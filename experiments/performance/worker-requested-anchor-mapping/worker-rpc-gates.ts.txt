import * as Effect from "effect/Effect"
import { commandIntentDigest } from "./src/client/worker-document/command-intent"
import { createWorkerSessionClient } from "./src/client/worker-document/worker-client"

type Worker = Effect.Success<ReturnType<typeof createWorkerSessionClient>>

export function verifyRequestedAnchors(
 worker: Worker,
 subject:string,document:string,
 source: string,
 peer: (source: string) => Effect.Effect<void, unknown>,
 check: (condition: unknown, message: string) => void,
) {
 return Effect.gen(function* () {
  if (!source.startsWith("A😀") || source.length<1536) throw Error("Dedicated Unicode512 seed required")
  let basis = yield* worker.Begin()
  check(basis.source === source, "real worker Begin exact seed")
  let ranges: {start:number;end:number;direction:"forward"|"backward"|"none"}[] = []
  for (let index=0;index<512;index++) {
   let start=(index*3)%source.length
   ranges.push({start,end:Math.min(source.length,start+1),direction:index%2?"backward":"forward"})
  }
  let unchanged = yield* worker.MergeCandidate({epoch:basis.epoch,sequence:basis.sequence,start:2,end:3,ranges})
  check(unchanged.source===source && unchanged.start===2 && unchanged.end===3, "stable Unicode intra-atom boundaries preserved")
  check(JSON.stringify(unchanged.ranges)===JSON.stringify(ranges), "all512 stable ranges and directions preserved")
  yield* worker.ReleaseCandidate({candidate:unchanged.candidate})
  yield* peer("PEER"+source)
  yield* worker.Synchronize()
  let inserted = yield* worker.MergeCandidate({epoch:basis.epoch,sequence:basis.sequence,start:2,end:3,ranges})
  check(inserted.source==="PEER"+source && inserted.start===6 && inserted.end===7, "remote prefix shifts Unicode intra-atom anchors exactly")
  let expected = ranges.map(range=>({...range,start:range.start?range.start+4:0,end:range.end?range.end+4:0}))
  check(JSON.stringify(inserted.ranges)===JSON.stringify(expected), "all512 remote ranges preserve offset-zero affinity and directions")
  yield* worker.Adopt({epoch:basis.epoch,sequence:basis.sequence,candidate:inserted.candidate})
  let adopted = yield* worker.EditorRead()
  if (adopted.epoch===undefined || adopted.sequence===undefined) throw Error("Missing adopted editor epoch")
  check(adopted.source===inserted.source, "Adopt keeps exact source/new epoch")
  let stale = yield* worker.MergeCandidate({epoch:basis.epoch,sequence:basis.sequence,start:0,end:0}).pipe(Effect.result)
  check(stale._tag==="Failure", "old epoch rejected without mutation")
  check((yield* worker.EditorRead()).source===adopted.source, "stale mapping leaves exact source unchanged")
  yield* peer("PEER"+source.slice(1))
  yield* worker.Synchronize()
  let deleted = yield* worker.MergeCandidate({epoch:adopted.epoch,sequence:adopted.sequence,start:5,end:5})
  check(deleted.source==="PEER"+source.slice(1) && deleted.start===4 && deleted.end===4, "removed anchor maps to exact surviving predecessor through original fallback")
  yield* worker.ReleaseCandidate({candidate:deleted.candidate})
  let input={epoch:adopted.epoch,sequence:adopted.sequence+1,admissionEpoch:0,ranges:[{start:adopted.source.length,remove:0,text:"x"}]}
  let intentDigest=yield* commandIntentDigest({subject,document,...input})
  let receipt=yield* worker.ReplaceReceipt({...input,intentDigest})
  check(receipt.agreement&&receipt.intentDigest===intentDigest,"authenticated queued receipt preserves exact intent")
  let insertedLocal=yield* worker.EditorRead()
  check(insertedLocal.source===adopted.source+"x", "local queued basis preserves its exact stale source")
  yield* worker.Synchronize()
  let undone=yield* worker.UndoQueued({request:crypto.randomUUID(),epoch:adopted.epoch,sequence:adopted.sequence+2,redo:false,admissionEpoch:0})
  check(undone.status==="applied", "real accepted local insertion Undo applies once")
  let missing=yield* worker.MergeCandidate({epoch:adopted.epoch,sequence:adopted.sequence+2,start:insertedLocal.source.length,end:insertedLocal.source.length})
  let final="PEER"+source.slice(1)
  check(missing.source===final && missing.start===final.length && missing.end===final.length, "local Undo deleted-anchor fallback preserves exact tail caret and peer deletion")
  yield* worker.ReleaseCandidate({candidate:missing.candidate})
  yield* worker.Synchronize()
  yield* peer("PEER"+source.slice(1020))
  yield* worker.Synchronize()
  let longDeleted=yield* worker.MergeCandidate({epoch:adopted.epoch,sequence:adopted.sequence+2,start:1004,end:1024})
  check(longDeleted.source==="PEER"+source.slice(1020) && longDeleted.start===4 && longDeleted.end===4, "long removed predecessor chain maps to baseline surviving prefix")
  yield* worker.ReleaseCandidate({candidate:longDeleted.candidate})
 })
}
