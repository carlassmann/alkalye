import * as Effect from "effect/Effect"
import { Replica as Baseline } from "./src/shared/crdt-baseline"
import { Replica as Candidate } from "./src/shared/crdt"
import type { ReplicaCheckpoint } from "./src/shared/checkpoint"
import type { Atom } from "./src/shared/model"
interface ReadReplica {
	forkEffect(actor: string): Effect.Effect<ReadReplica, unknown>
	edit(target: string, source: string): import("./src/shared/model").Operation | null
	restore(snapshot: ReplicaCheckpoint): void
	text(target: string): string
	visible(target: string): Atom[]
	atomTarget(id: string): string | undefined
}
let source = "---\ntitle: Index cost\n---\n" + "Body TOKEN 😀\n".repeat(90000)
source = source.slice(0, 1_048_576)
let { snapshot, initial } = seed()
function seed() {
	let replica = new Baseline("fixture-" + "x".repeat(48))
	let initial = replica.edit("document", source)
	if (!initial) throw Error("Missing seed")
	return { snapshot: replica.checkpoint(), initial }
}
function hash(value: string) {
	return new Bun.CryptoHasher("sha256").update(value).digest("hex")
}
let rows: {
	arm: string
	mode: string
	trial: number
	restoreMs: number
	textMs: number
	lookupMs: number
	forkMs: number
	forkHeapBefore: number
	forkHeapAfter: number
	forkRSSBefore: number
	forkRSSAfter: number
	heapBefore: number
	heapAfter: number
	rssBefore: number
	rssAfter: number
	atomCount: number
	sourceSHA: string
}[] = []
async function sample(
	arm: string,
	mode: string,
	trial: number,
	create: () => Promise<ReadReplica>,
) {
	Bun.gc(true)
	let before = process.memoryUsage()
	let start = performance.now()
	let replica = await create()
	let restored = performance.now()
	let content = replica.text("document")
	let textDone = performance.now()
	if (content !== source) throw Error("Exact source mismatch")
	let atoms = replica.visible("document")
	let ids: string[] = []
	for (let index = 0; index < 512; index++) {
		let atom = atoms[Math.floor((index * atoms.length) / 512)]
		if (!atom) throw Error("Missing sampled atom")
		ids.push(atom.id)
	}
	let lookupStart = performance.now()
	for (let repeat = 0; repeat < 20; repeat++)
		for (let id of ids)
			if (replica.atomTarget(id) !== "document")
				throw Error("Atom identity mismatch")
	let lookupDone = performance.now()
	let after = process.memoryUsage()
	let forkBefore = process.memoryUsage()
	let forkStart = performance.now()
	let forked = await Effect.runPromise(replica.forkEffect("fork-" + arm + "-" + mode + "-" + trial))
	let forkDone = performance.now()
	let forkAfter = process.memoryUsage()
	if (forked.text("document") !== content) throw Error("Fork source mismatch")
	for (let id of ids) if (forked.atomTarget(id) !== "document") throw Error("Fork identity mismatch")
	let forkEdit = forked.edit("document", content + "Fork 🦋")
	if (!forkEdit || !forkEdit.insert.length) throw Error("Missing fork edit")
	for (let atom of forkEdit.insert) if (replica.atomTarget(atom.id) !== undefined) throw Error("Fork identity leaked to owner")
	if (replica.text("document") !== content) throw Error("Fork edit changed owner")
	let ownerEdit = replica.edit("document", content + "Owner 🌻")
	if (!ownerEdit || !ownerEdit.insert.length) throw Error("Missing owner edit")
	for (let atom of ownerEdit.insert) if (forked.atomTarget(atom.id) !== undefined) throw Error("Owner identity leaked to fork")
	if (forked.text("document") !== content + "Fork 🦋") throw Error("Owner edit changed fork")
	rows.push({
		arm,
		mode,
		trial,
		restoreMs: restored - start,
		textMs: textDone - restored,
		lookupMs: lookupDone - lookupStart,
		forkMs: forkDone - forkStart,
		forkHeapBefore: forkBefore.heapUsed,
		forkHeapAfter: forkAfter.heapUsed,
		forkRSSBefore: forkBefore.rss,
		forkRSSAfter: forkAfter.rss,
		heapBefore: before.heapUsed,
		heapAfter: after.heapUsed,
		rssBefore: before.rss,
		rssAfter: after.rss,
		atomCount: atoms.length,
		sourceSHA: hash(content),
	})
	await Bun.write("../cost-results.json", JSON.stringify(rows, null, 2) + "\n")
}
for (let trial = 0; trial < 3; trial++) {
	for (let arm of trial % 2
		? ["candidate", "baseline"]
		: ["baseline", "candidate"]) {
		await sample(arm, "apply", trial, async () => {
			let replica =
				arm === "baseline" ? new Baseline("applied") : new Candidate("applied")
			replica.apply([initial])
			return replica
		})
		await sample(arm, "sync", trial, async () => {
			let replica: ReadReplica =
				arm === "baseline" ? new Baseline("cold") : new Candidate("cold")
			replica.restore(snapshot)
			return replica
		})
		await sample(arm, "cooperative", trial, () =>
			arm === "baseline"
				? Effect.runPromise(Baseline.fromCheckpointEffect(snapshot, "cold"))
				: Effect.runPromise(Candidate.fromCheckpointEffect(snapshot, "cold")),
		)
	}
}
console.log(JSON.stringify({ stage: "complete", samples: rows.length, exactForkIndependence: true }))
