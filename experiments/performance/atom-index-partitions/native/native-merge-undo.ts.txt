function wait(expression: string) {
	browser(session, "wait", "--fn", expression)

function accepted(id: string, source: string) {
	return Effect.runPromise(
		Effect.gen(function* () {
			while (true) {
				let value = yield* api.read(id)
				if (value.content === source) return
				yield* Effect.sleep("100 millis")
			}
		}).pipe(
			Effect.timeout("30 seconds"),
			Effect.provide(FetchHttpClient.layer),
		),
	)
}
function beginTrace(
	label: string,
	readyExpression = native + ".value===window.workerMCExpected",
) {
	browser(
		session,
		"profiler",
		"start",
		"--categories",
		"devtools.timeline,disabled-by-default-devtools.timeline,toplevel,blink.user_timing,v8",
	)
	tracing = true
	evaluate(
		session,
		`(()=>{let probe=window.workerMCProbe={label:${JSON.stringify(label)},start:0,ready:0,localSaved:0,end:0,events:[],heapBefore:performance.memory?.usedJSHeapSize},controller=new AbortController();for(let name of ['keydown','beforeinput','input'])document.addEventListener(name,event=>{if(!event.isTrusted||event.timeStamp===0)return;probe.events.push({type:name,inputType:event.inputType,key:event.key,time:performance.now()});if(!probe.start){probe.start=performance.now();performance.mark('worker-mc:start');setTimeout(()=>{probe.end=performance.now();probe.heapAfter=performance.memory?.usedJSHeapSize;performance.mark('worker-mc:end');controller.abort()},3000)}},{capture:true,signal:controller.signal});function frame(){if(probe.start&&!probe.ready&&${readyExpression}){probe.ready=performance.now();performance.mark('worker-mc:ready')}if(probe.ready&&!probe.localSaved&&document.querySelector('#save-status').getAttribute('aria-busy')==='false'&&document.querySelector('#save-status').dataset.pending==='0'){probe.localSaved=performance.now();performance.mark('worker-mc:local-saved')}if(!probe.end)requestAnimationFrame(frame)}requestAnimationFrame(frame);return true})()`,
	)
}
function finishTrace(path: string) {
	wait("window.workerMCProbe.end>0")
	browser(session, "profiler", "stop", path)
	tracing = false
	return evaluate(session, "window.workerMCProbe")
}

		wait(
