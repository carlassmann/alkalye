import {
	assert,
	browser,
	currentDocument,
	evaluate,
	wait,
} from "./browser-driver"

// Source reads go through the documented EditorElement API; #editor.value is
// the projection the native textarea shows.
let editor = 'document.querySelector("alkalye-editor").source'
let native = 'document.querySelector("#editor")'
let source = [
	"# Links",
	"Read [Example](https://example.org/a_(b)) and [[Welcome to Alkalye]].",
	"`[code](https://c.example)` ![Icon](/i.png)",
	"last",
].join("\n")
let projected = source
	.replace("[Example](https://example.org/a_(b))", "\u2003\u2060Example")
	.replace("[[Welcome to Alkalye]]", "\u2003\u2060Welcome to Alkalye")
	.replace("![Icon](/i.png)", "\u2003\u2060Icon")
let expanded = `${native}.value.includes("[Example](")`
let collapsed = `${native}.value.includes("Read \u2003\u2060Example and \u2003\u2060Welcome to Alkalye")`

let server = Bun.spawn(["bun", "tests/browser-document-app.ts", "--serve"], {
	stdout: "pipe",
	stderr: "inherit",
})
let session = "editor-prettification-" + Date.now()
try {
	let output = await server.stdout.getReader().read()
	if (!output.value) throw new Error("Missing app URL")
	let url = new URL("/app", new TextDecoder().decode(output.value).trim()).href
	console.log(url)
	browser(session, "set", "viewport", "1280", "800")
	browser(session, "open", url)
	wait(session, `Boolean(${native})`)
	let previousDocument = evaluate(session, currentDocument)
	browser(session, "click", '[data-action="new"]')
	wait(
		session,
		`${currentDocument} !== ${JSON.stringify(previousDocument)} && ${editor}.value === "" && !${native}.readOnly`,
	)
	browser(session, "fill", "#editor", "# Pointer cancellation")
	evaluate(
		session,
		`(() => { let input = ${native}; input.dispatchEvent(new PointerEvent("pointerdown", {pointerId:1,pointerType:"mouse",bubbles:true})); input.dispatchEvent(new PointerEvent("pointercancel", {pointerId:1,pointerType:"mouse",bubbles:true})); ${editor}.setSelectionRange(${editor}.value.length,${editor}.value.length); input.focus(); return true })()`,
	)
	browser(session, "keyboard", "inserttext", " resumes painting")
	wait(
		session,
		`${editor}.value === "# Pointer cancellation resumes painting" && document.querySelector("#highlight").textContent === ${native}.value + "\\n" && document.querySelector("#save-status").getAttribute("aria-busy") === "false"`,
	)
	assert(
		true,
		"Cancelled pointer drags resume native editing and syntax painting",
	)
	let carriageSource =
		"# CRLF\r\n\r\nRead [Example](https://example.org).\r\nLast 🦊"
	evaluate(
		session,
		`(() => { ${editor}.replace(${JSON.stringify(carriageSource)},${carriageSource.length},${carriageSource.length}); ${native}.dispatchEvent(new InputEvent("input",{bubbles:true,inputType:"insertText"})); ${native}.focus(); return true })()`,
	)
	wait(
		session,
		`${editor}.value === ${JSON.stringify(carriageSource)} && document.querySelector("#save-status").textContent === "Saved"`,
	)
	selectSource(carriageSource.length)
	browser(session, "keyboard", "inserttext", "x")
	wait(
		session,
		`${editor}.value === ${JSON.stringify(carriageSource + "x")} && ${editor}.selectionEnd === ${carriageSource.length + 1} && document.querySelector("#save-status").textContent === "Saved"`,
	)
	assert(
		evaluate(
			session,
			`!${native}.value.includes("\\r") && ${editor}.sourceIndex.value === ${JSON.stringify(carriageSource + "x")}`,
		),
		"CRLF edits preserve raw source and use normalized native text",
	)
	browser(session, "press", "Control+z")
	wait(
		session,
		`${editor}.value === ${JSON.stringify(carriageSource)} && document.querySelector("#save-status").textContent === "Saved"`,
	)
	browser(session, "reload")
	wait(
		session,
		`${editor}?.value === ${JSON.stringify(carriageSource)} && !${native}.readOnly`,
	)
	selectSource(carriageSource.length)
	assert(
		evaluate(session, `${editor}.selectionEnd === ${carriageSource.length}`),
		"CRLF Undo survives reload and restores the source EOF caret",
	)
	browser(session, "fill", "#editor", source)
	wait(session, `${editor}.value === ${JSON.stringify(source)}`)
	wait(session, `${native}.value === ${JSON.stringify(projected)}`)
	assert(
		evaluate(
			session,
			`document.querySelector("#highlight").textContent === ${native}.value + "\\n" && [...document.querySelectorAll("#highlight .syntax-link")].map(link => link.textContent).join("|") === "\u2003\u2060Example|\u2003\u2060Welcome to Alkalye|\u2003\u2060Icon"`,
		),
		"inactive links and wiki-links collapse in the textarea and its overlay alike",
	)

	wait(
		session,
		'document.querySelector("#save-status").textContent === "Saved"',
	)
	selectSource(2)
	browser(session, "keyboard", "inserttext", "x")
	wait(session, `${editor}.value === ${JSON.stringify(source.slice(0, 2) + "x" + source.slice(2))} && document.querySelector("#save-status").textContent === "Saved"`)
	browser(session, "press", "Control+z")

	wait(session, `${editor}.value === ${JSON.stringify(source)} && ${editor}.selectionStart === 3 && ${editor}.selectionEnd === 3`)

	wait(session, collapsed)

		session,

		evaluate(session, collapsed),
		"saving and reopening keeps the Markdown source, not the projection",

		session,

		evaluate(

		})()`,

		session,

		true,

	assert(
		evaluate(
			session,

		),
		"repainting compact links preserves keyboard focus",

	selectSource(0)
	assert(
		evaluate(
			session,
			`(() => { let anchor = document.querySelector('[data-editor-links] a[data-kind="link"]'); let rect = anchor.getBoundingClientRect(); anchor.dispatchEvent(new PointerEvent("pointerdown", { pointerType:"mouse", shiftKey:true, bubbles:true, cancelable:true, clientX:rect.left + 3, clientY:rect.top + 3 })); let cancelled = !anchor.dispatchEvent(new MouseEvent("click", { shiftKey:true, bubbles:true, cancelable:true })); return cancelled && ${editor}.selectionStart === 0 && ${editor}.selectionEnd > 0 })()`,

	assert(

	wait(session, collapsed)

		session,

	assert(
		evaluate(
			session,
			`${expanded} && ${editor}.value.slice(${editor}.selectionStart, ${editor}.selectionEnd) === "example.org" && ${native}.value.slice(${native}.selectionStart, ${native}.selectionEnd) === "example.org"`,

		"source selections inside hidden destinations reveal the link",
	)

	let last = source.indexOf("last")
	selectSource(last)
	wait(session, collapsed)
	browser(session, "press", "ArrowUp")
	browser(session, "press", "ArrowUp")
	assert(
		evaluate(session, collapsed),
		"links stay compact when the cursor is elsewhere on their line",
	)
	selectSource(source.indexOf("[Example]") - 1)
	browser(session, "press", "ArrowRight")
	browser(session, "press", "ArrowRight")

	assert(
		evaluate(
			session,

	browser(session, "keyboard", "type", " more")

	let lineEnd = source.indexOf("\n`")

	wait(
		session,
		`${native}.value.includes("Read \u2003\u2060Example and \u2003\u2060Welcome to Alkalye.!")`,

	assert(true, "edits on a raw link line are kept when it collapses")
	undoUntil(source)

	selectSource(last)
	wait(session, collapsed)
	evaluate(
		session,
		`${native}.dispatchEvent(new CompositionEvent("compositionstart", {bubbles: true})); ${native}.setSelectionRange(${native}.value.indexOf("Example") + 2, ${native}.value.indexOf("Example") + 2)`,
	)

	assert(
		evaluate(session, collapsed),
		"composition keeps the textarea unchanged",
	)
	evaluate(
		session,
		`${native}.dispatchEvent(new CompositionEvent("compositionend", {bubbles: true, data: ""}))`,
	)
	wait(session, expanded)
	assert(
		evaluate(
			session,
			`${editor}.selectionStart === ${source.indexOf("Example") + 2}`,
		),
		"composition end reprojects around the caret's source offset",
	)

	selectSource(last)
	evaluate(
		session,
		`{
		${native}.dispatchEvent(new CompositionEvent("compositionstart", {bubbles:true}));
		let from = ${native}.value.indexOf("last");

		true,
		"composition end accepts final native text before the final input event",
	)
	undoUntil(source)

	selectSource(last)
	wait(session, collapsed)
	evaluate(
		session,
		`${native}.dispatchEvent(new PointerEvent("pointerdown", {bubbles: true})); ${native}.setSelectionRange(0, ${native}.value.indexOf("last"))`,
	)
	browser(session, "wait", "300")
	assert(evaluate(session, collapsed), "a pointer drag keeps the text under it")
	assert(
		evaluate(
			session,
			`(() => { let clipboardData = new DataTransfer(); let event = new ClipboardEvent("copy", {bubbles: true, cancelable: true, clipboardData}); ${native}.dispatchEvent(event); return event.defaultPrevented && clipboardData.getData("text/plain") === ${JSON.stringify(source.slice(0, last))} })()`,

	wait(session, collapsed)

	selectSource(last)
	wait(session, collapsed)
	browser(session, "press", "Control+Alt+ArrowUp")

		assert(
			true,
			"Deleting synthetic icon space preserves valid source across a durable reload",
		)
	}
	browser(session, "set", "viewport", "1280", "800")
	browser(session, "click", '[data-action="library"]')
	browser(session, "click", '[data-action="new"]')
	let large =
		"# Large note\n\n" +
		"Paragraph **bold** [Example](https://example.org) and ==marker==.\n\n".repeat(
			2048,
		)
	evaluate(
		session,
		`(() => { ${editor}.value = ${JSON.stringify(large)}; ${native}.dispatchEvent(new Event("input", { bubbles:true })); return true })()`,
	)
	wait(
		session,
		`${editor}.value === ${JSON.stringify(large)} && document.querySelector("#save-status").getAttribute("aria-busy") === "false"`,
	)
	selectSource(large.length)
	browser(session, "keyboard", "inserttext", "Last paragraph.")
	wait(
		session,
		`${editor}.value.endsWith("Last paragraph.") && document.querySelector("#save-status").getAttribute("aria-busy") === "false"`,
	)
	assertViewportGeometry(
		"Large wrapped documents keep native input and visible syntax aligned",
	)
	browser(session, "reload")
	wait(
		session,
		`${editor}.value === ${JSON.stringify(large + "Last paragraph.")} && !${native}.readOnly`,
	)
	selectSource(large.indexOf("Paragraph"))
	browser(session, "keyboard", "inserttext", "Introduction.\n\n")
	wait(
		session,
		`${editor}.value.includes("Introduction.\\n\\nParagraph") && document.querySelector("#save-status").getAttribute("aria-busy") === "false"`,
	)
	assertViewportGeometry(
		"Inserting lines before thousands of links preserves visible offsets and wrapping",
	)
	selectSource(0)
	evaluate(
		session,
		`(() => { ${native}.parentElement.scrollTop = 0; ${native}.parentElement.dispatchEvent(new Event("scroll")); return true })()`,
	)
	wait(
		session,
		`Boolean(document.querySelector('[data-editor-links] a[data-kind="link"]'))`,
	)
	let focusedLink = evaluate(
		session,
		`(() => { let anchor = document.querySelector('[data-editor-links] a[data-kind="link"]'); anchor.focus(); let source = anchor.dataset.editorLink; ${native}.parentElement.scrollTop = ${native}.parentElement.scrollHeight; ${native}.parentElement.dispatchEvent(new Event("scroll")); return source })()`,
	)
	wait(
		session,
		`document.activeElement?.dataset.editorLink === ${JSON.stringify(focusedLink)} && document.activeElement.style.left === "-10000px"`,
	)
	browser(session, "press", "Tab")
	wait(
		session,
		`document.activeElement?.tagName === "A" && document.activeElement.dataset.editorLink !== ${JSON.stringify(focusedLink)} && document.activeElement.getBoundingClientRect().top > 0 && document.activeElement.getBoundingClientRect().bottom < window.innerHeight`,
	)
	assert(
		true,
		"Focused links keep their destination while scrolling, and Tab reaches offscreen links",
	)
	let wrappedLink =
		"# Wrapped link\n\n[" +
		"English עברית العربية words ".repeat(14) +
		"](https://example.org/wrapped)\n"
	browser(session, "set", "viewport", "600", "800")
	evaluate(
		session,
		`(()=>{let app=document.querySelector('#app');if(app.classList.contains('library-open'))document.querySelector('[data-action=library]').click();if(app.classList.contains('tools-open'))document.querySelector('[data-action=toggle-tools]').click();${native}.focus()})();${editor}.value=${JSON.stringify(wrappedLink)};${editor}.dispatchEvent(new Event('input',{bubbles:true}));${editor}.setSelectionRange(0,0);${native}.parentElement.scrollTop=0`,
	)
	wait(
		session,
		`${editor}.value===${JSON.stringify(wrappedLink)} && document.querySelector('#save-status').textContent==='Saved' && document.querySelectorAll('[data-editor-links] a[data-kind=link]').length>1`,
	)
	assert(
		evaluate(
			session,
			`(()=>{let anchors=[...document.querySelectorAll('[data-editor-links] a[data-kind=link]')];return anchors.filter(anchor=>anchor.tabIndex===0).length===1 && anchors.every(anchor=>{let bounds=anchor.getBoundingClientRect();return document.elementFromPoint(bounds.x+bounds.width/2,bounds.y+bounds.height/2)===anchor})})()`,
		),
		"Wrapped bidirectional links retain separate actionable fragments and one keyboard stop",
	)
} finally {
	try {
		browser(session, "close")
	} finally {
		server.kill()
		await server.exited
	}
}

function assertViewportGeometry(label: string) {
	let geometry = `(() => {
		let input = ${native}, overlay = document.querySelector("#highlight"), outer = input.parentElement
		let prefix = /^[\\u200e\\u200f]/.test(input.value) ? 1 : 0
		window.viewportGeometryFailure = undefined
		if (overlay.textContent !== input.value.slice(prefix) + "\\n" || input.scrollTop !== 0 || input.value.length >= ${editor}.value.length || outer.scrollHeight <= outer.clientHeight) { window.viewportGeometryFailure = {textMatches:overlay.textContent === input.value.slice(prefix)+"\\n",nativeScroll:input.scrollTop,nativeLength:input.value.length,sourceLength:${editor}.value.length,outerHeight:outer.scrollHeight,outerClient:outer.clientHeight}; return false }
		let mirror = document.createElement("pre"), style = getComputedStyle(input), bounds = input.getBoundingClientRect()
		for (let property of ["font-family", "font-size", "font-weight", "font-style", "line-height", "letter-spacing", "tab-size", "white-space", "overflow-wrap", "word-break", "unicode-bidi", "direction", "text-align", "padding-top", "padding-right", "padding-bottom", "padding-left", "border-top-width", "border-right-width", "border-bottom-width", "border-left-width", "box-sizing"]) mirror.style.setProperty(property, style.getPropertyValue(property))
		Object.assign(mirror.style, {position:"fixed", left:bounds.left+"px", top:bounds.top+"px", width:bounds.width+"px", margin:"0", visibility:"hidden", pointerEvents:"none"})
		mirror.textContent = input.value + "\\u200b"
		document.body.append(mirror)
		try {
			let walker = document.createTreeWalker(overlay, NodeFilter.SHOW_TEXT), nodes = [], offset = 0, node
			while (node = walker.nextNode()) { nodes.push({node,start:offset,end:offset+node.textContent.length}); offset += node.textContent.length }
			let checked = 0, rows = new Map(), host = outer.getBoundingClientRect()
			for (let entry of nodes) for (let local = 0; local < entry.node.textContent.length; local++) {
				let character = entry.node.textContent[local]
				if (!/[A-Za-z]/.test(character)) continue
				let actual = document.createRange()
				actual.setStart(entry.node, local); actual.setEnd(entry.node, local + 1)
				let rect = actual.getBoundingClientRect()
				if (rect.top < host.top || rect.bottom > host.bottom || rect.width === 0) continue
				let row = Math.round(rect.top), samples = rows.get(row) ?? 0
				if (samples >= 3) continue
				rows.set(row, samples + 1)
				let expected = document.createRange(), position = prefix + entry.start + local
				expected.setStart(mirror.firstChild, position); expected.setEnd(mirror.firstChild, position + 1)
				let nativeRect = expected.getBoundingClientRect()
				if (Math.abs(rect.left-nativeRect.left)>1 || Math.abs(rect.top-nativeRect.top)>1) {window.viewportGeometryFailure={position,character,overlayX:rect.left,overlayY:rect.top,nativeX:nativeRect.left,nativeY:nativeRect.top,inputFont:style.font,inputLine:style.lineHeight,overlayFont:getComputedStyle(overlay).font,overlayLine:getComputedStyle(overlay).lineHeight,inputBounds:{top:bounds.top,width:bounds.width},overlayBounds:{top:overlay.getBoundingClientRect().top,width:overlay.getBoundingClientRect().width},lines:[...overlay.children].slice(0,5).map(line=>({height:line.getBoundingClientRect().height,font:getComputedStyle(line).font,lineHeight:getComputedStyle(line).lineHeight}))}; return false}
				checked++
				if (checked === 32) return true
			}
			window.viewportGeometryFailure = {checked,rows:rows.size}
			return checked >= 3 && rows.size >= 2
		} finally {mirror.remove()}
	})()`
	try {
		wait(session, geometry)
	} catch (error) {
		console.error(evaluate(session, "window.viewportGeometryFailure"))
		throw error
	}
	assert(evaluate(session, geometry), label)
}

function selectSource(offset: number) {
	evaluate(
		session,
		`(() => { if (${native}.readOnly) throw new Error("Document session is not ready for editing"); ${native}.focus(); ${editor}.setSelectionRange(${offset}, ${offset}); return true })()`,
	)
}
function undoUntil(expected: string) {
	for (let attempt = 0; attempt < 10; attempt++) {
		if (evaluate(session, `${editor}.value === ${JSON.stringify(expected)}`))
			return
		browser(session, "press", "Control+z")
		browser(session, "wait", "100")
	}
	wait(session, `${editor}.value === ${JSON.stringify(expected)}`)
}
