import json,pathlib,collections,hashlib
p=pathlib.Path('/tmp/alkalye-rich-current');events=json.loads((p/'sampled-input.trace.json').read_text())['traceEvents'];m={e['name']:e for e in events if e.get('name') in ['rich:start','rich:end']};a,b=m['rich:start'],m['rich:end'];thread=a['pid'],a['tid'];tasks=[e for e in events if e.get('name')=='RunTask' and e.get('ph')=='X' and (e.get('pid'),e.get('tid'))==thread and e['ts']<=b['ts'] and e['ts']+e.get('dur',0)>=a['ts']];initial=max((e for e in tasks if e['ts']<=a['ts']<=e['ts']+e.get('dur',0)),key=lambda e:e.get('dur',0));start,end=initial['ts'],initial['ts']+initial.get('dur',0)
groups={}
for e in sorted(events,key=lambda e:e.get('ts',0)):
 if e.get('name') not in ['Profile','ProfileChunk']:continue

 if e['name']=='Profile':g['time']=d.get('startTime',e['ts']);continue

  g['self'][ident]+=delta;seen=set();ancestor=ident

   seen.add(ancestor);g['inclusive'][ancestor]+=delta;ancestor=g['nodes'].get(ancestor,{}).get('parent')
profiles=[]
for (pid,_),g in groups.items():
 if not g['self']:continue
 profiles.append({'main':(pid,g['tid'])==thread,'sampledInputSelfMs':sum(g['self'].values())/1000,'self':[{'frame':g['nodes'].get(i,{}).get('callFrame',{}),'selfMs':v/1000,'inclusiveMs':g['inclusive'][i]/1000} for i,v in g['self'].most_common(25)],'inclusive':[{'frame':g['nodes'].get(i,{}).get('callFrame',{}),'selfMs':g['self'][i]/1000,'inclusiveMs':v/1000} for i,v in g['inclusive'].most_common(30)]})
r={'cpuSampling':True,'inputRunTaskMs':initial.get('dur',0)/1000,'allTaskMaxMs':max(e.get('dur',0)/1000 for e in tasks),'deferredMaxMs':max(e.get('dur',0)/1000 for e in tasks if e is not initial),'profiles':profiles,'traceSha256':hashlib.sha256((p/'sampled-input.trace.json').read_bytes()).hexdigest()};(p/'sampled-analysis.json').write_text(json.dumps(r,indent=2)+'\n');print(json.dumps(r))
