import * as Effect from "effect/Effect"
import { startServer } from "../control-runtime/src/server/index"
let data = Bun.env.CHILDREN_FIXTURE_DATA
if (!data?.startsWith("/tmp/alkalye-production-visibility-memo/")) throw Error("Owned artificial fixture data required")
let instance = await Effect.runPromise(startServer({port:4498,hostname:"127.0.0.1",publicOrigin:"http://127.0.0.1:4498",data,assets:import.meta.dir+"/dist",development:false,build:false}))
console.log(JSON.stringify({url:instance.server.url.href}))
for (let signal of ["SIGINT","SIGTERM"]) process.once(signal,async()=>{await Effect.runPromise(instance.close());process.exit(0)})
