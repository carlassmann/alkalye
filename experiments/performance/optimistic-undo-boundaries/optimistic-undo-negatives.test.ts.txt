import { expect, test } from "bun:test"
import { mkdtemp, rm } from "node:fs/promises"
import { join } from "node:path"
import { tmpdir } from "node:os"
import * as Effect from "effect/Effect"
import * as Context from "effect/Context"
import * as Layer from "effect/Layer"
import * as Fiber from "effect/Fiber"
import * as Exit from "effect/Exit"
import * as Scope from "effect/Scope"
import * as Deferred from "effect/Deferred"
import * as EventJournal from "effect/eventlog/EventJournal"
import { DocumentJournal } from "../src/platform/document-journal"
import type { DocumentSessionOptions } from "../src/shared/document-session"
import {
	cliDocumentSessionLayer,
	cliDocumentPlatform,
} from "../src/cli/document-platform"
import {
	DocumentSession,
	documentDatabaseName,
} from "../src/shared/document-session"
import type { DocumentTransport } from "../src/shared/document-session"
import { WorkspaceStore } from "../src/server/document-store"
import { Identity, personalWorkspaceRequest } from "../src/shared/workspace"
import { Replica } from "../src/shared/crdt"
import {
	buildAction,
	ResourceUnavailable,
	restoreSnapshot,
} from "../src/shared/document-actions"
import { newDocument } from "../src/shared/documents"

test.each(["interrupted", "peer", "storage", "acl", "close"])(
	"optimistic Undo preserves later typing across interruption and accepted peer insertion (%j)",
	async phase => {
		let directory = await mkdtemp(
			join(tmpdir(), "alkalye-journal-interruption-"),
		)
		let source =
			'---\ntitle: "Replacement"\n---\n' +
			"Body 😀\n".repeat(4) +
			"TOKEN suffix 😀\n"
		let offsets: number[] = []
		for (
			let offset = source.indexOf("TOKEN");
			offset !== -1;
			offset = source.indexOf("TOKEN", offset + 5)
		)
			offsets.push(offset)
		let expected = source.replaceAll("TOKEN", "xTOKEN")
		let failStorage = false
		let completed = false
		let armed = false
		let ready = await Effect.runPromise(Deferred.make<void>())
		let release = await Effect.runPromise(Deferred.make<void>())
		function controlledSessionLayer(
			directory: string,
			options: DocumentSessionOptions,
		) {
			let platform = cliDocumentPlatform(directory)
			let controlled = Layer.effectContext(
				Effect.gen(function* () {
					let storage = yield* DocumentJournal
					let journal = EventJournal.EventJournal.of({
						...storage.journal,
						write: options => Effect.suspend(() => failStorage ? Effect.fail(new EventJournal.EventJournalError({ method: "write", cause: new Error("Injected pre-write storage fault") })) : storage.journal.write(options)),
						withLock: store => effect => {
							let boundary = Effect.suspend(() => {
								if (!armed) return Effect.void
								armed = false
								return Deferred.succeed(ready, undefined).pipe(
									Effect.andThen(Deferred.await(release)),
								)
							})
							let admitted = boundary.pipe(Effect.andThen(storage.journal.withLock(store)(effect)))
							return admitted
						},
					})
					return Context.make(DocumentJournal, { ...storage, journal }).pipe(
						Context.add(EventJournal.EventJournal, journal),
					)
				}),
			).pipe(Layer.provide(platform.journal(documentDatabaseName(options))))
			return DocumentSession.layer(options).pipe(
				Layer.provide(controlled),
				Layer.provide(platform.lifecycle),
			)
		}
		try {
			await Effect.runPromise(
				Effect.scoped(
					Effect.gen(function* () {
						let services = yield* Layer.build(
							WorkspaceStore.layer(join(directory, "server.sqlite")),
						)
						let store = Context.get(services, WorkspaceStore)
						let subject = "replacement-writer"
						let workspaceOwner = "workspace-owner"
						let creation = yield* buildAction(
							new Replica("birth"),
							"document",
							"Seed",
							staged => newDocument(staged, "document", "Replacement", source),
						)
						yield* store
							.create("workspace", "Workspace", personalWorkspaceRequest)
							.pipe(Effect.provideService(Identity, { subject: workspaceOwner, actors: [] }))
						yield* store.createDocument("workspace", creation).pipe(
							Effect.provideService(Identity, {
								subject: workspaceOwner,
								actors: [creation.actor],
							}),
						)
						yield* store.membership("workspace", subject, "writer", "grant-writer").pipe(Effect.provideService(Identity, { subject: workspaceOwner, actors: [] }))
						function owner<A, E>(effect: Effect.Effect<A, E, Identity>) {
							return effect.pipe(
								Effect.provideService(Identity, { subject, actors: [] }),
							)
						}
						let transport: DocumentTransport = {
							snapshot: document =>
								owner(store.snapshot(document)).pipe(
									Effect.flatMap(value =>
										value
											? Effect.succeed(value)
											: Effect.fail(
													new ResourceUnavailable({
														message: "Missing snapshot",
													}),
												),
									),
								),
							commit: action =>
								store.commit(action).pipe(
									Effect.provideService(Identity, {
										subject,
										actors: [action.actor],
									}),
								),
							history: (document, after, log, limit) =>
								owner(store.history(document, after, log, limit)),
							action: (document, id) =>
								owner(store.action(document, id)).pipe(
									Effect.flatMap(value =>
										value
											? Effect.succeed(value)
											: Effect.fail(
													new ResourceUnavailable({
														message: "Missing action",
													}),
												),
									),
								),
							chain: (document, id) =>
								owner(store.chain(document, id)).pipe(
									Effect.flatMap(value =>
										value
											? Effect.succeed(value)
											: Effect.fail(
													new ResourceUnavailable({ message: "Missing chain" }),
												),
									),
								),
						}
						let sessionScope = yield* Scope.make()
						yield* Effect.addFinalizer(() => Scope.close(sessionScope, Exit.void))
						let sessionServices = yield* Layer.buildWithScope(
							controlledSessionLayer(join(directory, "client"), {
								document: "document",
								subject,
								author: "Writer",
								transport,
								cooperativeWork: true,
							}),
							sessionScope,
						)
						let session = Context.get(sessionServices, DocumentSession)
						expect(session.replica.text("document")).toBe(source)
						let basis = yield* session.replica.forkEffect()
						basis.text("document")
						let edited = session.replace(
							offsets.map(start => ({ start, remove: 0, text: "x" })),
							basis.visible("document"),
						)
						yield* edited.saved
						let editId = session.pending()[0]?.id
						expect(editId).toBeDefined()
						yield* session.sync
						expect(session.pending()).toHaveLength(0)
						expect(session.replica.text("document")).toBe(expected)
						let editSnapshot = yield* transport.snapshot("document")
						let acceptedEdit = yield* restoreSnapshot(
							editSnapshot,
							"independent-edit",
						)
						expect(acceptedEdit.text("document")).toBe(expected)
						armed = true
						let pending = yield* session.undo().pipe(Effect.forkScoped)
						yield* Deferred.await(ready).pipe(Effect.timeout("5 seconds"))
						expect(session.replica.text("document")).toBe(source)
						let later = session.replace([{ start: source.length, remove: 0, text: "later 😀\n" }])
						expect(session.replica.text("document")).toBe(source + "later 😀\n")
						let finalSource = source + "later 😀\n"
						if (phase === "peer") {
							yield* store.membership("workspace", "peer", "writer", "grant-peer").pipe(Effect.provideService(Identity, { subject: workspaceOwner, actors: [] }))
							let peerBasis = yield* restoreSnapshot(editSnapshot, "peer-fixture")
							let peerAction = yield* buildAction(peerBasis, "document", "Peer insertion", staged => [staged.edit("document", expected.replace("Body", "peer 😀 Body"))])
							yield* store.commit(peerAction).pipe(Effect.provideService(Identity, { subject: "peer", actors: [peerAction.actor] }))
							let siblingServices = yield* Layer.build(cliDocumentSessionLayer(join(directory, "client"), {
								document: "document", subject, author: "Writer", transport, cooperativeWork: true,
							}))
							yield* Context.get(siblingServices, DocumentSession).sync
							finalSource = source.replace("Body", "peer 😀 Body") + "later 😀\n"
						}
						if (phase === "interrupted") {
							yield* Fiber.interrupt(pending)
							let interrupted = yield* Fiber.await(pending)
							expect(Exit.hasInterrupts(interrupted)).toBe(true)
							expect(session.replica.text("document")).toBe(finalSource)
						}
						if (phase === "acl") yield* store.membership("workspace", subject, "reader", "revoke-writer").pipe(Effect.provideService(Identity, { subject: workspaceOwner, actors: [] }))
						if (phase === "storage") failStorage = true
						let closing = phase === "close" ? yield* Scope.close(sessionScope, Exit.void).pipe(Effect.forkScoped) : undefined
						yield* Deferred.succeed(release, undefined)
						if (phase === "storage") {
							let failed = yield* Fiber.join(pending).pipe(Effect.result)
							expect(failed._tag).toBe("Failure")
							expect(session.replica.text("document")).toBe(finalSource)
							let unchanged = yield* restoreSnapshot(yield* transport.snapshot("document"), "storage-independent")
							expect(unchanged.text("document")).toBe(expected)
							failStorage = false
							yield* session.flush
						} else if (phase === "interrupted") yield* session.flush
						else {
							let undo = yield* Fiber.join(pending)
							expect(undo.undoOf).toBe(editId)
						}
						yield* later.saved
						if (closing) yield* Fiber.join(closing)
						if (phase === "acl") {
							let queuedIds = session.pending().map(action => action.id)
							expect(queuedIds).toHaveLength(2)
							let denied = yield* session.sync.pipe(Effect.result)
							expect(denied._tag).toBe("Failure")
							expect(session.pending().map(action => action.id)).toEqual(queuedIds)
							expect(session.replica.text("document")).toBe(finalSource)
							let unchanged = yield* restoreSnapshot(yield* transport.snapshot("document"), "acl-independent")
							expect(unchanged.text("document")).toBe(expected)
							yield* store.membership("workspace", subject, "writer", "restore-writer").pipe(Effect.provideService(Identity, { subject: workspaceOwner, actors: [] }))
						}
						if (phase === "close") {
							let reopenedServices = yield* Layer.build(cliDocumentSessionLayer(join(directory, "client"), { document: "document", subject, author: "Writer", transport, cooperativeWork: true }))
							yield* Context.get(reopenedServices, DocumentSession).sync
						} else yield* session.sync
						let accepted = yield* transport.snapshot("document")
						let reconstructed = yield* restoreSnapshot(accepted, "independent-optimistic-undo")
						expect(reconstructed.text("document")).toBe(finalSource)
						expect(session.replica.text("document")).toBe(finalSource)
						expect(accepted.position).toBe(editSnapshot.position + (phase === "peer" ? 3 : 2))
						let cold = yield* Effect.scoped(
							DocumentSession.use(reopened =>
								Effect.succeed(reopened.replica.text("document")),
							).pipe(
								Effect.provide(
									cliDocumentSessionLayer(join(directory, "client"), {
										document: "document",
										subject,
										author: "Writer",
										transport,
										cooperativeWork: true,
									}),
								),
							),
						)
						expect(cold).toBe(finalSource)
					}),
				),
			)
			completed = true
		} finally {
			if (completed) await rm(directory, { recursive: true, force: true })
			else console.error("Retained optimistic Undo failure fixture", directory)
		}
	},
)
