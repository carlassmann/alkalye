function coordinate(raw: number) {

 let remaining=source.toView(raw)-source.viewport.viewport.start
 if(remaining<0)throw Error("Expected coordinate outside painted window")
 let walker=document.createTreeWalker(highlight,NodeFilter.SHOW_TEXT)
 for(let node=walker.nextNode();node;node=walker.nextNode()){
  let size=node.textContent?.length??0
  if(remaining<size){
   let range=document.createRange()
   range.setStart(node,remaining);range.setEnd(node,remaining+1)
   let bounds=range.getBoundingClientRect()
   if(!bounds.width || !bounds.height)throw Error("Expected glyph not rendered")
   let direction=node.parentElement?getComputedStyle(node.parentElement).direction:"ltr"
   return {raw,x:direction==="rtl"?bounds.right-bounds.width*.2:bounds.left+bounds.width*.2,y:bounds.top+bounds.height*.5,bounds:bounds.toJSON(),direction}
  }
  remaining-=size
 }
 throw Error("No public rendered source point")
}
