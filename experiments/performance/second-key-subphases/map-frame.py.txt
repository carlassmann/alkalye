root=pathlib.Path('/tmp/alkalye-second-key-subphases')
chars='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'
def vlq(s):
 out=[]; value=shift=0
 for c in s:
  n=chars.index(c); value|=(n&31)<<shift
  if n&32: shift+=5
  else:
   out.append(-(value>>1) if value&1 else value>>1); value=shift=0
 return out
