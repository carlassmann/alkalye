import json,statistics,sys
from pathlib import Path
root=Path(__file__).parent
name=sys.argv[1] if len(sys.argv)>1 else 'cost-results.json'
rows=json.loads((root/name).read_text());result=[]
keys=['loadMs','textMs','lookupMs','checkpointMs','durableCheckpointMs','checkpointBytes','durableCheckpointBytes','forkMs','firstForkWriteMs','firstOwnerWriteMs','heapBefore','heapAfterLoad','heapAfterFork','rssBefore','rssAfterLoad','rssAfterFork']
for fixture in dict.fromkeys(r['fixture'] for r in rows):
 for mode in ['apply','sync','cooperative']:
  for arm in ['baseline','candidate']:
   group=[r for r in rows if r['fixture']==fixture and r['mode']==mode and r['arm']==arm];entry={'fixture':fixture,'mode':mode,'arm':arm,'n':len(group),'metrics':{}}
   for key in keys:
    vals=[r[key] for r in group];entry['metrics'][key]={'median':statistics.median(vals),'min':min(vals),'max':max(vals)}
   result.append(entry)
(root/name.replace('results','summary')).write_text(json.dumps({'samples':len(rows),'summary':result},indent=2)+'\n')
for f in dict.fromkeys(r['fixture'] for r in rows):
 for mode in ['apply','sync','cooperative']:
  pair=[r for r in result if r['fixture']==f and r['mode']==mode];print(f,mode,{k:[round(r['metrics'][k]['median'],3) for r in pair] for k in keys[:10]})
