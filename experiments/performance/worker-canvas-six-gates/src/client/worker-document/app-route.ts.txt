import { frontmatter } from "../../shared/markdown"
import * as Fiber from "effect/Fiber"
import * as Semaphore from "effect/Semaphore"
import * as Option from "effect/Option"
import { attachmentsMarkup } from "../components/attachments"
import { openMediaPicker } from "../components/media-picker"
import type { MediaChoice } from "../components/media-picker"
import { openWorkerDrawing } from "./worker-drawing"
import { makeWorkerMedia } from "./worker-media"
import { downloadFile } from "../document-archives"
import { DocumentBlobs } from "../document-blobs"
import { commentsMarkup } from "../components/comments"
import { openDialog, mediaPreviewDialog, confirmDialog } from "../components/dialog"
import { setMarkup } from "../dom"
import { html } from "../language"
import type { OpenedThreadSchema } from "./worker-opened-schemas"
import { workerBootMark } from "./boot-diagnostics"
import * as Effect from "effect/Effect"
import * as FiberSet from "effect/FiberSet"
import * as Stream from "effect/Stream"
import { BrowserLibrary } from "../browser-library"
import { EditorElement } from "../components/editor"
import { openFind } from "../components/find"
import type { EditorPreferences } from "../../shared/preferences"
import type { DocumentLink } from "../../shared/model"
import { workerEditorBindings } from "./bindings"

export function mountWorkerAppRoute(
	host: HTMLElement,
	document: string,
	options: {
		name: string
		preferences: EditorPreferences
		documents(): readonly DocumentLink[]
		openDocument(id: string): void
		report(error: unknown): void
		title(title: string): void
		status(pending: number, unsaved: number, failed: boolean): void
		findBar: HTMLElement
  dialog: HTMLDialogElement
	},
) {
	return Effect.gen(function* () {
		let library = yield* BrowserLibrary
		workerBootMark("route:open-start")
		let opened = yield* library
			.openWorker(document)
			.pipe(
				Effect.tapError(error =>
					Effect.sync(() => workerBootMark("route:open-error", error)),
				),
			)
		workerBootMark("route:open-done")
		let owner = library.workerOwner(document)
		if (!owner)
			return yield* Effect.fail(new Error("Missing worker native owner"))
		let initialized = yield* Effect.gen(function* () {
			if (owner.recovery() !== undefined) return undefined
			return yield* opened.queue()
		}).pipe(
			Effect.catch(error =>
				owner.recovery() !== undefined
					? Effect.succeed(undefined)
					: Effect.fail(error),
			),
		)
		let tasks = yield* FiberSet.make()
		let fork = yield* FiberSet.runtime(tasks)()
		let captures = yield* FiberSet.make()
		let forkCapture = yield* FiberSet.runtime(captures)()
		if (!initialized) {
			let recovery = owner.recovery()
			if (recovery === undefined)
				return yield* Effect.fail(new Error("Recovery source unavailable"))
			workerBootMark("route:Editor-start")
			let editor = new EditorElement()
			function unsupported() {
				options.report(
					"Retained recovery is read-only; download before manual reconciliation",
				)
			}
			editor.configure({
				authority: "worker",
				content: recovery,
				atoms: [],
				editable: false,
				preferences: options.preferences,
				documents: options.documents,
				openDocument: options.openDocument,
				isDrawing: () => false,
				openDrawing: unsupported,
				createLinkedDocument: async () => {
					throw Error("Recovery is read-only")
				},
				canComment: () => false,
				canAttach: () => false,
				comments: () => [],
				report: options.report,
				change: () => ({ kind: "pending" }),
				undo: unsupported,
				select() {},
				requestAsset: unsupported,
				attach: unsupported,
				command: unsupported,
				find: unsupported,
			})
			let download = globalThis.document.createElement("button")
			download.textContent = "Download retained draft"
			download.addEventListener("click", () => {
				let url = URL.createObjectURL(
					new Blob([recovery], { type: "text/markdown" }),
				)
				let link = globalThis.document.createElement("a")
				link.href = url
				link.download = "retained-draft.md"
				link.click()
				URL.revokeObjectURL(url)
			})
   let commands = globalThis.document.createElement("button")
   commands.textContent = "Download retained commands"
   commands.addEventListener("click", () => {
    fork(opened.exportDomainRecords.pipe(Effect.tap(contents => Effect.sync(() => {
     let url = URL.createObjectURL(new Blob([contents], { type: "application/json" }))
     let link = globalThis.document.createElement("a")
     link.href = url; link.download = "retained-commands.json"; link.click(); URL.revokeObjectURL(url)
    })), Effect.catch(error => Effect.sync(() => { workerBootMark("comments:error", error instanceof Error ? error.name + ":" + error.message : String(error)); options.report(error) }))))
   })
			host.replaceChildren(editor, download, commands)
			owner.attachRecovered(editor)
			yield* Effect.addFinalizer(() =>
				Effect.sync(() => {
					owner.detach(editor)
					editor.remove()
					download.remove()
    commands.remove()
				}),
			)
			return {
				document,
				editor,
				save: Effect.void,
				synchronize: Effect.void,
				pending: owner.draft.pending,
				recovery: owner.recovery,
			}
		}
		if (frontmatter(initialized.initial.source).values.mode === "present")
			return yield* Effect.fail(new Error("Worker source-editor prototype does not support presentation mode"))
		let queue = initialized.queue
		let lastMergedRevision = opened.current().revision
		workerBootMark("route:Editor-start")
		let editor = new EditorElement()
		let bindings = workerEditorBindings(
			editor,
			queue,
			command => {
				fork(command)
			},
			options.report,
			owner.draft.retain,
		)
		function unsupported() {
			options.report(
				"Worker route prototype: this feature has not crossed to worker authority",
			)
		}
  let threads: readonly typeof OpenedThreadSchema.Type[] = []
  let threadBasis: { epoch: string; sequence: number } | undefined
  let commentsVisible = false
  let routeClosed = false
  let activeComposer: string | undefined
  let activeAssetSelection: string | undefined
  let blobs = yield* DocumentBlobs.pipe(Effect.provide(DocumentBlobs.layer(document)))
  let controls = globalThis.document.createElement("div")
  let toggle = globalThis.document.createElement("button")
  toggle.dataset.workerComments = ""
  toggle.textContent = "Comments"
  let panel = globalThis.document.createElement("aside")
  panel.dataset.workerCommentsPanel = ""
  panel.hidden = true
  let filesInput = globalThis.document.createElement("input")
  filesInput.type = "file"
  filesInput.multiple = true
  filesInput.dataset.workerAssets = ""
  filesInput.setAttribute("aria-label", "Attach files")
  let assetsPanel = globalThis.document.createElement("aside")
  assetsPanel.dataset.workerAssetsPanel = ""
  let pickerButton = globalThis.document.createElement("button")
  pickerButton.type = "button"
  pickerButton.dataset.workerMediaPicker = ""
  pickerButton.textContent = "Insert media"
  pickerButton.addEventListener("click", () => forkCapture(requestAsset().pipe(Effect.catch(error => Effect.sync(() => options.report(error))))))
  let drawingButton = globalThis.document.createElement("button")
  drawingButton.type = "button"
  drawingButton.dataset.workerNewDrawing = ""
  drawingButton.textContent = "New whiteboard"
  drawingButton.addEventListener("click", () => forkCapture(draw().pipe(Effect.catch(error => Effect.sync(() => options.report(error))))))
  controls.append(toggle, filesInput, pickerButton, drawingButton)
  let drawingOwner = owner
  let drawingGate = yield* Semaphore.make(1)
  let activeDrawing: { close: Effect.Effect<void>; epoch: number; token: string } | undefined
  function draw(board?: string) {
   return drawingGate.withPermitsIfAvailable(1)(Effect.scoped(Effect.gen(function* () {
    if (routeClosed || !opened.current().canRead) return yield* Effect.fail(new Error("This document is unavailable"))
    let epoch = opened.current().admissionEpoch
    let view = yield* Effect.acquireRelease(opened.boards.acquire(board), view => opened.boards.release(view.token).pipe(Effect.catchCause(() => Effect.void)))
    if (opened.current().admissionEpoch !== epoch) return yield* Effect.fail(new Error("Whiteboard admission changed"))
    let actor = opened.current().actor
    if (!actor) return yield* Effect.fail(new Error("Whiteboard actor is unavailable"))
    let drawing = yield* openWorkerDrawing({
     document, board: view.board, actor, name: options.name, dialog: options.dialog,
     view: () => { let current = opened.current().boardViews?.find(value => value.token === view.token); if (!current) throw new Error("This whiteboard view is unavailable"); return current },
     source: () => editor.source?.value ?? "",
     writable: () => opened.boards.writable() && drawingOwner.recovery() === undefined,
     changes: opened.changes, presenceChanges: opened.changes,
     people: () => opened.current().presence ?? [], announce: opened.boards.announce,
     write: (values, embed) => opened.boards.write(view.token, view.board, values, embed),
     undo: () => opened.undo(false), redo: () => opened.undo(true), report: options.report,
    })
    let active = { ...drawing, epoch, token: view.token }
    activeDrawing = active
    yield* drawing.closed.pipe(Effect.ensuring(Effect.sync(() => { if (activeDrawing === active) activeDrawing = undefined })))
   }))).pipe(Effect.flatMap(result => Option.isNone(result) ? Effect.fail(new Error("A whiteboard is already open")) : Effect.succeed(result.value)))
  }
  let media = yield* makeWorkerMedia({ blobs, current: opened.current, changes: opened.changes, source: () => editor.source?.value ?? "" })
  function renderAssets() {
   let current = opened.current()
   setMarkup(assetsPanel, attachmentsMarkup((current.assets ?? []).map(asset => ({ id: asset.id, fields: { type: "asset", name: asset.name, data: asset.reference ?? "", muted: String(asset.muted) }, audible: asset.audible })), (current.whiteboards ?? []).map(board => ({ id: board.id, fields: { type: "whiteboard", name: board.name } })), canEdit(), new Map(), false))
  }
  function runAssetCommand(work: Effect.Effect<unknown, unknown>) {
   return Effect.runPromise(Fiber.join(fork(Effect.suspend(() => canEdit() ? work : Effect.fail(new Error("This document is read-only"))))))
  }
  function previewAsset(id: string, name: string) {
   return Effect.scoped(Effect.gen(function* () {
    let reference = media.references(`![media](asset:${id})`).find(([asset]) => asset === id)?.[1]
    if (!reference || routeClosed || !opened.current().canRead) return yield* Effect.fail(new Error("Attachment unavailable"))
    let url = yield* media.acquire(reference)
    if (!url || routeClosed || !opened.current().canRead) return yield* Effect.fail(new Error("Attachment unavailable"))
    let type = opened.current().assets?.find(asset => asset.id === id)?.contentType ?? "image/svg+xml"
    yield* mediaPreviewDialog(options.dialog, options.report, url, name, type.startsWith("video/") ? "video" : type.startsWith("audio/") ? "audio" : "image")
   }))
  }
  function requestAsset() {
   return Effect.gen(function* () {
    if (!canEdit()) return yield* Effect.fail(new Error("This document is read-only"))
    let capture = yield* opened.assets.capture(true)
    activeAssetSelection = capture.token
    let choices: MediaChoice[] = (opened.current().assets ?? []).flatMap(asset => {
     let kind: MediaChoice["kind"] | undefined = asset.contentType.startsWith("video/") ? "video" : asset.contentType.startsWith("audio/") ? "audio" : asset.contentType.startsWith("image/") ? "image" : undefined
     return kind ? [{ id: asset.id, name: asset.name, kind }] : []
    })
    if (routeClosed) { yield* opened.assets.release(capture.token); return }
    openMediaPicker({ dialog: options.dialog, choices, canDraw: false, select: id => runAssetCommand(opened.assets.insert(id, capture.token)), link: async () => { throw new Error("Worker URL insertion not yet proved") }, upload: () => filesInput.click(), createDrawing: unsupported, report: options.report, closed: () => { if (activeAssetSelection === capture.token) activeAssetSelection = undefined; if (!routeClosed) fork(opened.assets.release(capture.token)) } })
   })
  }
  assetsPanel.addEventListener("click", event => {
   if (!(event.target instanceof Element)) return
   let action = event.target.closest<HTMLElement>("[data-action]")?.dataset.action
   if (!action) return
   let id = action.slice(action.indexOf(":") + 1)
   let asset = opened.current().assets?.find(asset => asset.id === id)
   let board = opened.current().whiteboards?.find(board => board.id === id)
   if (action === "asset-upload") filesInput.click()
   else if (action.startsWith("asset-insert:")) fork(Effect.gen(function* () { let capture = yield* opened.assets.capture(); yield* Effect.acquireUseRelease(Effect.succeed(capture), token => opened.assets.insert(id, token.token), token => opened.assets.release(token.token)) }))
   else if (action.startsWith("asset-rename:") && asset) openDialog(options.dialog, options.report, () => {}, "Rename attachment", html`<label>Name<input name="name" value="${asset.name}" required></label>`, form => runAssetCommand(opened.assets.rename(id, String(form.get("name") ?? ""))))
   else if (action.startsWith("asset-delete:") && asset) fork(Effect.gen(function* () { if (!(yield* confirmDialog(options.dialog, "Remove this attachment and its source references?", "Remove"))) return; if (!canEdit()) return yield* Effect.fail(new Error("This document is read-only")); yield* opened.assets.delete(id) }).pipe(Effect.catch(error => Effect.sync(() => options.report(error)))))
   else if (action.startsWith("asset-rename:") && board) openDialog(options.dialog, options.report, () => {}, "Rename whiteboard", html`<label>Name<input name="name" value="${board.name}" required></label>`, form => runAssetCommand(withBoard(id, view => opened.boards.write(view.token, view.board, [["name", String(form.get("name") ?? "")]], false))))
   else if (action.startsWith("asset-delete:") && board) fork(Effect.gen(function* () { if (!(yield* confirmDialog(options.dialog, "Remove this whiteboard and its source references?", "Remove"))) return; if (!canEdit()) return yield* Effect.fail(new Error("This document is read-only")); yield* withBoard(id, view => opened.boards.delete(view.token, view.board)) }).pipe(Effect.catch(error => Effect.sync(() => options.report(error)))))
   else if (action.startsWith("asset-mute:") && asset) fork(opened.assets.mute(id, !asset.muted))
   else if (action.startsWith("asset-download:") && asset) fork(Effect.gen(function* () { let reference = yield* opened.assets.reference(id); let blob = yield* blobs.read(reference.reference); let after = yield* opened.assets.reference(id); if (!opened.current().canRead || after.version !== reference.version || after.reference !== reference.reference) return yield* Effect.fail(new Error("Attachment changed while downloading")); yield* downloadFile({ name: reference.name, blob }) }))
   else if (action.startsWith("drawing:")) forkCapture(draw(id).pipe(Effect.catch(error => Effect.sync(() => options.report(error)))))
   else unsupported()
  })
  function withBoard<Value, Failure>(id: string, use: (view: typeof import("./worker-opened-schemas").BoardViewSchema.Type) => Effect.Effect<Value, Failure>) {
   return Effect.acquireUseRelease(opened.boards.acquire(id), use, view => opened.boards.release(view.token).pipe(Effect.catchCause(() => Effect.void)))
  }
  let commentOwner = owner
  function canEdit() { return !routeClosed && opened.current().canWrite && commentOwner.recovery() === undefined }
  function canComment() { return canEdit() && opened.current().metadata?.commentsDisabled !== true }
  function attachFiles(files: readonly File[]) {
   return Effect.gen(function* () {
    if (!canEdit() || files.length < 1 || files.length > 8) return yield* Effect.fail(new Error("Attachments are unavailable"))
    let selection = yield* opened.assets.capture()
    yield* Effect.acquireUseRelease(Effect.succeed(selection), captured => Effect.gen(function* () {
     let references: { reference: string; name: string }[] = []
     for (let file of files) references.push({ reference: yield* blobs.store(file), name: file.name })
     if (!canEdit()) return yield* Effect.fail(new Error("This document is read-only"))
     yield* opened.assets.attach(captured.token, references)
    }), captured => opened.assets.release(captured.token).pipe(Effect.timeout("1 second"), Effect.catchCause(() => Effect.void)))
   })
  }
  filesInput.addEventListener("change", () => {
   let files = Array.from(filesInput.files ?? [])
   fork(attachFiles(files).pipe(Effect.catch(error => Effect.sync(() => options.report(error)))))
  })
  function renderComments() {
   workerBootMark("comments:render", JSON.stringify({ writable: opened.current().canWrite, recovery: commentOwner.recovery() !== undefined, nativeReadOnly: editor.source?.readOnly, disabled: opened.current().metadata?.commentsDisabled === true, connected: panel.isConnected }))
   let markup = commentsMarkup(threads, opened.current().metadata?.commentsDisabled === true, opened.current().canWrite && commentOwner.recovery() === undefined)
   setMarkup(panel, markup)
   workerBootMark("comments:markup", JSON.stringify({ markup: String(markup), panel: panel.outerHTML }))
  }
  let refreshComments = Effect.suspend(() => commentsVisible && !routeClosed ? opened.discussion.view().pipe(
   Effect.tap(view => Effect.sync(() => { threads = view.threads; threadBasis = { epoch: view.epoch, sequence: view.sequence }; renderComments() })),
   Effect.catch(error => Effect.sync(() => { threadBasis = undefined; options.report(error) })),
  ) : Effect.void)
  function compose(reply?: string) {
   return Effect.gen(function* () {
    workerBootMark("comments:compose", JSON.stringify({ allowed: canComment(), start: editor.source?.selectionStart, end: editor.source?.selectionEnd }))
    if (!canComment()) return yield* Effect.fail(new Error("Comments are unavailable"))
    let composer = yield* reply === undefined ? opened.discussion.capture() : opened.discussion.reply(reply)
    if (routeClosed) { yield* opened.discussion.release(composer.token); return }
    activeComposer = composer.token
    workerBootMark("comments:composer-ready")
    let quote = "quote" in composer && typeof composer.quote === "string" ? composer.quote : undefined
    openDialog(options.dialog, options.report, () => { fork(refreshComments) }, reply === undefined ? "Add comment" : "Reply", html`${quote ? html`<blockquote>${quote}</blockquote>` : ""}<label>Comment<textarea name="body" rows="4" required></textarea></label>`, form => Effect.runPromise(Effect.suspend(() => routeClosed || !canComment() ? Effect.fail(new Error("This document is read-only")) : opened.discussion.submit(composer.token, String(form.get("body") ?? "")))))
    options.dialog.addEventListener("close", () => { if (activeComposer === composer.token) { activeComposer = undefined; if (!routeClosed) fork(opened.discussion.release(composer.token).pipe(Effect.catch(() => Effect.void))) } }, { once: true })
   })
  }
  let stopCaptures = FiberSet.clear(captures).pipe(Effect.andThen(Effect.suspend(() => {
   let token = activeComposer
   let assetToken = activeAssetSelection
   activeComposer = undefined
   activeAssetSelection = undefined
   if (options.dialog.open) options.dialog.close("cancel")
   return Effect.all([token === undefined ? Effect.void : opened.discussion.release(token).pipe(Effect.timeout("1 second"), Effect.catchCause(() => Effect.void)), assetToken === undefined ? Effect.void : opened.assets.release(assetToken).pipe(Effect.timeout("1 second"), Effect.catchCause(() => Effect.void))], { concurrency: 2 }).pipe(Effect.asVoid)
  })))
  toggle.addEventListener("click", () => { commentsVisible = !commentsVisible; panel.hidden = !commentsVisible; renderAssets(); if (commentsVisible) fork(refreshComments) })
  panel.addEventListener("click", event => {
   if (!(event.target instanceof Element)) return
   let target = event.target.closest("[data-action]")
   if (!(target instanceof HTMLElement)) return
   event.stopPropagation()
   let action = target.dataset.action ?? ""
   workerBootMark("comments:click", action)
   if (action === "comment") forkCapture(compose().pipe(Effect.catch(error => Effect.sync(() => { workerBootMark("comments:error", error instanceof Error ? error.name + ":" + error.message : String(error)); options.report(error) }))))
   else if (action.startsWith("reply:")) forkCapture(compose(action.slice(6)).pipe(Effect.catch(error => Effect.sync(() => { workerBootMark("comments:error", error instanceof Error ? error.name + ":" + error.message : String(error)); options.report(error) }))))
   else if (action.startsWith("jump-comment:")) { let thread = threads.find(value => value.id === action.slice(13)); if (thread?.range) editor.selectRange(thread.range.start, thread.range.end) }
   else {
    let command = action.startsWith("resolve:") ? opened.discussion.resolve(action.slice(8), true) : action.startsWith("reopen:") ? opened.discussion.resolve(action.slice(7), false) : action.startsWith("delete-comment:") ? opened.discussion.delete(action.slice(15)) : action === "toggle-comments" ? opened.properties.update({ commentsDisabled: opened.current().metadata?.commentsDisabled !== true }) : undefined
    if (command) fork(command.pipe(Effect.andThen(refreshComments), Effect.catch(error => Effect.sync(() => { workerBootMark("comments:error", error instanceof Error ? error.name + ":" + error.message : String(error)); options.report(error) }))))
   }
  })
		editor.configure({
			authority: "worker",
			content: initialized.initial.source,
			atoms: [],
			editable: opened.current().canWrite,
			preferences: options.preferences,
			documents: options.documents,
			openDocument: options.openDocument,
			isDrawing: id => (opened.current().whiteboards ?? []).some(board => board.id === id),
			openDrawing: id => forkCapture(draw(id).pipe(Effect.catch(error => Effect.sync(() => options.report(error))))),
			createLinkedDocument: async () => {
				throw new Error("Worker linked creation unavailable")
			},
			canComment,
			canAttach: canEdit,
			comments: () => {
    let state = queue.state()
    return threadBasis?.epoch === state.epoch && threadBasis.sequence === state.sequence && editor.source?.value === state.source ? threads.filter(thread => thread.fields.resolved !== "true").flatMap(thread => thread.range ? [thread.range] : []) : []
   },
			report: options.report,
			change: (...args) => { threadBasis = undefined; return bindings.change(...args) },
			undo: bindings.undo,
			select: bindings.resumeUndo,
			requestAsset: () => forkCapture(requestAsset().pipe(Effect.catch(error => Effect.sync(() => options.report(error))))),
   previewImage: (url, name) => { if (url.startsWith("asset:")) forkCapture(previewAsset(url.slice(6), name).pipe(Effect.catch(error => Effect.sync(() => options.report(error))))); else unsupported() },
			attach: files => fork(attachFiles(files).pipe(Effect.catch(error => Effect.sync(() => options.report(error))))),
			command: action => { if (action === "comment") forkCapture(compose().pipe(Effect.catch(error => Effect.sync(() => { workerBootMark("comments:error", error instanceof Error ? error.name + ":" + error.message : String(error)); options.report(error) })))); else unsupported() },
			find: query => openFind(query, false, { editor, bar: options.findBar }),
		})
		workerBootMark("route:Editor-configured")
		host.replaceChildren(controls, editor, panel, assetsPanel)
		owner.attach(editor, queue, initialized.initial.source)
		yield* Effect.addFinalizer(() =>
			Effect.gen(function* () {
    routeClosed = true
    yield* stopCaptures
				owner.detach(editor)
     if (options.dialog.open) options.dialog.close("cancel"); controls.remove(); panel.remove(); assetsPanel.remove()
				editor.remove()
				options.findBar.hidden = true
				options.findBar.replaceChildren()
			}),
		)
		yield* opened.changes.pipe(
			Stream.runForEach(state =>
				Effect.sync(() => {
					editor.setEditable(state.canWrite && owner.recovery() === undefined)
     filesInput.disabled = !canEdit()
     drawingButton.disabled = !opened.boards.writable() || owner.recovery() !== undefined
					options.title(state.metadata?.title ?? "Untitled")
     renderAssets()
     if (!state.canRead) fork(stopCaptures)
     else if (activeDrawing && (activeDrawing.epoch !== state.admissionEpoch || !state.boardViews?.some(view => view.token === activeDrawing?.token))) fork(activeDrawing.close)
     if (commentsVisible) fork(refreshComments)
					options.status(
						state.status.pending,
						state.status.unsaved,
						state.failed || state.status.failures.save !== undefined,
					)
				}),
			),
			Effect.forkScoped({ startImmediately: true }),
		)
		yield* Stream.fromEventListener(editor, "keyup").pipe(
			Stream.merge(Stream.fromEventListener(editor, "pointerup")),
			Stream.merge(Stream.fromEventListener(editor, "compositionend")),
			Stream.runForEach(() => Effect.sync(bindings.resumeUndo)),
			Effect.forkScoped({ startImmediately: true }),
		)
		yield* opened.changes.pipe(
			Stream.map(() => undefined),
			Stream.merge(
				Stream.fromEventListener(editor, "keyup").pipe(
					Stream.map(() => undefined),
				),
			),
			Stream.merge(
				Stream.fromEventListener(editor, "pointerup").pipe(
					Stream.map(() => undefined),
				),
			),
			Stream.merge(
				Stream.fromEventListener(editor, "compositionend").pipe(
					Stream.map(() => undefined),
				),
			),
			Stream.buffer({ capacity: 1, strategy: "sliding" }),
			Stream.runForEach(() =>
				queue.settle.pipe(
					Effect.flatMap(settled =>
						Effect.suspend(() => {
							let state = opened.current()
							if (
								!settled ||
								!state.canWrite ||
								owner.recovery() !== undefined ||
								state.revision <= lastMergedRevision
							)
								return Effect.void
							let revision = state.revision
							return bindings.merge().saved.pipe(
								Effect.tap(accepted =>
									Effect.sync(() => {
										if (accepted) { lastMergedRevision = revision; fork(refreshComments) }
									}),
								),
								Effect.asVoid,
								Effect.catch(error => Effect.sync(() => { workerBootMark("comments:error", error instanceof Error ? error.name + ":" + error.message : String(error)); options.report(error) })),
							)
						}),
					),
				),
			),
			Effect.forkScoped({ startImmediately: true }),
		)
		return {
			document,
			editor,
			save: stopCaptures.pipe(Effect.andThen(Effect.suspend(() =>
				owner.recovery() !== undefined && !owner.draft.pending()
					? Effect.void
					: queue.settle.pipe(
							Effect.flatMap(settled =>
								settled
									? opened.save
									: Effect.fail(new Error("Native queue not settled")),
							),
						),
			))),
			synchronize: stopCaptures.pipe(Effect.andThen(opened.synchronize)),
			pending: owner.draft.pending,
			recovery: owner.recovery,
		}
	})
}
