import * as Schema from "effect/Schema"
import { DocumentMetadataSchema } from "../../shared/document-metadata"
import { LivePresenceSchema } from "../../shared/document-live"
import { IdentifierSchema } from "../../shared/model"

export let EditorBasisSchema = Schema.Struct({
 epoch: Schema.String,
 sequence: Schema.Natural,
 source: Schema.String,
})
export let OpenedDocumentRecordSchema = Schema.Struct({
 id: IdentifierSchema,
 kind: Schema.Literals(["document", "slides", "whiteboard"]),
 title: Schema.String,
 content: Schema.String,
 folder: Schema.String,
 archived: Schema.Boolean,
 updated: Schema.Natural,
 room: Schema.String,
 tags: Schema.Array(Schema.String),
 pinned: Schema.Boolean,
 created: Schema.Natural,
 deleted: Schema.Boolean,
})
export let OpenedDocumentDescriptorSchema = Schema.Struct({
 id: OpenedDocumentRecordSchema.fields.id,
 kind: OpenedDocumentRecordSchema.fields.kind,
 title: OpenedDocumentRecordSchema.fields.title,
 folder: OpenedDocumentRecordSchema.fields.folder,
 archived: OpenedDocumentRecordSchema.fields.archived,
 updated: OpenedDocumentRecordSchema.fields.updated,
 room: OpenedDocumentRecordSchema.fields.room,
 tags: OpenedDocumentRecordSchema.fields.tags,
 pinned: OpenedDocumentRecordSchema.fields.pinned,
 created: OpenedDocumentRecordSchema.fields.created,
 deleted: OpenedDocumentRecordSchema.fields.deleted,
})
export let OpenedThemeSchema = Schema.Struct({ id: Schema.String, fields: Schema.Record(Schema.String, Schema.String), source: Schema.String })
export let OpenedAssetSchema = Schema.Struct({
 id: IdentifierSchema,
 name: Schema.String,
 reference: Schema.optional(Schema.String),
 contentType: Schema.String,
 referenceVersion: Schema.optional(Schema.String),
 muted: Schema.Boolean,
 audible: Schema.Boolean,
})
export let OpenedWhiteboardSchema = Schema.Struct({ id: IdentifierSchema, name: Schema.String, referenceLight: Schema.optional(Schema.String), referenceDark: Schema.optional(Schema.String) })
export let OpenedThreadSchema = Schema.Struct({
 id: IdentifierSchema,
 fields: Schema.Record(Schema.String, Schema.String),
 range: Schema.optional(Schema.Struct({ start: Schema.Natural, end: Schema.Natural })),
 orphaned: Schema.Boolean,
})
export let EditorThreadRangeSchema = Schema.Struct({
 id: IdentifierSchema,
 start: Schema.Natural,
 end: Schema.Natural,
})
export let OpenedConnectionSchema = Schema.Struct({
 status: Schema.Literals(["offline", "syncing", "synced", "unavailable", "failed"]),
 message: Schema.optional(Schema.String),
 reason: Schema.optional(Schema.Literals(["access", "authentication", "unavailable"])),
})
let failure = Schema.optional(Schema.Literals(["access", "conflict", "storage", "network", "failure"]))
export let WorkerOpenedSnapshotSchema = Schema.Struct({
 revision: Schema.Natural,
 editor: Schema.optional(Schema.Struct({
  basis: EditorBasisSchema,
  ranges: Schema.Array(EditorThreadRangeSchema),
 })),
 projection: Schema.optional(Schema.Struct({
  document: OpenedDocumentRecordSchema,
  metadata: DocumentMetadataSchema,
  presentationItem: Schema.optional(Schema.Natural),
  themes: Schema.Array(OpenedThemeSchema),
  assets: Schema.Array(OpenedAssetSchema),
  whiteboards: Schema.Array(OpenedWhiteboardSchema),
  threads: Schema.Array(OpenedThreadSchema),
 })),
 canWrite: Schema.Boolean,
 connection: OpenedConnectionSchema,
 status: Schema.Struct({
  unsaved: Schema.Natural,
  pending: Schema.Natural,
  rejected: Schema.Natural,
  failures: Schema.Struct({ save: failure, snapshot: failure, projection: failure }),
 }),
})
export let WorkerRecordSchema = Schema.Struct({
 id: IdentifierSchema,
 fields: Schema.Record(Schema.String, Schema.String),
})
export let WorkerPresenceSchema = Schema.Array(LivePresenceSchema)
export type WorkerOpenedSnapshot = typeof WorkerOpenedSnapshotSchema.Type
export type WorkerEditorProjection = NonNullable<WorkerOpenedSnapshot["editor"]>
export type WorkerRecord = typeof WorkerRecordSchema.Type

export let WorkerAdmissionSchema = Schema.Struct({
 subject: Schema.String,
 epoch: Schema.Natural,
 exists: Schema.Boolean,
 deleted: Schema.Boolean,
 role: Schema.optional(Schema.Literals(["owner", "writer", "reader"])),
 account: Schema.Literals(["ready", "signedOut", "accountChanged"]),
 online: Schema.Boolean,
})
export let BoardViewSchema = Schema.Struct({ token: Schema.String, generation: Schema.Natural, board: Schema.String, fields: Schema.Record(Schema.String, Schema.String), documentWhiteboard: Schema.optional(Schema.String) })
export let WorkerOpenedNoticeSchema = Schema.Struct({
 document: Schema.optional(OpenedDocumentDescriptorSchema),
 presentationItem: Schema.optional(Schema.Natural),
 themes: Schema.optional(Schema.Array(OpenedThemeSchema)),
 boardViews: Schema.optional(Schema.Array(BoardViewSchema)),
 boardGeneration: Schema.optional(Schema.Natural),
 actor: Schema.optional(Schema.String),
 presence: Schema.optional(WorkerPresenceSchema),
 assets: Schema.optional(Schema.Array(OpenedAssetSchema)),
 whiteboards: Schema.optional(Schema.Array(OpenedWhiteboardSchema)),
 revision: Schema.Natural,
 admissionEpoch: Schema.Natural,
 metadata: Schema.optional(DocumentMetadataSchema),
 canWrite: Schema.Boolean,
 connection: OpenedConnectionSchema,
 status: WorkerOpenedSnapshotSchema.fields.status,
})
