import { LivePresenceSchema } from "../../shared/document-live"
import { BoardViewSchema } from "./worker-opened-schemas"
import { OpenedThreadSchema } from "./worker-opened-schemas"
import { DomainIntentSchema, DomainOutcomeSchema, DomainPatchSchema } from "./domain-schemas"
import * as Rpc from "effect/rpc/Rpc"
import * as RpcGroup from "effect/rpc/RpcGroup"
import * as Schema from "effect/Schema"
import { DocumentMetadataSchema } from "../../shared/document-metadata"
import { WorkerOpenedEventSchema } from "./worker-opened-observation"
import { WorkerAdmissionSchema, WorkerOpenedNoticeSchema, WorkerOpenedSnapshotSchema } from "./worker-opened-schemas"

export class WorkerDocumentFailure extends Schema.TaggedError<WorkerDocumentFailure>()("WorkerDocumentFailure", {
 reason: Schema.Literals(["Closed", "AlreadyOpen", "AlreadyObserving", "SessionFailure", "PermissionDenied", "AppliedUnknown", "DomainNotApplied", "HandleCapacity", "SelectionUnavailable", "BasisUnavailable", "SequenceUnavailable", "CandidateUnavailable", "ClipboardUnavailable", "ClipboardChanged", "Rejected", "Unsaved", "QueuedBlocked"]),
 category: Schema.String,
}) {}
let selectedRange = Schema.Struct({ start: Schema.Natural, end: Schema.Natural, direction: Schema.Literals(["forward", "backward", "none"]) })
let undoReceipt = Schema.Struct({
	request: Schema.String,
	epoch: Schema.String,
	sequence: Schema.Natural,
	status: Schema.Literals(["applied", "not-applied", "unknown"]),
	action: Schema.optional(Schema.String),
	failure: Schema.optional(Schema.String),
})

let token = Schema.Struct({ token: Schema.String })
let range = Schema.Struct({ epoch: Schema.String, sequence: Schema.Natural, start: Schema.Natural, end: Schema.Natural })
let history = Schema.Struct({ token: Schema.String, complete: Schema.Boolean, moments: Schema.Array(Schema.Struct({ position: Schema.Natural, label: Schema.String, time: Schema.Natural, subject: Schema.String })) })
let moment = Schema.Struct({ token: Schema.String, position: Schema.Natural, source: Schema.String, metadata: DocumentMetadataSchema, records: Schema.Array(Schema.Struct({ id: Schema.String, kind: Schema.String, label: Schema.String })) })
let patch = DomainPatchSchema
export let NativeReceiptSchema = Schema.Struct({ intentDigest: Schema.String, epoch: Schema.String, sequence: Schema.Natural, beforeLength: Schema.Natural, afterLength: Schema.Natural, agreement: Schema.Literal(true), unsaved: Schema.Natural, operation: Schema.optional(Schema.String) })
export let WorkerDocumentRpc = RpcGroup.make(
 Rpc.make("ObserveNotice", { success: WorkerOpenedNoticeSchema, error: WorkerDocumentFailure, stream: true }),
 Rpc.make("ExecuteAdmitted", { payload: { token: Schema.String, admissionEpoch: Schema.Natural }, success: Schema.Struct({ action: Schema.String }), error: WorkerDocumentFailure }),
 Rpc.make("ExecuteDomain", { payload: { intent: DomainIntentSchema }, success: DomainOutcomeSchema, error: WorkerDocumentFailure }),
 Rpc.make("RecoverDomain", { payload: { request: Schema.String, intentDigest: Schema.String }, success: Schema.Struct({ found: Schema.Boolean, outcome: Schema.optional(DomainOutcomeSchema) }), error: WorkerDocumentFailure }),
 Rpc.make("Synchronize", { error: WorkerDocumentFailure }),
 Rpc.make("ReplaceReceipt", { payload: { intentDigest: Schema.String, epoch: Schema.String, sequence: Schema.Natural, admissionEpoch: Schema.Natural, clipboardToken: Schema.optional(Schema.String), ranges: Schema.Array(Schema.Struct({ start: Schema.Natural, remove: Schema.Natural, text: Schema.String })) }, success: NativeReceiptSchema, error: WorkerDocumentFailure }),
 Rpc.make("RecoverReceipt", { payload: { intentDigest: Schema.String }, success: Schema.Struct({ found: Schema.Boolean, receipt: Schema.optional(NativeReceiptSchema) }), error: WorkerDocumentFailure }),
 Rpc.make("Connectivity", { payload: { online: Schema.Boolean }, success: WorkerOpenedNoticeSchema, error: WorkerDocumentFailure }),

 Rpc.make("FixtureFault", { payload: { openFailure: Schema.Boolean, writeFailure: Schema.Boolean }, error: WorkerDocumentFailure }),
 Rpc.make("FixtureDraft", { payload: { source: Schema.String }, error: WorkerDocumentFailure }),
 Rpc.make("EditorRead", { success: Schema.Struct({ epoch: Schema.optional(Schema.String), sequence: Schema.optional(Schema.Natural), source: Schema.String }), error: WorkerDocumentFailure }),
	Rpc.make("Begin", {
		success: Schema.Struct({ epoch: Schema.String, sequence: Schema.Natural, source: Schema.String }),
		error: WorkerDocumentFailure,
	}),
	Rpc.make("ReplaceQueued", {
		payload: {
			epoch: Schema.String,
			sequence: Schema.Natural,
			clipboardToken: Schema.optional(Schema.String),
			ranges: Schema.Array(Schema.Struct({ start: Schema.Natural, remove: Schema.Natural, text: Schema.String })),
		},
		success: Schema.Struct({ epoch: Schema.String, sequence: Schema.Natural, source: Schema.String, unsaved: Schema.Natural }),
		error: WorkerDocumentFailure,
	}),
	Rpc.make("UndoQueued", {
		payload: { request: Schema.String, epoch: Schema.String, sequence: Schema.Natural, redo: Schema.Boolean, admissionEpoch: Schema.Natural },
		success: undoReceipt,
		error: WorkerDocumentFailure,
	}),
	Rpc.make("MergeCandidate", {
		payload: { epoch: Schema.String, sequence: Schema.Natural, start: Schema.Natural, end: Schema.Natural, ranges: Schema.optional(Schema.Array(selectedRange)) },
		success: Schema.Struct({ candidate: Schema.String, source: Schema.String, start: Schema.Natural, end: Schema.Natural, ranges: Schema.optional(Schema.Array(selectedRange)) }),
		error: WorkerDocumentFailure,
	}),
	Rpc.make("Adopt", {
		payload: { epoch: Schema.String, sequence: Schema.Natural, candidate: Schema.String },
		error: WorkerDocumentFailure,
	}),
	Rpc.make("ReleaseCandidate", {
		payload: { candidate: Schema.String },
		error: WorkerDocumentFailure,
	}),
	Rpc.make("CaptureClipboard", {
		payload: { epoch: Schema.String, sequence: Schema.Natural, start: Schema.Natural, end: Schema.Natural },
		success: Schema.Struct({ token: Schema.String, selected: Schema.String }),
		error: WorkerDocumentFailure,
	}),
	Rpc.make("ResolveClipboard", {
		payload: { token: Schema.String, epoch: Schema.String, sequence: Schema.Natural },
		success: Schema.Struct({ epoch: Schema.String, sequence: Schema.Natural, source: Schema.String, start: Schema.Natural, end: Schema.Natural }),
		error: WorkerDocumentFailure,
	}),
	Rpc.make("ReleaseClipboard", {
		payload: { token: Schema.String },
		error: WorkerDocumentFailure,
	}),

 Rpc.make("Protection", { success: Schema.Struct({ unsaved: Schema.Natural, dirty: Schema.Boolean }), error: WorkerDocumentFailure }),
 Rpc.make("Open", { payload: { document: Schema.String, workspace: Schema.String, subject: Schema.String, origin: Schema.String, online: Schema.Boolean, activityChannel: Schema.NonEmptyString, cooperativeWork: Schema.Boolean, admission: WorkerAdmissionSchema }, success: WorkerOpenedSnapshotSchema, error: WorkerDocumentFailure }),
 Rpc.make("Observe", { success: WorkerOpenedEventSchema, error: WorkerDocumentFailure, stream: true }),
 Rpc.make("Read", { success: WorkerOpenedSnapshotSchema, error: WorkerDocumentFailure }),
 Rpc.make("PropertiesUpdate", { payload: { patch, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("Present", { payload: { index: Schema.Natural, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("Presentation", { payload: { enabled: Schema.Boolean, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("AssetCapture", { payload: { range, inline: Schema.Boolean, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("BoardAcquire", { payload: { board: Schema.optional(Schema.String), admissionEpoch: Schema.Natural }, success: BoardViewSchema, error: WorkerDocumentFailure }),
 Rpc.make("BoardRelease", { payload: { token: Schema.String }, success: Schema.Natural, error: WorkerDocumentFailure }),
 Rpc.make("BoardWrite", { payload: { token: Schema.String, board: Schema.String, values: Schema.Array(Schema.Tuple([Schema.String, Schema.String])).check(Schema.isMaxLength(900)), embed: Schema.Boolean, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("BoardDelete", { payload: { token: Schema.String, board: Schema.String, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("BoardPresence", { payload: { presence: LivePresenceSchema, admissionEpoch: Schema.Natural }, error: WorkerDocumentFailure }),
 Rpc.make("AssetInsert", { payload: { asset: Schema.String, selection: Schema.String, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("AssetDelete", { payload: { asset: Schema.String, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("AssetRename", { payload: { asset: Schema.String, name: Schema.String, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("AssetMute", { payload: { asset: Schema.String, muted: Schema.Boolean, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("AssetAttach", { payload: { selection: Schema.String, files: Schema.Array(Schema.Struct({ reference: Schema.String, name: Schema.String })).check(Schema.isMinLength(1), Schema.isMaxLength(8)), admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("AssetReference", { payload: { asset: Schema.String }, success: Schema.Struct({ id: Schema.String, name: Schema.String, reference: Schema.String, version: Schema.String }), error: WorkerDocumentFailure }),
 Rpc.make("AssetRelease", { payload: { token: Schema.String }, error: WorkerDocumentFailure }),
 Rpc.make("CommentView", { success: Schema.Struct({ revision: Schema.Natural, epoch: Schema.String, sequence: Schema.Natural, disabled: Schema.Boolean, threads: Schema.Array(OpenedThreadSchema) }), error: WorkerDocumentFailure }),
 Rpc.make("CommentDelete", { payload: { thread: Schema.String, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),

 Rpc.make("CommentSubmit", { payload: { composer: Schema.String, body: Schema.String, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("CommentRelease", { payload: { token: Schema.String }, error: WorkerDocumentFailure }),
 Rpc.make("CommentResolve", { payload: { thread: Schema.String, resolved: Schema.Boolean, admissionEpoch: Schema.Natural }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("TaskToggle", { payload: { range }, success: token, error: WorkerDocumentFailure }),
 Rpc.make("HistoryLoad", { success: history, error: WorkerDocumentFailure }),
 Rpc.make("HistorySelect", { payload: { token: Schema.String, position: Schema.Natural }, success: moment, error: WorkerDocumentFailure }),

 Rpc.make("HistoryRelease", { payload: { token: Schema.String }, error: WorkerDocumentFailure }),

 Rpc.make("Execute", { payload: { token: Schema.String }, success: Schema.Struct({ action: Schema.String }), error: WorkerDocumentFailure }),
 Rpc.make("Release", { payload: { token: Schema.String }, error: WorkerDocumentFailure }),
 Rpc.make("Admission", { payload: { writable: Schema.Boolean, online: Schema.Boolean }, error: WorkerDocumentFailure }),
 Rpc.make("Flush", { error: WorkerDocumentFailure }),
 Rpc.make("LiveStatus", { success: Schema.Struct({ connected: Schema.Boolean, transport: Schema.Literals(["socket", "poll"]), opened: Schema.Boolean }), error: WorkerDocumentFailure }),
 Rpc.make("Close", { error: WorkerDocumentFailure }),
