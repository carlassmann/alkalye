from pathlib import Path
import json,hashlib
root=Path(__file__).parent
prior=Path('/tmp/alkalye-main-current72')
manifest=json.loads((prior/'undo-cwd-manifest.private.json').read_text())
version=next(v for v in manifest['versions'] if v['name']=='rewrite')
required=[root/'run-gate0.js',root/'extract-counts.py',root/'work.config.js',prior/'rewrite-oracle.private.json',Path(version['fixtures'][0]['sourceFile'])]
for arm in ['baseline-runtime','candidate-runtime']:
 required += [root/arm/p for p in ['prepare-timeline.ts','server-refresh.ts','tests/browser-driver.ts','experiments/performance/interaction-matrix/rewrite-oracle.ts','experiments/performance/main-current-preparation/network-with-receives.js','experiments/performance/main-current-preparation/runner-consumption.ts']]
for path in required:
 if not path.is_file() or path.stat().st_size==0:raise RuntimeError('Missing preflight dependency: '+str(path))
for name in ['profile-seed','data-seed']:
 if not (root/name).is_dir():raise RuntimeError('Missing seed '+name)
(root/'preflight.json').write_text(json.dumps({'status':'PASS','files':len(required),'sha256':{str(p.relative_to(root)) if p.is_relative_to(root) else p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in required}},indent=2)+'\n')
print('Preflight PASS: '+str(len(required))+' dependencies; no browser keys')
