import { browser, evaluate, serverAddress, assert } from "./browser-driver"
import { object } from "../src/shared/model"
if (process.argv.includes("--serve")) {
	let built = await Bun.build({
		entrypoints: ["tests/certified-source-fixture.ts"],
		target: "browser",
	})
	if (!built.success)
		throw new AggregateError(built.logs, "Lexical fixture build failed")
	let script = built.outputs[0]
	if (!script) throw new Error("Missing script")
	let style =
		(await Bun.file("src/client/components/editor.css").text()) +
		(await Bun.file("src/client/components/editor-viewport.css").text())
	let server = Bun.serve({
		port: 0,
		routes: {
			"/": new Response(
				`<!doctype html><style>:root{--font-mono:monospace;--font-size:18px;--line-height:1.8;--background:#fff;--foreground:#111}html,body{height:800px;margin:0}${style}</style><script type="module" src="/fixture.js"></script>`,
				{ headers: { "Content-Type": "text/html" } },
			),
			"/fixture.js": new Response(script, {
				headers: { "Content-Type": "text/javascript" },
			}),
		},
		fetch: () => new Response("Missing", { status: 404 }),
	})
	console.log(server.url.href)
	process.once("SIGTERM", () => {
		server.stop(true)
		process.exit(0)
	})
	await new Promise(() => {})
} else {
	let server = Bun.spawn(
		["bun", "tests/browser-certified-source.ts", "--serve"],
		{ stdout: "pipe", stderr: "inherit" },
	)
	let session = "lexical-source-" + Date.now()
	try {
		let url = await serverAddress(server.stdout)
		browser(session, "open", url)
		for (let index = 0; index < 13; index++) {
			evaluate(session, "window.certifiedPrepare(" + index + ")")
			browser(session, "press", index === 8 ? "[" : index === 9 ? "Enter" : "x")
			browser(session, "wait", "--fn", "window.certifiedDone===true")
			let state = object(evaluate(session, "window.certifiedVerify()"))
			assert(state.trusted === true && state.source === true, "Trusted exact native fixture " + index)
			console.log(JSON.stringify({ index, ...state }))
			if (index === 0 || index === 11) {
				for (let repeat = 0; repeat < 3; repeat++) {
					evaluate(session, "window.certifiedPrepareAgain()")
					browser(session, "press", "x")
					browser(session, "wait", "--fn", "window.certifiedDone===true")
					let repeated = object(evaluate(session, "window.certifiedVerify()"))
					assert(repeated.trusted === true && repeated.source === true, "Repeated trusted native edit " + index + ":" + repeat)
					console.log(JSON.stringify({ index, repeat, ...repeated }))
				}
			}
		}
	} finally {
		try {
			browser(session, "close")
		} finally {
			server.kill()
			await server.exited
		}
	}
}
