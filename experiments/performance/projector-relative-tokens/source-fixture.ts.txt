import { SourceTextarea } from "../src/client/source-textarea"
import { SourceTextarea as BaselineTextarea } from "../src/client/source-textarea-oracle"
import type { ProjectionMode } from "../src/client/source-textarea"
import { Projector } from "../src/shared/projection"
import { project, identity, toView } from "../src/shared/projection-oracle"

let linkedDocument = { id: "first", title: "First", archived: false }
let documents = [linkedDocument]
let mode: ProjectionMode = "project"
let native = document.createElement("textarea")
native.id = "editor"
document.body.append(native)
let source = new SourceTextarea(native, "", {
	documents: () => documents,
	mode: () => mode,
	caretTop: () => undefined,
	painted() {},
	fullDocument: true,
})
let controlNative = document.createElement("textarea")
document.body.append(controlNative)
let control = new BaselineTextarea(controlNative, "", {
	documents: () => documents,
	mode: () => mode,
	caretTop: () => undefined,
	painted() {},
	fullDocument: true,
})
let crlfNormalizationUnresolved: {
	requested: number
	actual: number
	native: number
}[] = []
let assertions = 0
let stages: string[] = []
function check(condition: boolean, message: string) {
	assertions++
	if (!condition) throw new Error(message)
}
function validate(content: string, start: number, end = start) {
	let previous = source.projection,
		previousOutput = JSON.stringify(previous)
	source.setSelectionRange(start, end)
	control.value = content
	control.setSelectionRange(start, end)
	let expected =
		mode === "raw"
			? identity(content)
			: project(content, { documents, active: { start, end }, iconSpace: true })
	check(
		JSON.stringify(previous) === previousOutput,
		"previous owned output changed",
	)
	check(source.value === content, "source changed")
	check(
		JSON.stringify(source.projection) === JSON.stringify(expected),
		"fresh projection differs at " + stages.at(-1),
	)
	if (content.includes("\r\n")) {
		check(
			source.selectionStart === control.selectionStart &&
				source.selectionEnd === control.selectionEnd,
			"CRLF baseline caret differs",
		)
		check(
			source.viewport.selectionStart === control.viewport.selectionStart &&
				source.viewport.selectionEnd === control.viewport.selectionEnd,
			"CRLF baseline native mapping differs",
		)
		if (source.selectionStart !== start)
			crlfNormalizationUnresolved.push({
				requested: start,
				actual: source.selectionStart,
				native: source.viewport.selectionStart,
			})
	} else {
		check(
			source.selectionStart === start && source.selectionEnd === end,
			"LF source selection moved",
		)
		check(
			source.viewport.selectionStart === toView(expected, start) &&
				source.viewport.selectionEnd === toView(expected, end),
			"LF native view mapping differs",
		)
	}
}
try {
	let body =
		"Body [Label](https://example.org) [[First]] 😀 end.\nNext [Two](https://two.example).\n"
	let cases = [
		body,
		"---\ntitle: Doc\n---\n" + body,
		"---\ntitle: Doc\n--\n" + body,
		"---\r\ntitle: Doc\r\n...  \r\n" + body,
		body,
		"```\n" + body + "```\nOutside [Three](https://three.example)",
		"````\n" + body + "```\nOutside [Three](https://three.example)",
		"````\n" + body + "```` trailing\nOutside [Three](https://three.example)",
		"````\n" + body + "````  \nOutside [Three](https://three.example)",
		"~~~\n" + body + "~~~\nOutside [Three](https://three.example)",
		"\n\n" + body,
		"`[Code](https://code.example)`\n    [Indented](https://indent.example)\n\t[[First]]\n   [Visible](https://visible.example)\n" +
			body,
	]
	for (let [index, content] of cases.entries()) {
		stages.push("transition" + index)
		source.value = content
		let link = content.indexOf("[Label]")
		for (let position of [
			0,
			Math.max(0, link - 1),
			Math.max(0, link),
			Math.max(0, link + 1),
			content.length,
		])
			validate(content, position)
		if (link >= 0) validate(content, link, link + 7)
		let firstLink = project(content, { documents, iconSpace: true }).links[0]
		if (firstLink) {
			validate(content, firstLink.source.start)
			validate(content, firstLink.source.end)
			validate(content, firstLink.source.start, firstLink.source.end)
		}
	}
	stages.push("two shifted suffixes")
	for (let content of ["Prefix\n\n" + body, body, "Another\n" + body, body]) {
		source.value = content
		validate(content, content.indexOf("[[First]]") + 2)
	}
	stages.push("same array retitle")
	source.value = body
	linkedDocument.title = "Renamed"
	source.sync()
	validate(body, 0)
	stages.push("raw then hold then project")
	for (let next of ["raw", "hold", "project"] satisfies ProjectionMode[]) {
		mode = next
		source.sync()
		validate(body, 0)
		validate(body, body.indexOf("[Label]") + 1)
	}
	stages.push("public output ownership and token boundaries")
	let projector = new Projector()
	let owned = projector.project(body, { documents, iconSpace: true })
	let firstOwned = owned.links[0]
	if (firstOwned) {
		firstOwned.source.start = -5
		let part = firstOwned.parts[0]
		if (part) part.source = -7
	}
	check(
		JSON.stringify(projector.project(body, { documents, iconSpace: true })) ===
			JSON.stringify(project(body, { documents, iconSpace: true })),
		"mutable public output poisoned private tokens",
	)
	let full = project(body, { documents }).links
	for (let left of [
		undefined,
		{ start: 0, end: 0 },
		{ start: NaN, end: 0 },
		{ start: -Infinity, end: Infinity },
		...full.flatMap(link => [
			{ start: link.source.start, end: link.source.start },
			{ start: link.source.end, end: link.source.end },
		]),
	]) {
		for (let right of [
			undefined,
			{ start: 0, end: 0 },
			{ start: body.length, end: body.length },
		]) {
			function active(range: { start: number; end: number } | undefined) {
				return range
					? full
							.filter(
								link =>
									link.source.end >=
										Math.max(
											0,
											Math.min(body.length, range.start, range.end),
										) &&
									link.source.start <=
										Math.max(
											0,
											Math.min(body.length, Math.max(range.start, range.end)),
										),
							)
							.map(link => link.source)
					: undefined
			}
			let expected =
				Boolean(left) === Boolean(right) &&
				(!left ||
					!right ||
					(![left.start, left.end, right.start, right.end].some(Number.isNaN) &&
						JSON.stringify(active(left)) === JSON.stringify(active(right))))
			check(
				projector.sameActiveTokens(body, left, right) === expected,
				"inclusive public active bounds differ",
			)
		}
	}
	stages.push("throwing title and reentrant publication")
	let calls = 0,
		rejected = false
	let failingDocument = {
		id: "first",
		archived: false,
		get title() {
			calls++
			if (calls === 2) throw Error("controlled title failure")
			return "First"
		},
	}
	try {
		projector.project(body, { documents: [failingDocument] })
	} catch {
		rejected = true
	}
	check(rejected, "title failure did not surface")
	let changedDocuments = [{ id: "first", title: "First", archived: false }]
	check(
		JSON.stringify(projector.project(body, { documents: changedDocuments })) ===
			JSON.stringify(project(body, { documents: changedDocuments })),
		"failed title rebuild poisoned next output",
	)
	projector.project(body, { documents })
	calls = 0
	let reentrantDocument = {
		id: "first",
		archived: false,
		get title() {
			calls++
			if (calls === 2) projector.project(body, { documents: changedDocuments })
			return linkedDocument.title
		},
	}
	let shifted = "New [[First]]\n" + body
	check(
		JSON.stringify(
			projector.project(shifted, { documents: [reentrantDocument] }),
		) === JSON.stringify(project(shifted, { documents })),
		"reentrant rebuild changed token ownership",
	)
	check(calls >= 2, "reentrant title getter was not exercised")
	stages.push("large small and duplicate rows")
	for (let content of [
		body.repeat(256),
		"Plain",
		body + body,
		"Prefix\n" + body + body,
		body,
	]) {
		source.value = content
		validate(content, 0)
		validate(content, content.length)
	}
	linkedDocument.title = "First"
	mode = "project"
	source.value = body
	source.setSelectionRange(2, 2)
	source.focus()
	native.addEventListener("input", () => {
		source.input()
		validate(source.value, source.selectionStart, source.selectionEnd)
		Object.assign(window, { lexicalTyped: source.value })
	})
	Object.assign(window, {
		lexicalDone: true,
		lexicalResults: { assertions, stages, crlfNormalizationUnresolved },
		lexicalInitial: body,
	})
} catch (error) {
	Object.assign(window, {
		lexicalDone: true,
		lexicalFailure: String(error),
		lexicalResults: { assertions, stages, crlfNormalizationUnresolved },
	})
}
