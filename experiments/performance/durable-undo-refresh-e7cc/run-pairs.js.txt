import { mkdir, rm, cp } from "node:fs/promises"
let root="/tmp/alkalye-durable-undo-refresh",env={...process.env,WORK_STATE_ROOT:root+"/work-state"}
async function command(args,cwd,log){let child=Bun.spawn(args,{cwd,env,stdout:"pipe",stderr:"pipe"});let [stdout,stderr,code]=await Promise.all([new Response(child.stdout).text(),new Response(child.stderr).text(),child.exited]);if(log)await Bun.write(log,stdout+stderr);if(code!==0)throw Error("Phase command failed: "+args[0]+" exit "+code);return stdout}
let phases=[]
try {
 for(let pair=0;pair<3;pair++)for(let arm of pair%2===0?["baseline","candidate"]:["candidate","baseline"]){
  let label=arm+"-"+pair,path=root+"/"+label
  await command(["work","down"],root)
  await rm(root+"/data-active",{recursive:true,force:true});await cp(root+"/data-seed",root+"/data-active",{recursive:true})
  await mkdir(path)
  let started=await command(["work","run",arm],root,path+"/work.log.txt")
  console.log(JSON.stringify({phase:label,stage:"server-started",process:started.trim()}))
  await command(["bun","prepare-timeline.ts",label,arm],root+"/"+arm,path+"/pins.log")
  console.log(JSON.stringify({phase:label,stage:"pins-ready"}))
  await command(["bun","experiments/performance/main-current-preparation/runner-with-redo.ts",path+"/manifest.private.json"],root+"/"+arm,path+"/run.log")
  let rows=await Bun.file(path+"/results/results.json").json()
  if(rows.length!==1||rows[0].status!=="complete"||!rows[0].undoReceipt||!rows[0].redoReceipt)throw Error("Incomplete phase: "+label)
  phases.push({label,status:"complete",results:path+"/results/results.json"})
  await Bun.write(root+"/phases.json",JSON.stringify(phases))
  console.log(JSON.stringify({phase:label,stage:"complete"}))
 }
} finally {await command(["work","down"],root)}
