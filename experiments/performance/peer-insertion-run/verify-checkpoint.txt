import * as Effect from "effect/Effect"
import * as Redacted from "effect/Redacted"
import * as HttpApiClient from "effect/http-api/HttpApiClient"
import * as FetchHttpClient from "effect/http/FetchHttpClient"
import * as HttpClient from "effect/http/HttpClient"
import * as HttpClientRequest from "effect/http/HttpClientRequest"
import {Signatures} from "../../../src/platform/signatures"
import {webCryptoLayer} from "../../../src/platform/web-crypto"
import {encode} from "../../../src/shared/base64"
import {DocumentApi} from "../../../src/shared/document-api"
import {restoreSnapshot} from "../../../src/shared/document-actions"
import {documentMetadata} from "../../../src/shared/document-metadata"
import {frontmatter} from "../../../src/shared/markdown"
import {wikiTargets} from "../../../src/shared/portable-links"
import {performanceFixture} from "../fixture"
let base=new URL("http://127.0.0.1:4470"),document="benchmark-twenty-document-final"
let output=Bun.argv[2]
if(!output)throw Error("Output required")
await Effect.runPromise(Effect.gen(function*(){
 let signatures=yield* Signatures, identity=yield* signatures.account(Redacted.make(encode(new Uint8Array(32).fill(170))))
 let anonymous=yield* HttpApiClient.make(DocumentApi,{baseUrl:base})
 let challenge=yield* anonymous.authentication.challenge({payload:{publicKey:identity.publicKey,purpose:"bearer"}})
 let signature=yield* identity.signAuthentication(challenge,{origin:base.origin,purpose:"bearer"})
 let login=yield* anonymous.authentication.bearerLogin({payload:{id:challenge.id,publicKey:identity.publicKey,signature}})
 let api=yield* HttpApiClient.make(DocumentApi,{baseUrl:base,transformClient:client=>HttpClient.mapRequest(client,HttpClientRequest.bearerToken(Redacted.make(login.token)))})
 let snapshot=yield* api.documents.snapshot({params:{document},query:{}}),replica=yield* restoreSnapshot(snapshot,"checkpoint-verifier")
 let first=yield* api.documents.history({params:{document},query:{after:1,log:snapshot.log,limit:100}})
 let second=yield* api.documents.history({params:{document},query:{after:first.at(-1)?.position??1,log:snapshot.log,limit:100}})
 let history=[...first,...second],source=replica.text(document),header=frontmatter(source)
 let initial=performanceFixture(128).replace("title: Benchmark 128 KB","title: Cohort slides\nmode: present").replace("A responsiveness fixture","[[other-doc]] A responsiveness fixture")
 let exactMarkers=true
 for(let writer=0;writer<20;writer++)for(let round=0;round<6;round++)if(source.split(`\nBENCHMARK_WRITER_${writer}_ROUND_${round}\n`).length!==2)exactMarkers=false
 let row=(yield* api.workspaces.library()).documents.find(row=>row.document===document)
 let report={snapshotPosition:snapshot.position,acceptedCursor:history.at(-1)?.position,historyCount:history.length,contiguous:history.every((fact,index)=>fact.position===index+2),prefixExact:source.startsWith(initial),exactMarkers,projectionExact:row!==undefined&&row.position===snapshot.position&&Bun.deepEquals(row.metadata,documentMetadata(source))&&row.kind==="slides"&&row.excerpt===header.body.replace(/[#*`>\n]/g," ").trim().slice(0,72)&&Bun.deepEquals(row.links??[],wikiTargets(header.body)),kind:row?.kind,links:row?.links,sourceLength:source.length}
 yield* Effect.promise(()=>Bun.write(output,JSON.stringify(report,null,2)))
 if(report.snapshotPosition!==121||report.acceptedCursor!==121||report.historyCount!==120||!report.contiguous||!report.prefixExact||!report.exactMarkers||!report.projectionExact)throw Error("Retained checkpoint verification failed")
}).pipe(Effect.provide(Signatures.layer),Effect.provide(FetchHttpClient.layer),Effect.provide(webCryptoLayer),Effect.timeout("20 seconds")))
