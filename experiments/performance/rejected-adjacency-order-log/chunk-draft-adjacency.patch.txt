+ constructor(buckets?: Map<string, readonly Atom[]>[], insertionOrder = Chunk.empty<string>()) {
+  this.buckets = buckets ?? Array.from({ length: 256 }, () => new Map())
+  this.insertionOrder = insertionOrder
+  if (!buckets) this.ownedBuckets.fill(1)
+ }
+ fork() {

+ }

+ set(id: string, siblings: readonly Atom[]) {

+  }

+  entries.set(id, siblings)
+ }
