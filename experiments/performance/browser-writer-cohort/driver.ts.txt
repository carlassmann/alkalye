  if (!Number.isSafeInteger(pid) || !cpuTime || !Number.isFinite(rssKiB)) throw new Error("Malformed resource bracket")
  return {role:pid===process.pid?"driver":"server",pid,cpuTime,rssKiB}
 })

 return {stage,time:performance.timeOrigin+performance.now(),rows,scope:"process endpoint brackets; excludes browser descendants; CPU not precisely trace-aligned"}
