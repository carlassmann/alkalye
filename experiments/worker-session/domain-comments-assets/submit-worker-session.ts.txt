type Capability = Effect.Success<ReturnType<typeof makeWorkerOpenedCapabilities>>

 function failure(reason: WorkerDocumentFailure["reason"]) { return new WorkerDocumentFailure({ reason, category: "conflict" }) }
 function requireOpen() { return Effect.suspend(() => opened ? Effect.succeed(opened) : Effect.fail(failure("Closed"))) }
 function admitted(current: Opened, epoch: number): Effect.Effect<void, WorkerDocumentFailure> { return current.admissionEpoch === epoch && current.capability.current().canWrite ? Effect.void : Effect.fail(failure("PermissionDenied")) }
 function prepareAdmitted(current: Opened, epoch: number, command: Effect.Effect<{ token: string }, unknown>, declaration: DomainCommand) {
  return Effect.uninterruptible(Effect.gen(function* () {
   yield* admitted(current, epoch)
   let digest = yield* domainCommandDigest(declaration)
   let receipt = yield* command
   current.commandEpochs.set(receipt.token, epoch)
   current.commandDigests.set(receipt.token, digest)
   return receipt
  }))
 }
 function captureComposer<A extends { token: string }, E, R>(current: Opened, epoch: number, capture: Effect.Effect<A, E, R>) {
  return Effect.gen(function* () {
   yield* admitted(current, epoch)
   let composer = yield* capture
   yield* admitted(current, epoch).pipe(Effect.tapError(() => current.capability.discussion.release(composer.token)))
   current.composerEpochs.set(composer.token, epoch)
   return composer
  })
 }
 function notice(current: Opened) {
  let snapshot = current.capability.current()
  return { revision: snapshot.revision, admissionEpoch: current.admissionEpoch, metadata: snapshot.projection?.metadata, canWrite: snapshot.canWrite, connection: snapshot.connection, status: snapshot.status }
 }
 function checked<A, E, R>(work: Effect.Effect<A, E, R>) {
  return work.pipe(Effect.tapError(error => Effect.tryPromise({ try: async () => {
   let message = error instanceof Error ? error.message : typeof error === "string" ? error : ""
   let bytes = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(message))
   let fingerprint = Array.from(new Uint8Array(bytes), value => value.toString(16).padStart(2, "0")).join("")
   let tag = typeof error === "object" && error !== null && "_tag" in error && typeof error._tag === "string" ? error._tag : undefined
   let frames = error instanceof Error ? error.stack?.split("\n").slice(1, 13).map(frame => frame.trim().replace(/\(.*/, "").slice(0, 120)) : undefined
   workerBootMark("command:error", JSON.stringify({ frames, class: error instanceof Error ? error.name : typeof error, tag, reason: error instanceof WorkerDocumentFailure ? error.reason : message === "Replica changed while being copied" ? "ReplicaChangedWhileCopied" : message === "Replica changed while being read" ? "ReplicaChangedWhileRead" : undefined, fingerprint }))
  }, catch: () => undefined }).pipe(Effect.catch(() => Effect.void))), Effect.ensuring(Effect.sync(() => opened?.activity.publish())), Effect.mapError(error => error instanceof WorkerDocumentFailure ? error : new WorkerDocumentFailure({ reason: error instanceof PermissionDenied ? "PermissionDenied" : "SessionFailure", category: documentFailureCategory(error) })))
 }

    let options = { document: input.document, subject: input.subject, cooperativeWork: input.cooperativeWork, transport: {

     history: (...arguments_: Parameters<typeof transport.history>) => account.request(transport.history(...arguments_)),

Effect.fail(new EventJournal.EventJournalError({ method: "readAfter", cause: "fixture open failure" })) : journalStorage.readAfter(...arguments_), cache: Effect.suspend(() => openFailure ? Effect.fail(new EventJournal.EventJournalError({ method: "cache", cause: "fixture open failure" })) : journalStorage.cache) })

   if (input.admission.subject !== current.subject || input.admission.epoch <= current.admissionEpoch) return yield* Effect.fail(failure("PermissionDenied"))

  })))),
  ExecuteAdmitted: () => Effect.fail(failure("Rejected")),
  ExecuteDomain: input => execute(requireOpen().pipe(Effect.flatMap(current => Effect.uninterruptibleMask(restore => Effect.gen(function* () {
   let intent = input.intent

    return Effect.gen(function* () {
     let selected: DomainOutcome | undefined
     let created = false
     yield* current.storage.modify(key, stored => {
      if (stored !== undefined) {
       let previous = Schema.decodeUnknownSync(DomainOutcomeSchema)(stored)
       if (!matchesDomain(previous.intent, intent)) throw failure("Rejected")
       if (!finish || previous.status !== "unknown") { selected = previous; return stored }

      selected = next
      return next
     })
     if (!selected) return yield* Effect.fail(failure("AppliedUnknown"))
     return { outcome: selected, created }
    })
   }
   function refuse() { let outcome: DomainOutcome = { intent, status: "not-applied" }; return record(outcome).pipe(Effect.map(recorded => recorded.outcome)) }

   }

   if (!basis || basis.epoch !== intent.epoch || basis.sequence !== intent.sequence) return yield* refuse()

   if (!claimed.created) return claimed.outcome

   current.commandEpochs.delete(intent.token)
   current.commandDigests.delete(intent.token)
   yield* current.capability.commands.release(intent.token)
   if (Exit.isFailure(completed)) return started
   let outcome: DomainOutcome = { intent, status: "applied", action: completed.value.action }
   return (yield* record(outcome, true)).outcome
  }))))),
  RecoverDomain: input => execute(requireOpen().pipe(Effect.flatMap(current => Effect.gen(function* () {
   let stored = yield* current.storage.get(domainOutcomeKey(current.subject, current.document, input.request))
   if (stored === undefined) return { found: false }

  Observe: () => Stream.unwrap(Effect.acquireRelease(

function sourceAfterRanges(source: string, ranges: readonly { start: number; remove: number; text: string }[]) {
 let parts: string[] = []
 let position = 0
 for (let range of ranges) {
  if (range.start < position || range.start + range.remove > source.length) return
  parts.push(source.slice(position, range.start), range.text)
  position = range.start + range.remove
 }
 parts.push(source.slice(position))
 return parts.join("")
}
