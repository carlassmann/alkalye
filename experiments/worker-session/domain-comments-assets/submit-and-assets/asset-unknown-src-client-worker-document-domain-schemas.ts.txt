import * as Schema from "effect/Schema"
export let DomainIntentSchema = Schema.Struct({ subject: Schema.String, document: Schema.String, request: Schema.String, token: Schema.String, epoch: Schema.String, sequence: Schema.Natural, admissionEpoch: Schema.Natural, basisDigest: Schema.String, commandDigest: Schema.String, intentDigest: Schema.String })
export let DomainOutcomeSchema = Schema.Struct({ intent: DomainIntentSchema, status: Schema.Literals(["applied", "not-applied", "unknown"]), action: Schema.optional(Schema.String) })

export let DomainPatchSchema = Schema.Struct({ title: Schema.optional(Schema.String), path: Schema.optional(Schema.String), tags: Schema.optional(Schema.Array(Schema.String)), pinned: Schema.optional(Schema.Boolean), archived: Schema.optional(Schema.Boolean), commentsDisabled: Schema.optional(Schema.Boolean) })
export let DomainCommandSchema = Schema.Union([Schema.Struct({ type: Schema.Literal("properties"), patch: DomainPatchSchema }), Schema.Struct({ type: Schema.Literal("presentation"), enabled: Schema.Boolean }), Schema.Struct({ type: Schema.Literal("present"), index: Schema.Natural }), Schema.Struct({ type: Schema.Literal("commentResolve"), thread: Schema.String, resolved: Schema.Boolean }), Schema.Struct({ type: Schema.Literal("commentSubmit"), composer: Schema.String, body: Schema.String }), Schema.Struct({ type: Schema.Literal("assetAttach"), selection: Schema.String, files: Schema.Array(Schema.Struct({ reference: Schema.String, name: Schema.String })).check(Schema.isMinLength(1), Schema.isMaxLength(8)) })])
export let QueuedDomainSchema = Schema.Struct({ subject: Schema.String, document: Schema.String, request: Schema.String, epoch: Schema.String, sequence: Schema.Natural, admissionEpoch: Schema.Natural, command: DomainCommandSchema, commandDigest: Schema.String, basisDigest: Schema.String, queuedDigest: Schema.String })
export let PendingDomainSchema = Schema.Struct({ queued: QueuedDomainSchema, source: Schema.String, phase: Schema.Literals(["NeverDispatched", "Preparing", "Prepared", "STARTED", "Known"]), intent: Schema.optional(DomainIntentSchema), outcome: Schema.optional(DomainOutcomeSchema) })

export let DomainRecoverySchema = Schema.Struct({ format: Schema.Literal("worker-domain-recovery/v1"), subject: Schema.String, document: Schema.String, source: Schema.String, sourceDigest: Schema.String, corrupt: Schema.Boolean, bindings: Schema.Array(Schema.Struct({ request: Schema.String, queuedDigest: Schema.String, intentDigest: Schema.optional(Schema.String) })) })
