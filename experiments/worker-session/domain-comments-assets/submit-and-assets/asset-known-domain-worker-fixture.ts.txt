let Pins = Schema.Array(Schema.Struct({ path: Schema.String, sha256: Schema.String }))
function verifyBodies() {
 return Effect.tryPromise({ try: async () => {
  let pins = Schema.decodeUnknownSync(Pins)(await (await fetch("/__tests/asset-pins", { cache: "no-store" })).json())
  let urls = performance.getEntriesByType("resource").map(entry => entry.name).filter(url => new URL(url).pathname.endsWith(".js"))
  for (let url of new Set([...urls, ...workerUrls])) {
   let path = new URL(url).pathname
   let expected = pins.find(pin => pin.path === path)
   if (!expected) throw new Error("Unpinned script path")
   let response = await fetch(url, { cache: "no-store" })
   if (!response.ok) throw new Error("Script fetch failed")
   let bytes = await crypto.subtle.digest("SHA-256", await response.arrayBuffer())
   let digest = Array.from(new Uint8Array(bytes), byte => byte.toString(16).padStart(2, "0")).join("")
   if (digest !== expected.sha256) throw new Error("Served script differs from frozen fixture build")
  }
  check(workerUrls.length > 0 && urls.length > 0, "requested main/worker scripts re-fetched against frozen served body hashes before invocation")
 }, catch: error => error })
}
