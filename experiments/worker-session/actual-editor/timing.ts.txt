import { assert, browser, evaluate, wait } from "./tests/browser-driver"
let url = process.argv[2]
let output = process.argv[3]
if (!url || !output) throw Error("URL/output required")
let session = "actual-worker-timing-" + Date.now()
let api = "window.workerEditorProof"
let source = 'document.querySelector("alkalye-editor").source'
let rows: unknown[] = []
try {
	browser(session, "set", "viewport", "1280", "800")
	browser(session, "open", url)
	wait(session, "document.body.dataset.ready==='true'||!!document.body.dataset.failure")
	assert(evaluate(session, "!document.body.dataset.failure"), "Actual native editor fixture ready")
	let multicursor = new URL(url).searchParams.has("cursors")
	for (let trial = 0; trial < 3; trial++) {
		evaluate(session, `(()=>{window.inputExpected=${api}.inspect().source;${source}.focus();let at=${multicursor ? "window.inputExpected.indexOf('TOKEN')" : "window.inputExpected.length"};${source}.setSelectionRange(at,at+${multicursor ? "5" : "0"});return true})()`)
		if (multicursor) {
			browser(session, "press", "Control+Shift+l")
			browser(session, "press", "ArrowRight")
			assert(evaluate(session, `${api}.inspect().busy`), "Actual multicursor selections are active")
		}
		let geometry = evaluate(session, `(()=>{let native=${source}.element,host=native.parentElement;return {height:host.clientHeight,scrollHeight:host.scrollHeight,nativeCharacters:native.value.length,sourceCharacters:${api}.inspect().source.length,readonly:native.readOnly}})()`)
		assert(evaluate(session, `(()=>{let native=${source}.element,host=native.parentElement;return host.clientHeight===800&&host.scrollHeight>host.clientHeight&&native.value.length<32768&&!native.readOnly})()`), "Bounded actual editor geometry precedes profiling")
		let marker = ["X", "Y", "Z"][trial]
		if (!marker) throw Error("Missing marker")
		evaluate(session, `(()=>{window.inputExpected=${multicursor ? `window.inputExpected.replaceAll('TOKEN','TOKEN'+${JSON.stringify(marker)})` : `window.inputExpected+${JSON.stringify(marker)}`};return true})()`)
		browser(session, "profiler", "start", "--categories", "devtools.timeline,disabled-by-default-devtools.timeline,toplevel,blink.user_timing,v8")
		evaluate(session, `(()=>{let controller=new AbortController(),state=window.inputProof={trusted:0,before:0,input:null,microtask:0,raf2:0,controller};performance.clearMarks();document.addEventListener('beforeinput',event=>{state.before=performance.now();state.trusted+=Number(event.isTrusted);performance.mark('worker:input:start');queueMicrotask(()=>{state.microtask=performance.now()-state.before;requestAnimationFrame(()=>requestAnimationFrame(()=>{state.raf2=performance.now()-state.before;performance.mark('worker:input:raf2')}))})},{capture:true,signal:controller.signal});window.addEventListener('input',()=>{state.input=performance.now()-state.before;},{signal:controller.signal});return {heap:performance.memory?.usedJSHeapSize,native:${source}.element.value.length,nodes:document.querySelector('#highlight').querySelectorAll('*').length}})()`)
		browser(session, "keyboard", "inserttext", marker)
		wait(session, `${api}.inspect().source===window.inputExpected&&${api}.inspect().pending===0&&window.inputProof.raf2>0`)
		assert(evaluate(session, `${api}.flush().then(value=>value===window.inputExpected)`), "Native queue drains and durable source equals exact input")
		let observation = evaluate(session, `(()=>{let state=window.inputProof;performance.mark('worker:input:durable');state.controller.abort();return {trusted:state.trusted,inputMs:state.input,microtaskMs:state.microtask,raf2Ms:state.raf2,durableMs:performance.now()-state.before,sourceCharacters:window.inputExpected.length,pending:${api}.inspect().pending,failures:${api}.inspect().failures,heap:performance.memory?.usedJSHeapSize,nativeCharacters:${source}.element.value.length,nodes:document.querySelector('#highlight').querySelectorAll('*').length}})()`)
		let aftermath = evaluate(session, `new Promise(resolve=>setTimeout(()=>{performance.mark('worker:input:aftermath');resolve({pending:${api}.inspect().pending,failures:${api}.inspect().failures,heap:performance.memory?.usedJSHeapSize})},3000))`)
		let trace = output + "-" + trial + ".trace.json"
		browser(session, "profiler", "stop", trace)
		rows.push({ trial, multicursor, geometry, observation, aftermath, trace })
		await Bun.write(output, JSON.stringify(rows, null, 2))
	}
	evaluate(session, `${api}.dispose()`)
} finally { browser(session, "close") }
