interface Inspection { source?: string; readonly?: boolean; pending: number; queueSource: string; recovery?: string; observations: number; failures: string[]; clientExits?: string[]; signals?: string[]; mutationCalls?: number; failureReasons?: string[] }
declare global { interface Window { openedNative?: { stage: string; assertions: string[]; failure?: string; inspect(): Inspection; accepted(): Promise<{ source: string; position: number }>; undo(): void; fault(): Promise<void>; hold(): Promise<void>; interrupt(): Promise<void>; failClose(): Promise<void> } } }
let assertions: string[] = [], failures: string[] = [], failureReasons: string[] = []
function assert(value: unknown, label: string) { if (!value) throw Error(label); assertions.push(label) }
let program = Effect.scoped(Effect.gen(function* () {
 let account = yield* bootstrapBrowserAccount(assert)
 let documentId = "opened-native-" + crypto.randomUUID()
 let source = "---\ntitle: Worker fixture\n---\n# Native writer\nUnicode 🦊 café.\n"
 yield* Effect.promise(async () => {
  let response = await fetch("/__tests/seed", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ subject: account.subject, document: documentId, source }) })
  assert(response.ok, "real HTTP fixture seeded without main Replica")
 })
 let owner = yield* Scope.Scope
 let workerScope = yield* Scope.fork(owner)
 let worker = yield* createWorkerSessionClient(new URL("/worker.js", location.href).href).pipe(Scope.provide(workerScope))
 let queue: Effect.Success<ReturnType<typeof makeEditorQueue>> | undefined
 let recovery: HTMLTextAreaElement | undefined
 let channel = "opened-native-activity-" + crypto.randomUUID()
 let activity = yield* forwardWorkerActivity(channel, () => {
  let state = queue?.state()
  return recovery !== undefined || !!state && (state.pending.length > 0 || state.commandFrozen || state.failure !== undefined)
 }, () => worker.Protection())
 let signals: string[] = []
 let signalChannel = yield* Effect.acquireRelease(Effect.sync(() => new BroadcastChannel(channel + ":fixture")), value => Effect.sync(() => value.close()))
 signalChannel.onmessage = event => { if (typeof event.data === "string") signals.push(event.data) }
 function waitSignal(name: string) { return Effect.sync(() => signals.includes(name)).pipe(Effect.flatMap(found => found ? Effect.void : Effect.fail("SignalPending")), Effect.retry(Schedule.spaced("20 millis")), Effect.timeout("5 seconds")) }
 let durableSource = source
 let input = { document: documentId, workspace: "personal-" + account.subject, subject: account.subject, origin: location.origin, online: true, activityChannel: channel, cooperativeWork: true }
 yield* worker.Open(input)
 let opened = yield* worker.Read().pipe(Effect.flatMap(value => value.projection ? Effect.succeed(value) : Effect.fail("NotReady")), Effect.retry(Schedule.spaced("20 millis")), Effect.timeout("10 seconds"))
 assert(opened.projection?.document.content === source, "worker opened projection exact")
 let observations = 0
 let observed = yield* Deferred.make<void>()
 yield* worker.Observe().pipe(Stream.runForEach(value => Effect.gen(function* () {
  if (value.type === "document") { observations++; yield* Deferred.succeed(observed, undefined) }
 })), Effect.forkScoped({ startImmediately: true }))
 yield* Deferred.await(observed).pipe(Effect.timeout("5 seconds"))
 let initial = yield* worker.Begin()
 let queueScope = yield* Scope.fork(owner)
 let clientExits: string[] = []
 let mutationCalls = 0
 queue = yield* makeEditorQueue(worker, initial, exit => clientExits.push(Exit.isFailure(exit) ? Cause.hasInterrupts(exit.cause) ? "interrupted" : "failed" : "success"), () => { mutationCalls++ }).pipe(Scope.provide(queueScope))
 let currentQueue = queue
 let tasks = yield* FiberSet.make()
 let fork = yield* FiberSet.runtime(tasks)()
 let editor = new EditorElement()
 function report(error: unknown) {
  failures.push(String(error))
  if (error instanceof WorkerDocumentFailure) failureReasons.push(error.reason)
  if (!recovery) {
   recovery = document.createElement("textarea")
   recovery.id = "recovery-draft"
   recovery.readOnly = true
   recovery.value = editor.source?.value ?? currentQueue.state().source
   document.body.append(recovery)
  }
  activity.uncertain()
 }
 let bindings = workerEditorBindings(editor, currentQueue, effect => fork(effect.pipe(Effect.ensuring(Effect.sync(activity.update)))), report)
 editor.configure({ authority: "worker", content: initial.source, atoms: [], editable: true, preferences: editorPreferences({ autocomplete: false }), documents: () => [], isDrawing: () => false, openDrawing() {}, openDocument() {}, comments: () => [], createLinkedDocument: async () => "unused", report, change: bindings.change, clipboard: { capture: (start,end,expected) => currentQueue.captureClipboard(start,end,expected), resolve: token => currentQueue.resolveClipboard(token), release: token => currentQueue.releaseClipboard(token) }, undo: bindings.undo, select: bindings.resumeUndo, requestAsset() {}, attach() {}, command() {}, find() {} })
 document.body.append(editor)
 yield* Effect.addFinalizer(() => Effect.sync(() => { editor.remove(); recovery?.remove() }))
 let done = yield* Deferred.make<void>()
 window.openedNative = {
  stage: "ready", assertions,
  inspect() { let state = currentQueue.state(); return { source: editor.source?.value, readonly: editor.source?.readOnly, pending: state.pending.length, queueSource: state.source, recovery: recovery?.value, observations, failures, clientExits, signals, mutationCalls, failureReasons } },
  accepted() { return Effect.runPromise(Effect.gen(function* () {
   let view = yield* worker.Read().pipe(Effect.flatMap(value => value.status.pending === 0 && value.status.unsaved === 0 && value.connection.status === "synced" ? Effect.succeed(value) : Effect.fail("NotAccepted")), Effect.retry(Schedule.spaced("20 millis")), Effect.timeout("10 seconds"))
   let response = yield* Effect.promise(() => fetch("/__tests/accepted?document=" + encodeURIComponent(documentId) + "&subject=" + encodeURIComponent(account.subject)))
   let accepted = yield* Effect.promise(() => response.json())
   assert(response.ok && typeof accepted.source === "string" && accepted.source === editor.source?.value && accepted.source === view.projection?.document.content, "native local source exactly accepted by real server")
   durableSource = accepted.source
   return accepted
  })) },
  undo() { bindings.undo(false) },
  fault() { return Effect.runPromise(worker.FixtureFault({ openFailure:false, writeFailure:true })) },
  hold() { return Effect.runPromise(Effect.gen(function* () { signalChannel.postMessage("hold"); yield* waitSignal("armed"); assert(signals.includes("armed"), "real worker journal hold armed") })) },
  interrupt() { return Effect.runPromise(Effect.gen(function* () {
   yield* Effect.sync(() => signals.some(value => value.startsWith("held:"))).pipe(Effect.flatMap(found => found ? Effect.void : Effect.fail("HoldPending")), Effect.retry(Schedule.spaced("20 millis")), Effect.timeout("5 seconds"))
   let held = signals.find(value => value.startsWith("held:"))
   let id = held?.slice(5)
   assert(!!id && signals.filter(value => value.startsWith("held:")).length === 1, "exactly one native-marked held journal write has an identity")
   assert(currentQueue.state().pending.some(edit => edit.delivery === "sending"), "actual native mutation RPC sending while real journal held")
   let native = editor.source?.value
   let dispatched = mutationCalls
   yield* Scope.close(queueScope, Exit.void)
   assert(clientExits.includes("interrupted"), "actual ReplaceQueued client fiber exit interrupted")
   yield* Effect.sync(() => recovery?.value === native && editor.source?.readOnly).pipe(Effect.flatMap(found => found ? Effect.void : Effect.fail("RecoveryPending")), Effect.retry(Schedule.spaced("20 millis")), Effect.timeout("5 seconds"))
   assert(recovery?.value === native && editor.source?.readOnly, "interrupted native FIFO retains real native draft readonly")
   assert(currentQueue.state().pending.some(edit => edit.delivery === "unknown") && failureReasons.includes("AppliedUnknown"), "interrupted sending RPC marked unknown without blind replay")
   signalChannel.postMessage("release-fail")
   yield* waitSignal("released")
   yield* Effect.sync(() => signals.includes("interrupted:" + id) || signals.includes("resumed:" + id) && signals.includes("write-failed:" + id)).pipe(Effect.flatMap(found => found ? Effect.void : Effect.fail("HoldOutcomePending")), Effect.retry(Schedule.spaced("20 millis")), Effect.timeout("5 seconds"))
   assert(signals.includes("interrupted:" + id) || signals.includes("resumed:" + id) && signals.includes("write-failed:" + id), "held journal write identity has an explicit interrupt or resumed refusal outcome")
   let closed = yield* worker.Close().pipe(Effect.exit)
   let cause = Exit.isFailure(closed) ? Cause.squash(closed.cause) : undefined
   assert(cause instanceof WorkerDocumentFailure && cause.reason === "SessionFailure", "interrupted suffix Close reports journal session failure")
   yield* Scope.close(workerScope, Exit.void)
   let reopenedScope = yield* Scope.fork(owner)
   let fresh = yield* createWorkerSessionClient(new URL("/worker.js", location.href).href).pipe(Scope.provide(reopenedScope))
   yield* Effect.promise(() => fetch("/__tests/network?blocked=1&reset=1"))
   yield* fresh.Open({ ...input, online: false, activityChannel: channel + "-reopen" })
   let reopened = yield* fresh.Read().pipe(Effect.flatMap(value => value.projection ? Effect.succeed(value) : Effect.fail("ReopenPending")), Effect.retry(Schedule.spaced("20 millis")), Effect.timeout("5 seconds"))
   assert(native?.startsWith(durableSource) && native !== durableSource && reopened.projection?.document.content === durableSource && durableSource.endsWith("PREFIX"), "offline independent worker reopens exact native committed prefix excluding interrupted suffix")
   let network = yield* Effect.promise(async () => (await fetch("/__tests/network?blocked=1")).json())
   assert(network.apiRequests === 0, "offline prefix reopen makes zero HTTP API requests")
   yield* fresh.Close()
   yield* Scope.close(reopenedScope, Exit.void)
   assert(mutationCalls === dispatched, "interrupted native suffix is never dispatched again")
   assert(editor.source?.value === native && recovery?.value === native, "independent reopen never replaces retained native unknown draft")
   editor.remove()
   let dirty = new Event("beforeunload", { cancelable: true }); window.dispatchEvent(dirty)
   assert(dirty.defaultPrevented && !activity.closedCleanly(), "fixture-owned recovery retains unload protection after disposal")
   if (window.openedNative) window.openedNative.stage = "passed"
   yield* Deferred.succeed(done, undefined)
  })) },
  failClose() { return Effect.runPromise(Effect.gen(function* () {
   let native = editor.source?.value
   assert(!!native && recovery?.value === native && editor.source?.readOnly, "refused actual native edit retains full source in recovery owner")
   assert(currentQueue.state().pending.length > 0 && currentQueue.state().failure !== undefined, "failed native FIFO suffix remains recoverable")
   let dirty = new Event("beforeunload", { cancelable:true }); window.dispatchEvent(dirty)
   assert(dirty.defaultPrevented, "actual native recovery retains unload protection")
   let closed = yield* worker.Close().pipe(Effect.exit)
   assert(Exit.isFailure(closed), "failed native durable Close remains failed")
   yield* Scope.close(workerScope, Exit.void)
   editor.remove()
   assert(recovery?.value === native, "native recovery owner survives editor and worker disposal")
   assert(!activity.closedCleanly(), "dirty owner refuses clean-close acknowledgement")
   if (window.openedNative) window.openedNative.stage = "passed"
   yield* Deferred.succeed(done, undefined)
  })) },
 }
 yield* Deferred.await(done)
})).pipe(Effect.provide(FetchHttpClient.layer), Effect.catchCause(cause => Effect.sync(() => {
 if (window.openedNative) { window.openedNative.stage = "failed"; window.openedNative.failure = String(cause) }
 else window.openedNative = { stage:"failed", assertions, failure:String(cause), inspect:()=>({pending:0,queueSource:"",observations:0,failures}), accepted:async()=>({source:"",position:0}),undo(){},fault:async()=>{},hold:async()=>{},interrupt:async()=>{},failClose:async()=>{} }
})))
Effect.runFork(program)
