import { assert, browser, evaluate, wait } from "./tests/browser-driver"
let url = process.argv[2]
if (!url) throw Error("URL required")
let session = "actual-worker-undo-" + Date.now()
let api = "window.workerEditorProof"
let source = 'document.querySelector("alkalye-editor").source'
try {
	browser(session, "open", url)
	wait(session, "document.body.dataset.ready==='true'||!!document.body.dataset.failure")
	assert(evaluate(session, "!document.body.dataset.failure"), "Actual native undo fixture is ready")
	let original = evaluate(session, `${api}.inspect().source`)
	if (typeof original !== "string") throw Error("Original source missing")
	evaluate(session, `${source}.focus();${source}.setSelectionRange(${original.length},${original.length})`)
	browser(session, "press", "Control+z")
	wait(session, `${api}.inspect().source===${JSON.stringify(original)}&&!${api}.inspect().readonly`)
	assert(true, "Nothing-to-undo outcome unfreezes native input")
	browser(session, "keyboard", "inserttext", "😀")
	browser(session, "press", "Control+z")
	wait(session, `${api}.inspect().source===${JSON.stringify(original)}&&!${api}.inspect().readonly&&${api}.inspect().pending===0`)
	assert(evaluate(session, `${api}.inspect().start===${original.length}`), "Undo resolves deleted own emoji caret to nearest surviving predecessor")
	browser(session, "press", "Control+Shift+z")
	let appended = original + "😀"
	wait(session, `${api}.inspect().source===${JSON.stringify(appended)}&&!${api}.inspect().readonly&&${api}.inspect().pending===0`)
	assert(evaluate(session, `${api}.flush()`) === appended, "Redo preserves exact local durable source")
	let alpha = appended.indexOf("alpha")
	evaluate(session, `${source}.focus();${source}.setSelectionRange(${alpha},${alpha+5})`)
	browser(session, "press", "Control+d")
	browser(session, "press", "ArrowRight")
	browser(session, "keyboard", "inserttext", "X")
	let multi = appended.replaceAll("alpha", "alphaX")
	wait(session, `${api}.inspect().source===${JSON.stringify(multi)}&&${api}.inspect().pending===0`)
	browser(session, "press", "Control+z")
	wait(session, `${api}.inspect().source===${JSON.stringify(appended)}&&!${api}.inspect().readonly&&${api}.inspect().pending===0`)
	assert(evaluate(session, `${api}.inspect().busy`), "Undo retains multicursor selections through resolved candidate adoption")
	browser(session, "press", "Escape")
	evaluate(session, `${api}.peer()`)
	evaluate(session, `${api}.merge()`)
	let peer = "PEER\n" + appended
	wait(session, `${api}.inspect().source===${JSON.stringify(peer)}`)
	browser(session, "press", "Control+Shift+z")
	wait(session, `${api}.inspect().source===${JSON.stringify(peer)}&&!${api}.inspect().readonly&&${api}.inspect().pending===0`)
	assert(true, "New same-subject peer edit invalidates old redo")
	evaluate(session, `${source}.focus();${source}.setSelectionRange(${peer.length},${peer.length})`)
	browser(session, "keyboard", "inserttext", "Q")
	browser(session, "press", "Control+z")
	wait(session, `${api}.inspect().source===${JSON.stringify(peer)}&&!${api}.inspect().readonly&&${api}.inspect().pending===0`)
	assert(true, "Undo retains previously merged peer prefix")
	browser(session, "press", "Control+Shift+z")
	let redone = peer + "Q"
	wait(session, `${api}.inspect().source===${JSON.stringify(redone)}&&!${api}.inspect().readonly&&${api}.inspect().pending===0`)
	assert(evaluate(session, `${api}.flush()`) === redone, "Redo keeps previously merged peer prefix and exact local journal")
	assert(evaluate(session, `${api}.inspect().failures.length===0`), "Native undo delegation reports no failures")
	browser(session, "reload")
	wait(session, `document.body.dataset.ready==='true'&&${api}.inspect().source===${JSON.stringify(redone)}&&!${api}.inspect().readonly`)
	assert(true, "Undo/redo source survives actual worker/native reload")
	evaluate(session, `${api}.dispose()`)
	assert(evaluate(session, "document.body.dataset.closed==='true'"), "Undo lifecycle disposes scoped resources")
} finally { browser(session, "close") }
