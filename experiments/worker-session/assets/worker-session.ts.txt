import * as Context from "effect/Context"
import * as Cause from "effect/Cause"
import * as Effect from "effect/Effect"
import * as Exit from "effect/Exit"
import * as Layer from "effect/Layer"
import * as Scope from "effect/Scope"
import * as Semaphore from "effect/Semaphore"
import * as Stream from "effect/Stream"
import { indexedDbJournalLayer } from "./src/platform/indexeddb-journal"
import { DocumentLifecycle } from "./src/platform/document-lifecycle"
import { DocumentSession, documentDatabaseName } from "./src/shared/document-session"
import { makeDocumentTransport } from "./src/shared/document-transport"
import { documentFailureCategory } from "./src/shared/document-observability"
import { UndoUnavailable } from "./src/shared/document-errors"
import { workerBlobServices, workerAssets } from "./worker-assets"
import { DocumentDiscussion } from "./src/client/document-discussion"
import { selectionAnchor, commentRanges } from "./src/shared/documents"
import { webCryptoLayer } from "./src/platform/web-crypto"
import { PermissionDenied } from "./src/shared/document-actions"

import type { TextReplacement } from "./src/shared/crdt"
import type { Atom } from "./src/shared/model"
import { WorkerDocumentFailure, WorkerDocumentRpc } from "./worker-rpc"

let workerLifecycle = Layer.succeed(DocumentLifecycle, {
	changes: Stream.never,
	hidden: Stream.never,
	protect: () => Effect.succeed({ update() {} }),
})

export let workerSessionHandlers = WorkerDocumentRpc.toLayer(
	Effect.gen(function* () {
		let owner = yield* Scope.Scope
		let commands = yield* Semaphore.make(1)
		let opened: {
			document: string
			scope: Scope.Closeable
			assets: Effect.Success<ReturnType<typeof workerAssets>>
            discussion: Context.Service.Shape<typeof DocumentDiscussion>
            session: Context.Service.Shape<typeof DocumentSession>
		} | undefined
		let undoReceipts = new Map<string, {
			request: string; epoch: string; sequence: number; redo: boolean;
			status: "applied" | "not-applied" | "unknown"; action?: string; failure?: string;
		}>()
		let handles = new Map<string, Atom[]>()
		let clipboard = new Map<string, { document: string; start: ClipboardBoundary; end: ClipboardBoundary; selected: string }>()
		let commentsWritable = false
        let assetsWritable = false
        let commentComposers = new Map<string, Effect.Success<ReturnType<Context.Service.Shape<typeof DocumentDiscussion>["prepareComment"]>>>()
        let sequence = 0
		let epoch = 0
		let basis: { epoch: string; sequence: number; replica: Replica } | undefined
		let candidate: { epoch: string; priorEpoch: string; priorSequence: number; replica: Replica } | undefined
		function fail(reason: WorkerDocumentFailure["reason"]) {
			return new WorkerDocumentFailure({ reason, category: "conflict" })
		}
		function requireSession() {
			return Effect.suspend(() => opened ? Effect.succeed(opened) : Effect.fail(fail("Closed")))
		}
		function requireCapacity() {
			return Effect.suspend(() => handles.size < 8 ? Effect.void : Effect.fail(fail("HandleCapacity")))
		}
		function read() {
			return Effect.gen(function* () {
				yield* requireCapacity()
				let current = yield* requireSession()
				return yield* Effect.try(() => {
					let handle = String(++sequence)
					let atoms = current.session.replica.visible(current.document)
					let source = current.session.replica.text(current.document)
					handles.set(handle, atoms)
					return {
						handle,
						revision: current.session.version(),
						source,
						unsaved: current.session.unsaved(),
						pending: current.session.pending().length,
						rejected: current.session.rejected().length,
						...(basis ? { epoch: basis.epoch, sequence: basis.sequence, localSource: basis.replica.text(current.document) } : {}),
					}
				})
			})
		}
		function execute<A, E, R>(work: Effect.Effect<A, E, R>) {
			return commands.withPermit(work).pipe(Effect.mapError(error =>
				error instanceof WorkerDocumentFailure ? error : error instanceof PermissionDenied ? fail("PermissionDenied") : new WorkerDocumentFailure({
					reason: "SessionFailure", category: documentFailureCategory(error),
				}),
			), Effect.catchDefect(() => Effect.fail(new WorkerDocumentFailure({
				reason: "SessionFailure", category: "failure",
			}))))
		}
		return WorkerDocumentRpc.of({
			Open: input => execute(Effect.uninterruptibleMask(restore => Effect.gen(function* () {
				if (opened) return yield* Effect.fail(fail("AlreadyOpen"))
				let scope = yield* Scope.fork(owner)
				return yield* restore(Effect.gen(function* () {
                    let { account, blobs } = yield* workerBlobServices(input.document, input.subject, input.origin, scope)
                    let transport = yield* makeDocumentTransport(input.origin, input.subject)
                    let options = { ...input, transport: {
                        ...transport,
                        snapshot: (document: string) => account.request(transport.snapshot(document)),
                        action: (...arguments_: Parameters<typeof transport.action>) => account.request(transport.action(...arguments_)),
                        chain: (...arguments_: Parameters<typeof transport.chain>) => account.request(transport.chain(...arguments_)),
                        history: (...arguments_: Parameters<typeof transport.history>) => account.request(transport.history(...arguments_)),
                        withdraw: (...arguments_: Parameters<typeof transport.withdraw>) => account.request(transport.withdraw(...arguments_)),
                        create: (...arguments_: Parameters<typeof transport.create>) => account.request(transport.create(...arguments_)),
                        commit: (action: Parameters<typeof transport.commit>[0]) => blobs.upload(action).pipe(Effect.andThen(account.request(transport.commit(action)))),
                    } }
                    let context = yield* Layer.buildWithScope(DocumentSession.layer(options).pipe(
						Layer.provide(indexedDbJournalLayer(documentDatabaseName(options))),
						Layer.provide(workerLifecycle),
					), scope)
					let session = Context.get(context, DocumentSession)
                    commentsWritable = input.commentsWritable ?? false
                    let discussionContext = yield* Layer.buildWithScope(DocumentDiscussion.layer({
                        document: input.document, author: session.replica.author,
                        canWrite: () => commentsWritable,
                    }).pipe(Layer.provide(Layer.succeed(DocumentSession, session)), Layer.provide(webCryptoLayer)), scope)
                    assetsWritable = input.assetsWritable ?? false
                    let assets = yield* workerAssets(input.document, session, blobs, () => assetsWritable, scope)
                    opened = { document: input.document, scope, session, assets, discussion: Context.get(discussionContext, DocumentDiscussion) }
					return yield* read()
				})).pipe(Effect.onExit(exit => Exit.isSuccess(exit) ? Effect.void : Effect.sync(() => {
					opened = undefined
					basis = undefined
					candidate = undefined
					handles.clear()
				clipboard.clear()
                commentComposers.clear()
				}).pipe(Effect.andThen(Scope.close(scope, exit)))))
			}))),
			Read: () => execute(read()),
            Assets: () => execute(Effect.gen(function* () { let current = yield* requireSession(); return current.assets.current() })),
            CaptureAssetSelection: input => execute(Effect.gen(function* () {
                let current = yield* requireSession()
                let local = basis
                if (!local || local.epoch !== input.epoch || local.sequence !== input.sequence) return yield* Effect.fail(fail("BasisUnavailable"))
                return { token: yield* Effect.try({ try: () => current.assets.capture(local.replica, input.start, input.end, input.inline), catch: error => error }) }
            })),
            PrepareAttachment: input => execute(Effect.gen(function* () { let current = yield* requireSession(); return { token: yield* Effect.try({ try: () => current.assets.prepare(input.files, input.selection), catch: error => error }) } })),
            Attach: input => execute(Effect.gen(function* () { let current = yield* requireSession(); let action = yield* current.assets.attach(input.token); return { action: action.id } })),
            PrepareAssetChange: input => execute(Effect.gen(function* () { let current = yield* requireSession(); return { token: yield* Effect.try({ try: () => current.assets.prepareChange(input.kind, input.asset, input.name, input.selection), catch: error => error }) } })),
            ReadAsset: input => execute(Effect.gen(function* () { let current = yield* requireSession(); return yield* current.assets.read(input.asset) })),
            ReleaseAsset: input => execute(Effect.gen(function* () { let current = yield* requireSession(); current.assets.release(input.token) })),
            AssetAdmission: input => execute(Effect.gen(function* () { yield* requireSession(); assetsWritable = input.writable })),

            CaptureComment: input => execute(Effect.gen(function* () {
                let current = yield* requireSession()
                let local = basis
                if (!local || local.epoch !== input.epoch || local.sequence !== input.sequence) return yield* Effect.fail(fail("BasisUnavailable"))
                if (!commentsWritable) return yield* Effect.fail(fail("PermissionDenied"))
                if (commentComposers.size >= 8) return yield* Effect.fail(fail("HandleCapacity"))
                let source = local.replica.text(current.document)
                if (input.start >= input.end || input.end > source.length || /[\uD800-\uDBFF](?![\uDC00-\uDFFF])|(?<![\uD800-\uDBFF])[\uDC00-\uDFFF]/u.test(source.slice(input.start, input.end))) return yield* Effect.fail(fail("SelectionUnavailable"))
                let captured = selectionAnchor(local.replica.visible(current.document), source, source.slice(input.start, input.end), input.start)
                let composer = yield* current.discussion.prepareComment(captured)
                let token = crypto.randomUUID()
                commentComposers.set(token, composer)
                return { token, quote: captured.quote }
            })),
            SubmitComment: input => execute(Effect.gen(function* () {
                let current = yield* requireSession()
                if (current.session.replica.metadata(current.document).commentsDisabled) return yield* Effect.fail(fail("CommentsDisabled"))
                let composer = commentComposers.get(input.token)
                if (!composer) return yield* Effect.fail(fail("CommentUnavailable"))
                let action = yield* composer.submit(input.body)
                return { action: action.id }
            })),
            ReleaseComment: input => execute(Effect.sync(() => { commentComposers.delete(input.token) })),
            CommentAdmission: input => execute(Effect.gen(function* () { yield* requireSession(); commentsWritable = input.writable })),
            CommentsEnabled: input => execute(Effect.gen(function* () { let current = yield* requireSession(); yield* current.discussion.setEnabled(input.enabled) })),
            Threads: () => execute(Effect.gen(function* () {
                let current = yield* requireSession()
                let threads = current.discussion.current()
                let local = basis
                let ranges = commentRanges(local?.replica ?? current.session.replica, current.document, threads.map(thread => thread.fields))
                return {
                    ...(local ? { epoch: local.epoch, sequence: local.sequence } : {}),
                    revision: current.session.version(),
                    threads: threads.map((thread, index) => ({ id: thread.id, fields: thread.fields, range: ranges[index], orphaned: thread.orphaned })),
                }
            })),

			Replace: input => execute(Effect.gen(function* () {
				if (basis) return yield* Effect.fail(fail("BasisUnavailable"))
				yield* requireCapacity()
				let current = yield* requireSession()
				let base = handles.get(input.handle)
				if (!base) return yield* Effect.fail(fail("HandleUnavailable"))
				let edit = yield* Effect.try(() => current.session.replace(input.ranges, base))
				let saved: Effect.Effect<void, unknown> = edit.saved
				yield* saved.pipe(Effect.mapError(error => new WorkerDocumentFailure({
					reason: edit.operation ? "Unsaved" : "Rejected",
					category: documentFailureCategory(error),
				})))
				return yield* read()
			})),
			Begin: () => execute(Effect.gen(function* () {
				let current = yield* requireSession()
				if (basis) return yield* Effect.fail(fail("BasisUnavailable"))
				let replica = yield* current.session.replica.forkEffect()
				basis = { epoch: String(++epoch), sequence: 0, replica }
				return { epoch: basis.epoch, sequence: basis.sequence, source: replica.text(current.document) }
			})),
			ReplaceQueued: input => execute(Effect.gen(function* () {
				let current = yield* requireSession()
				let local = basis
				if (!local || local.epoch !== input.epoch) return yield* Effect.fail(fail("BasisUnavailable"))
				if (input.sequence !== local.sequence + 1) return yield* Effect.fail(fail("SequenceUnavailable"))
				candidate = undefined
				let replacements: readonly TextReplacement[] = input.ranges
				if (input.clipboardToken) {
					let captured = clipboard.get(input.clipboardToken)
					if (!captured || captured.document !== current.document) return yield* Effect.fail(fail("ClipboardUnavailable"))
					clipboard.delete(input.clipboardToken)
					let range = resolveClipboardRange(local.replica.visible(current.document), local.replica.text(current.document), captured)
					let remoteRange = resolveClipboardRange(current.session.replica.visible(current.document), current.session.replica.text(current.document), captured)
					let replacement = range && clipboardReplacement(local.replica.text(current.document), range, input.ranges)
					if (!replacement || !remoteRange) return yield* Effect.fail(fail("ClipboardChanged"))
					replacements = [replacement]
				}
				let edit = yield* Effect.try({
					try: () => {
						let edit = current.session.replace(replacements, local.replica.visible(current.document))
						if (edit.operation) {
							local.replica.apply([edit.operation])
							local.sequence = input.sequence
						}
						return edit
					},
					catch: () => {
						basis = undefined
					candidate = undefined
						return fail("AppliedUnknown")
					},
				})
				let saved: Effect.Effect<void, unknown> = edit.saved
				yield* saved.pipe(Effect.mapError(error => new WorkerDocumentFailure({
					reason: edit.operation ? "Unsaved" : "Rejected", category: documentFailureCategory(error),
				})))
				local.sequence = input.sequence
				return { epoch: local.epoch, sequence: local.sequence, source: local.replica.text(current.document), unsaved: current.session.unsaved() }
			})),
			UndoQueued: input => execute(Effect.uninterruptibleMask(restore => Effect.gen(function* () {
				let retained = undoReceipts.get(input.request)
				if (retained) {
					if (retained.epoch !== input.epoch || retained.sequence !== input.sequence || retained.redo !== input.redo)
						return yield* Effect.fail(fail("SequenceUnavailable"))
					return retained
				}
				let current = yield* requireSession()
				let local = basis
				if (!local || local.epoch !== input.epoch) return yield* Effect.fail(fail("BasisUnavailable"))
				if (input.sequence !== local.sequence + 1) return yield* Effect.fail(fail("SequenceUnavailable"))
				candidate = undefined
				let result = yield* restore(input.redo ? current.session.redo() : current.session.undo()).pipe(Effect.exit)
				let status: "applied" | "not-applied" | "unknown"
				let action: string | undefined
				let failure: string | undefined
				if (Exit.isSuccess(result)) {
					status = "applied"
					action = result.value.id
				} else {
					let error = Cause.squash(result.cause)
					status = error instanceof UndoUnavailable ? "not-applied" : "unknown"
					failure = documentFailureCategory(error)
				}
				local.sequence = input.sequence
				let receipt = { ...input, status, ...(action ? { action } : {}), ...(failure ? { failure } : {}) }
				undoReceipts.set(input.request, receipt)
				if (undoReceipts.size > 64) {
					let oldest = undoReceipts.keys().next().value
					if (oldest) undoReceipts.delete(oldest)
				}
				return receipt
			}))),
			MergeCandidate: input => execute(Effect.gen(function* () {
				let current = yield* requireSession()
				let local = basis
				if (!local || local.epoch !== input.epoch) return yield* Effect.fail(fail("BasisUnavailable"))
				if (local.sequence !== input.sequence) return yield* Effect.fail(fail("SequenceUnavailable"))
				candidate = undefined
				if (input.start > input.end || input.end > local.replica.text(current.document).length) return yield* Effect.fail(fail("SelectionUnavailable"))
				yield* current.session.flush
				let replica = yield* current.session.replica.forkEffect()
				let source = replica.text(current.document)
				let oldAtoms = local.replica.visible(current.document)
				let newAtoms = replica.visible(current.document)
				if (input.ranges?.some(range => range.start > range.end || range.end > local.replica.text(current.document).length)) return yield* Effect.fail(fail("SelectionUnavailable"))
				let requestedRanges = input.ranges ?? [{ start: input.start, end: input.end, direction: "none" }]
				let mapped = mapSelectionBoundaries(oldAtoms, newAtoms, [input.start, input.end, ...requestedRanges.flatMap(range => [range.start, range.end])])
				let start = mapped.get(input.start), end = mapped.get(input.end)
				if (start === undefined || end === undefined) return yield* Effect.fail(fail("SelectionUnavailable"))
				let mappedRanges: { start: number; end: number; direction: "forward" | "backward" | "none" }[] = []
				for (let range of requestedRanges) {
					let start = mapped.get(range.start), end = mapped.get(range.end)
					if (start === undefined || end === undefined) return yield* Effect.fail(fail("SelectionUnavailable"))
					mappedRanges.push({ start, end, direction: range.direction })
				}
				let next = String(++epoch)
				candidate = { epoch: next, priorEpoch: local.epoch, priorSequence: local.sequence, replica }
				return { candidate: next, source, start, end, ranges: mappedRanges }
			})),
			Adopt: input => execute(Effect.try({
				try: () => {
					if (!basis || !candidate || candidate.epoch !== input.candidate || basis.epoch !== input.epoch || basis.sequence !== input.sequence || candidate.priorEpoch !== input.epoch || candidate.priorSequence !== input.sequence) throw fail("CandidateUnavailable")
					basis = { epoch: candidate.epoch, sequence: 0, replica: candidate.replica }
					candidate = undefined
				},
				catch: error => error instanceof WorkerDocumentFailure ? error : fail("AppliedUnknown"),
			})),
			ReleaseCandidate: input => execute(Effect.sync(() => {
				if (candidate?.epoch === input.candidate) candidate = undefined
			})),
			CaptureClipboard: input => execute(Effect.gen(function* () {
				let current = yield* requireSession()
				let local = basis
				if (!local || local.epoch !== input.epoch || local.sequence !== input.sequence) return yield* Effect.fail(fail("BasisUnavailable"))
				if (clipboard.size >= 8) return yield* Effect.fail(fail("HandleCapacity"))
				let source = local.replica.text(current.document)
				if (input.start > input.end || input.end > source.length) return yield* Effect.fail(fail("SelectionUnavailable"))
				let atoms = local.replica.visible(current.document)
				let start = captureClipboardBoundary(atoms, input.start)
				let end = captureClipboardBoundary(atoms, input.end)
				if (!start || !end) return yield* Effect.fail(fail("SelectionUnavailable"))
				let token = crypto.randomUUID()
				let selected = source.slice(input.start, input.end)
				clipboard.set(token, { document: current.document, start, end, selected })
				return { token, selected }
			})),
			ResolveClipboard: input => execute(Effect.gen(function* () {
				let current = yield* requireSession()
				let local = basis
				if (!local || local.epoch !== input.epoch || local.sequence !== input.sequence) return yield* Effect.fail(fail("BasisUnavailable"))
				let captured = clipboard.get(input.token)
				if (!captured || captured.document !== current.document) return yield* Effect.fail(fail("ClipboardUnavailable"))
				let source = local.replica.text(current.document)
				let range = resolveClipboardRange(local.replica.visible(current.document), source, captured)
				let remoteSource = current.session.replica.text(current.document)
				let remoteRange = resolveClipboardRange(current.session.replica.visible(current.document), remoteSource, captured)
				if (!range || !remoteRange) return yield* Effect.fail(fail("ClipboardChanged"))
				return { epoch: local.epoch, sequence: local.sequence, source, ...range }
			})),
			ReleaseClipboard: input => execute(Effect.sync(() => { clipboard.delete(input.token) })),
			End: input => execute(Effect.gen(function* () {
				let current = yield* requireSession()
				if (!basis || basis.epoch !== input.epoch) return yield* Effect.fail(fail("BasisUnavailable"))
				if (input.sequence !== basis.sequence) return yield* Effect.fail(fail("SequenceUnavailable"))
				yield* current.session.flush
				let merged = yield* read()
                current.assets.close()
				basis = undefined
					candidate = undefined
				clipboard.clear()
                commentComposers.clear()
				return { ...merged, epoch: undefined, sequence: undefined, localSource: undefined }
			})),
			Release: input => execute(Effect.sync(() => { handles.delete(input.handle) })),
			Sync: () => execute(requireSession().pipe(Effect.flatMap(current => current.session.sync))),
			Flush: () => execute(requireSession().pipe(Effect.flatMap(current => current.session.flush))),
			Close: () => execute(Effect.uninterruptible(Effect.gen(function* () {
				let current = yield* requireSession()
				opened = undefined
				basis = undefined
					candidate = undefined
				handles.clear()
				clipboard.clear()
                commentComposers.clear()
				yield* Scope.close(current.scope, Exit.void)
			})).pipe(Effect.catchDefect(() => Effect.fail(new WorkerDocumentFailure({
				reason: "SessionFailure", category: "storage",
			}))))),
		})
	}),
)

function mapSelectionBoundaries(previous: readonly Atom[], next: readonly Atom[], offsets: number[]) {
	let nextPositions = new Map<string, { start: number; end: number }>()
	let nextOffset = 0
	for (let atom of next) {
		nextPositions.set(atom.id, { start: nextOffset, end: nextOffset + atom.value.length })
		nextOffset += atom.value.length
	}
	let requests = [...new Set(offsets)].sort((left, right) => left - right)
	let resolved = new Map<number, number>()
	if (requests[0] === 0) { resolved.set(0, 0); requests.shift() }
	let request = 0, position = 0, predecessor = 0
	for (let atom of previous) {
		let mapped = nextPositions.get(atom.id)
		let offset = requests[request]
		while (atom.value.length && offset !== undefined && offset <= position + atom.value.length) {
			resolved.set(offset, mapped ? mapped.start + Math.min(offset - position, mapped.end - mapped.start) : predecessor)
			request++
			offset = requests[request]
		}
		if (mapped) predecessor = mapped.end
		position += atom.value.length
	}
	return resolved
}

interface ClipboardBoundary { id: string; inside: number }
function captureClipboardBoundary(atoms: readonly Atom[], offset: number): ClipboardBoundary | undefined {
	if (offset === 0) return { id: "", inside: 0 }
	let position = 0
	for (let atom of atoms) {
		if (atom.value.length && offset <= position + atom.value.length) return { id: atom.id, inside: offset - position }
		position += atom.value.length
	}
}
function resolveClipboardBoundary(atoms: readonly Atom[], boundary: ClipboardBoundary) {
	if (!boundary.id) return 0
	let position = 0
	for (let atom of atoms) {
		if (atom.id === boundary.id && boundary.inside <= atom.value.length) return position + boundary.inside
		position += atom.value.length
	}
}
function resolveClipboardRange(atoms: readonly Atom[], source: string, captured: { start: ClipboardBoundary; end: ClipboardBoundary; selected: string }) {
	let start = resolveClipboardBoundary(atoms, captured.start)
	let end = resolveClipboardBoundary(atoms, captured.end)
	if (start === undefined || end === undefined || start > end || source.slice(start, end) !== captured.selected) return
	return { start, end }
}

function clipboardReplacement(source: string, range: { start: number; end: number }, replacements: readonly TextReplacement[]): TextReplacement | undefined {
	let pieces: string[] = []
	let position = 0
	for (let replacement of replacements) {
		if (replacement.start < position || replacement.start + replacement.remove > source.length) return
		pieces.push(source.slice(position, replacement.start), replacement.text)
		position = replacement.start + replacement.remove
	}
	pieces.push(source.slice(position))
	let next = pieces.join("")
	let before = source.slice(0, range.start)
	let after = source.slice(range.end)
	if (next.length < before.length + after.length || !next.startsWith(before) || !next.endsWith(after)) return
	return { start: range.start, remove: range.end - range.start, text: next.slice(before.length, next.length - after.length) }
}
