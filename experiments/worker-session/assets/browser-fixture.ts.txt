import * as Effect from "effect/Effect"
import * as Cause from "effect/Cause"
import * as Deferred from "effect/Deferred"
import { RpcClientError, RpcClientDefect } from "effect/rpc/RpcClientError"
import { WorkerDocumentFailure } from "./worker-rpc"
import * as Fiber from "effect/Fiber"
import * as FiberSet from "effect/FiberSet"
import * as FetchHttpClient from "effect/http/FetchHttpClient"
import { bootstrapBrowserAccount } from "./tests/browser-account-fixture"
import { createWorkerSessionClient } from "./worker-client"
import { makeEditorQueue } from "./editor-queue"
import { workerEditorBindings } from "./bindings"
import { EditorElement } from "./src/client/components/editor"
import { editorPreferences } from "./src/shared/preferences"

interface State {
	source: string
	start: number
	end: number
	busy: boolean
	pending: number
	readonly: boolean
	queueSource: string
	failures: string[]
}
declare global {
	interface Window {
		workerEditorProof?: {
			captureAsset(start: number, end: number): Promise<{ readonly token: string }>
            prepareAsset(file: File, selection: string): Promise<{ readonly token: string }>
            executeAsset(token: string): Promise<{ readonly action: string }>
            releaseAsset(token: string): Promise<void>
            assetAdmission(writable: boolean): Promise<void>
            assetDraftFiles(): string[]
            assets(): Promise<unknown>
            attachFiles(files: readonly File[]): Promise<boolean>
            assetChange(kind: "insert" | "rename" | "delete", asset: string, name?: string): Promise<boolean>
            assetFile(asset: string): Promise<Blob>
            fault(mode: string): void
			release(): void
			captureComment(start: number, end: number): Promise<{ token: string; quote: string }>
            submitComment(token: string, body: string): Promise<{ action: string }>
            releaseComment(token: string): Promise<void>
            threads(): Promise<unknown>
            peerRange(start: number, remove: number, text: string): Promise<void>
            commentsEnabled(enabled: boolean): Promise<void>
            commentAdmission(writable: boolean): Promise<void>
            probeLateAdoption(): boolean
            diagnostics(): { mergeRequests: number; requests: number; requestIds: string[]; held: boolean; closed: boolean; closeReceipt: string }
			expiredReceipt(): Promise<boolean>
			observeClosingUndo(): void
			inspect(): State
			paste(): void
			settleClipboard(value: string): void
			denyClipboard(): void
			written(): string | undefined
			settleWrite(): void
			denyWrite(): void
			captured(): number
			undo(redo: boolean): void
			format(action: string): void
			merge(): Promise<boolean>
			peer(selected?: boolean, needle?: string): Promise<void>
			editable(value: boolean): void
			flush(): Promise<string>
			dispose(): Promise<void>
		}
	}
}
let failures: string[] = []
let clipboardResult: { resolve: (value: string) => void; reject: (error: unknown) => void } | undefined
let clipboardCaptured = 0
let clipboardWritten: string | undefined
let clipboardWriting: { resolve: () => void; reject: (error: unknown) => void } | undefined
Object.defineProperty(navigator, "clipboard", { configurable: true, value: {
	readText() {
		return new Promise<string>((resolve, reject) => { clipboardResult = { resolve, reject } })
	},
	writeText(value: string) {
		clipboardWritten = value
		return new Promise<void>((resolve, reject) => { clipboardWriting = { resolve, reject } })
	},
} })
let lifetime: Fiber.Fiber<never, never>
let program = Effect.scoped(
	Effect.gen(function* () {
		let account = yield* bootstrapBrowserAccount((value, message) => {
			if (!value) throw Error(message)
		})
		let existingDocument = new URL(location.href).searchParams.get("document")
		let documentId = existingDocument ?? "actual-editor-" + crypto.randomUUID()
		let size = Number(new URL(location.href).searchParams.get("size") ?? 0)
		let cursors = Number(new URL(location.href).searchParams.get("cursors") ?? 0)
	let body = cursors ? ("TOKEN " + "z".repeat(Math.floor(size / cursors) - 7) + "\n").repeat(cursors) : size
			? "Paragraph text remains editable.\n"
					.repeat(Math.ceil(size / 33))
					.slice(0, size)
			: "# Writer\nalpha beta alpha\naa\nbar\n"
		let source = "---\ntitle: Worker fixture\n---\n" + body
		if (!existingDocument) yield* Effect.promise(async () => {
			let response = await fetch("/__tests/seed", {
				method: "POST",
				headers: { "content-type": "application/json" },
				body: JSON.stringify({
					subject: account.subject,
					document: documentId,
					source,
				}),
			})
			if (!response.ok) throw Error("Server seed failed " + response.status)
		})
		let rawWorker = yield* createWorkerSessionClient(
			new URL("/worker.js", location.href).href,
		)
        let mode = ""
        let held = false
        let closed = false
        let closeReceipt = ""
        let release = Deferred.makeUnsafe<void>()
        let requestIds: string[] = []
        let mergeRequests = 0
        let firstRequest: Parameters<typeof rawWorker.UndoQueued>[0] | undefined
        function dropped() {
            return new RpcClientError({ reason: new RpcClientDefect({ message: "Fixture dropped completed reply", cause: "controlled transport fault" }) })
        }
        let worker = {
            ...rawWorker,
            UndoQueued(payload: Parameters<typeof rawWorker.UndoQueued>[0]) {
                return Effect.gen(function* () {
                    requestIds.push(payload.request)
                    firstRequest ??= payload
                    let outcome = yield* rawWorker.UndoQueued(payload)
                    if (mode === "drop-once") { mode = ""; return yield* Effect.fail(dropped()) }
                    if (mode === "drop-always") return yield* Effect.fail(dropped())
                    if (mode === "hold-undo") { mode = ""; held = true; yield* Deferred.await(release); held = false }
                    return outcome
                })
            },
            MergeCandidate(payload: Parameters<typeof rawWorker.MergeCandidate>[0]) {
                return Effect.gen(function* () {
                    mergeRequests++
                    let candidate = yield* rawWorker.MergeCandidate(payload)
                    if (mode === "hold-merge") { mode = ""; held = true; yield* Deferred.await(release); held = false }
                    return candidate
                })
            },
            Adopt(payload: Parameters<typeof rawWorker.Adopt>[0]) {
                return Effect.gen(function* () {
                    let outcome = yield* rawWorker.Adopt(payload)
                    if (mode === "drop-adopt") { mode = ""; return yield* Effect.fail(dropped()) }
                    return outcome
                })
            },
        }
		let options = {
			document: documentId,
			subject: account.subject,
			origin: location.origin,
            commentsWritable: true,
            assetsWritable: true,
		}
		let opened = yield* worker.Open(options)
		yield* worker.Release({ handle: opened.handle })
		let initial = yield* worker.Begin()
		let queue = yield* makeEditorQueue(worker, initial)
		let tasks = yield* FiberSet.make()
		let scopedFork = yield* FiberSet.runtime(tasks)()
		let editor = new EditorElement()
        let commentDialog: HTMLDialogElement | undefined
        let commentsAlive = true
        let commentOpening = false
        function addComment() {
            let selection = editor.mergeSelection()
            if (!selection || !selection.editable || selection.busy || selection.start === selection.end || commentDialog || commentOpening) return
            commentOpening = true
            scopedFork(Effect.gen(function* () {
                let captured = yield* queue.captureComment(selection.start, selection.end, selection.source)
                if (!commentsAlive) { yield* queue.releaseComment(captured.token); return }
                let dialog = document.createElement("dialog")
                let body = document.createElement("textarea")
                body.setAttribute("aria-label", "Comment body")
                let save = document.createElement("button")
                save.textContent = "Save comment"
                let cancel = document.createElement("button")
                cancel.textContent = "Cancel comment"
                let report = document.createElement("p")
                let finished = false
                let submitting = false
                function close() {
                    if (finished) return
                    finished = true
                    dialog.remove()
                    commentDialog = undefined
                    scopedFork(queue.releaseComment(captured.token).pipe(Effect.catch(() => Effect.void)))
                }
                save.addEventListener("click", () => {
                    if (finished || submitting) return
                    submitting = true
                    scopedFork(queue.submitComment(captured.token, body.value).pipe(Effect.match({
                        onSuccess() { close() },
                        onFailure(error) { submitting = false; report.textContent = error instanceof WorkerDocumentFailure ? error.reason : "Comment could not be saved. Your draft is retained." },
                    })))
                })
                cancel.addEventListener("click", close)
                dialog.addEventListener("cancel", event => { event.preventDefault(); close() })
                dialog.append(body, save, cancel, report)
                document.body.append(dialog)
                commentDialog = dialog
                dialog.showModal()
                body.focus()
            }).pipe(Effect.catch(error => Effect.sync(() => failures.push(String(error)))), Effect.ensuring(Effect.sync(() => { commentOpening = false }))))
        }
        yield* Effect.addFinalizer(() => Effect.sync(() => { commentsAlive = false; commentDialog?.remove(); commentDialog = undefined }))
		let bindings = workerEditorBindings(
			editor,
			queue,
			task => {
				scopedFork(task)
			},
			error => failures.push(String(error)),
		)
        let retainedAssetFiles: File[] = []
        let assetBusy = false
        let pickerSelection: Deferred.Deferred<{ readonly token: string }, unknown> | undefined
        let fileInput = document.createElement("input")
        fileInput.type = "file"
        fileInput.multiple = true
        fileInput.hidden = true
        document.body.append(fileInput)
        yield* Effect.addFinalizer(() => Effect.sync(() => fileInput.remove()))
        function assetOperation(prepare: (captured: (token: string) => void) => ReturnType<typeof queue.prepareAttachment>) {
            return Effect.suspend(() => {
                let selected = editor.mergeSelection()
                if (assetBusy || !selected?.editable || selected.busy || !bindings.beginCommand()) {
                    failures.push("Attachment not admitted; native draft and picked file retained by caller")
                    return Effect.succeed(false)
                }
                assetBusy = true
                let selectionToken: string | undefined
                let commandToken: string | undefined
                let sending = false
                let ambiguous = false
                return Effect.gen(function* () {
                    let command = yield* prepare(token => { selectionToken = token })
                    commandToken = command.token
                    sending = true
                    yield* queue.attachAsset(command.token)
                    sending = false
                    bindings.finishCommand(true)
                    return true
                }).pipe(
                    Effect.catchCause(cause => {
                        if (Cause.hasInterruptsOnly(cause)) return Effect.failCause(cause)
                        let error = Cause.squash(cause)
                        return Effect.sync(() => {
                        let refused = error instanceof WorkerDocumentFailure && ["PermissionDenied", "QueuedBlocked", "Rejected"].includes(error.reason)
                        ambiguous = sending && !refused
                        if (ambiguous) bindings.failCommand(error)
                        else bindings.finishCommand(false)
                        failures.push(error instanceof WorkerDocumentFailure ? error.reason : "Attachment operation failed; native draft retained")
                        return false
                    })
                    }),
                    Effect.ensuring(Effect.gen(function* () {
                        if (selectionToken) yield* queue.releaseAsset(selectionToken).pipe(Effect.catch(() => Effect.void))
                        if (commandToken && !ambiguous) yield* queue.releaseAsset(commandToken).pipe(Effect.catch(() => Effect.void))
                        assetBusy = false
                    })),
                )
            })
        }
        function attachFiles(files: readonly File[], picker?: Deferred.Deferred<{ readonly token: string }, unknown>) {
            retainedAssetFiles.push(...files)
            let selected = editor.mergeSelection()
            if (!selected || !files.length) return Effect.succeed(false)
            return assetOperation(captured => Effect.gen(function* () {
                let selection = picker ? yield* Deferred.await(picker) : yield* queue.captureAssetSelection(selected.start, selected.end, selected.source)
                captured(selection.token)
                return yield* queue.prepareAttachments(files.map(blob => ({ blob, name: blob.name })), selection.token)
            })).pipe(Effect.tap(applied => Effect.sync(() => {
                if (applied) retainedAssetFiles = retainedAssetFiles.filter(file => !files.includes(file))
            })), Effect.ensuring(picker
                ? Deferred.await(picker).pipe(Effect.flatMap(value => queue.releaseAsset(value.token)), Effect.timeout("1 second"), Effect.catch(() => Effect.void))
                : Effect.void))
        }
        function assetChange(kind: "insert" | "rename" | "delete", asset: string, name?: string) {
            let selected = editor.mergeSelection()
            if (!selected) return Effect.succeed(false)
            return assetOperation(captured => Effect.gen(function* () {
                let selection = kind === "insert"
                    ? yield* queue.captureAssetSelection(selected.start, selected.end, selected.source)
                    : undefined
                if (selection) captured(selection.token)
                return yield* queue.prepareAssetChange(kind, asset, name, selection?.token)
            }))
        }
        function requestAssets() {
            let selected = editor.mergeSelection()
            if (assetBusy || !selected?.editable || selected.busy) return
            if (pickerSelection) cancelPicker()
            let pending = Deferred.makeUnsafe<{ readonly token: string }, unknown>()
            pickerSelection = pending
            scopedFork(Deferred.complete(pending, queue.captureAssetSelection(selected.start, selected.end, selected.source)))
            fileInput.click()
        }
        function cancelPicker() {
            let pending = pickerSelection
            pickerSelection = undefined
            if (pending) scopedFork(Deferred.await(pending).pipe(Effect.flatMap(value => queue.releaseAsset(value.token)), Effect.catch(() => Effect.void)))
        }
        fileInput.addEventListener("cancel", cancelPicker)
        fileInput.addEventListener("change", () => {
            let files = Array.from(fileInput.files ?? [])
            fileInput.value = ""
            let pending = pickerSelection
            pickerSelection = undefined
            if (!files.length) {
                if (pending) scopedFork(Deferred.await(pending).pipe(Effect.flatMap(value => queue.releaseAsset(value.token)), Effect.catch(() => Effect.void)))
                return
            }
            scopedFork(attachFiles(files, pending))
        })
		editor.configure({
			authority: "worker",
			content: initial.source,
			atoms: [],
			editable: true,
			preferences: editorPreferences({ autocomplete: false }),
			documents: () => [],
			isDrawing: () => false,
			openDrawing() {},
			openDocument() {},
			comments: () => [],
            workerComments: true,
            workerAssets: true,
			createLinkedDocument: async () => "unused",
			report: error => failures.push(String(error)),
			change: bindings.change,
			clipboard: {
				capture(start, end, expectedSource) { return queue.captureClipboard(start, end, expectedSource).pipe(Effect.tap(() => Effect.sync(() => { clipboardCaptured++ }))) },
				resolve(token) { return queue.resolveClipboard(token) },
				release(token) { return queue.releaseClipboard(token) },
			},
			undo: bindings.undo,
			select: bindings.resumeUndo,
			requestAsset() { requestAssets() },
			attach(files) { scopedFork(attachFiles(files)) },
			command(command) { if (command === "comment") addComment(); if (command === "asset") requestAssets() },
			find() {},
		})
		document.body.append(editor)
		yield* Effect.addFinalizer(() => Effect.sync(() => editor.remove()))
		let peer = yield* createWorkerSessionClient(
			new URL("/worker.js", location.href).href,
		)
		let peerOpened = yield* peer.Open(options)
		yield* peer.Release({ handle: peerOpened.handle })
		window.workerEditorProof = {
            captureAsset(start, end) { let selected = editor.mergeSelection(); if (!selected) return Promise.reject(Error("Editor removed")); return Effect.runPromise(queue.captureAssetSelection(start, end, selected.source)) },
            prepareAsset(file, selection) { return Effect.runPromise(queue.prepareAttachment(file, file.name, selection)) },
            executeAsset(token) { return Effect.runPromise(queue.attachAsset(token)) },
            releaseAsset(token) { return Effect.runPromise(queue.releaseAsset(token)) },
            assetAdmission(writable) { return Effect.runPromise(queue.assetAdmission(writable)) },
            assetDraftFiles() { return retainedAssetFiles.map(file => file.name) },
            assets() { return Effect.runPromise(queue.assets()) },
            attachFiles(files) { return Effect.runPromise(attachFiles(files)) },
            assetChange(kind, asset, name) { return Effect.runPromise(assetChange(kind, asset, name)) },
            assetFile(asset) { return Effect.runPromise(queue.readAsset(asset)) },
			inspect() {
				let text = editor.source
				let selected = editor.mergeSelection()
				if (!text || !selected) throw Error("Editor removed")
				let state = queue.state()
				return {
					source: text.value,
					start: selected.start,
					end: selected.end,
					busy: selected.busy,
					pending: state.pending.length,
					readonly: text.readOnly,
					queueSource: state.source,
					failures,
				}
			},
			paste() { editor.command("paste") },
			settleClipboard(value) {
				if (!clipboardResult) throw Error("Clipboard not requested")
				let pending = clipboardResult
				clipboardResult = undefined
				pending.resolve(value)
			},
			denyClipboard() {
				if (!clipboardResult) throw Error("Clipboard not requested")
				let pending = clipboardResult
				clipboardResult = undefined
				pending.reject(Error("Fixture clipboard permission denied"))
			},
			written() { return clipboardWritten },
			settleWrite() {
				if (!clipboardWriting) throw Error("Clipboard write not requested")
				let pending = clipboardWriting
				clipboardWriting = undefined
				pending.resolve()
			},
			denyWrite() {
				if (!clipboardWriting) throw Error("Clipboard write not requested")
				let pending = clipboardWriting
				clipboardWriting = undefined
				pending.reject(Error("Fixture clipboard write denied"))
			},
			captured() { return clipboardCaptured },
			undo(redo) { bindings.undo(redo) },
			format(action) {
				editor.format(action)
			},
			merge() {
				return Effect.runPromise(bindings.merge().saved)
			},
			editable(value) { editor.setEditable(value) },
			async peer(selected = false, needle = "alphabet") {
				await Effect.runPromise(peer.Sync())
				let view = await Effect.runPromise(peer.Read())
				let next = await Effect.runPromise(
					peer.Replace({
						handle: view.handle,
						ranges: selected
							? [{ start: view.source.indexOf(needle), remove: needle.length, text: "REMOTE" }]
							: [{ start: 0, remove: 0, text: "PEER\n" }],
					}),
				)
				await Effect.runPromise(peer.Release({ handle: view.handle }))
				await Effect.runPromise(peer.Release({ handle: next.handle }))
				await Effect.runPromise(peer.Flush())
				await Effect.runPromise(worker.Sync())
			},
			async peerRange(start, remove, text) {
                await Effect.runPromise(peer.Sync())
                let view = await Effect.runPromise(peer.Read())
                let next = await Effect.runPromise(peer.Replace({ handle: view.handle, ranges: [{ start, remove, text }] }))
                await Effect.runPromise(peer.Release({ handle: view.handle }))
                await Effect.runPromise(peer.Release({ handle: next.handle }))
                await Effect.runPromise(peer.Flush())
                await Effect.runPromise(worker.Sync())
            },
            async flush() {
				await Effect.runPromise(worker.Flush())
				let view = await Effect.runPromise(worker.Read())
				await Effect.runPromise(worker.Release({ handle: view.handle }))
				return view.source
			},
            fault(value) { mode = value; held = false; release = Deferred.makeUnsafe<void>() },
            release() { Deferred.doneUnsafe(release, Effect.void) },
            captureComment(start, end) { let selected = editor.mergeSelection(); if (!selected) return Promise.reject(Error("Editor removed")); return Effect.runPromise(queue.captureComment(start, end, selected.source)) },
            submitComment(token, body) { return Effect.runPromise(queue.submitComment(token, body)) },
            releaseComment(token) { return Effect.runPromise(queue.releaseComment(token)) },
            threads() { return Effect.runPromise(queue.threads()) },
            commentsEnabled(enabled) { return Effect.runPromise(queue.commentsEnabled(enabled)) },
            commentAdmission(writable) { return Effect.runPromise(queue.commentAdmission(writable)) },
            probeLateAdoption() { let selection = editor.mergeSelection(); return selection ? editor.adoptMerged("late candidate must not replace draft", selection) : false },
            diagnostics() { return { mergeRequests, requests: requestIds.length, requestIds: [...requestIds], held, closed, closeReceipt } },
            observeClosingUndo() {
                void Effect.runPromise(queue.undo(false).pipe(Effect.match({
                    onFailure(error) { closeReceipt = error instanceof WorkerDocumentFailure ? error.reason : String(error) },
                    onSuccess(receipt) { closeReceipt = receipt.status },
                })))
            },
            async expiredReceipt() {
                if (!firstRequest) throw Error("No retained request")
                let result = await Effect.runPromise(rawWorker.UndoQueued(firstRequest).pipe(Effect.exit))
                return result._tag === "Failure"
            },
			async dispose() {
				await Effect.runPromise(Fiber.interrupt(lifetime))
				closed = true
				document.body.dataset.closed = "true"
			},
		}
		let currentUrl = new URL(location.href)
		currentUrl.searchParams.set("document", documentId)
		history.replaceState(null, "", currentUrl.href)
		document.body.dataset.ready = "true"
		return yield* Effect.never
	}),
)
lifetime = Effect.runFork(
	program.pipe(
		Effect.provide(FetchHttpClient.layer),
		Effect.catch(error =>
			Effect.sync(() => {
				document.body.dataset.failure = String(error)
			}),
		),
		Effect.andThen(Effect.never),
	),
)
