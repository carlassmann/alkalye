import * as Effect from "effect/Effect"
import * as Fiber from "effect/Fiber"
import * as FiberSet from "effect/FiberSet"
import * as FetchHttpClient from "effect/http/FetchHttpClient"
import { bootstrapBrowserAccount } from "./tests/browser-account-fixture"
import { createWorkerSessionClient } from "./worker-client"
import { makeEditorQueue } from "./editor-queue"
import { workerEditorBindings } from "./bindings"
import { EditorElement } from "./src/client/components/editor"
import { accountHeader } from "./src/shared/authentication"
import { editorPreferences } from "./src/shared/preferences"

interface State {
	source: string
	start: number
	end: number
	busy: boolean
	pending: number
	readonly: boolean
	queueSource: string
	failures: string[]
}
declare global {
	interface Window {
		workerEditorProof?: {
			inspect(): State
			format(action: string): void
			merge(): Promise<boolean>
			peer(): Promise<void>
			flush(): Promise<string>
			dispose(): Promise<void>
            gauge(): { sequence: number; acknowledged: number; applied: number; pending: number; inserted: number; failures: number }
            startSustained(text: string, peers: number): void
            finishSustained(reopen?: boolean): Promise<unknown>
		}
	}
}
let failures: string[] = []
let lifetime: Fiber.Fiber<never, never>
let program = Effect.scoped(
	Effect.gen(function* () {
		let account = yield* bootstrapBrowserAccount((value, message) => {
			if (!value) throw Error(message)
		})
		let documentId = "actual-editor-" + crypto.randomUUID()
		let size = Number(new URL(location.href).searchParams.get("size") ?? 0)
		let cursors = Number(new URL(location.href).searchParams.get("cursors") ?? 0)
	let body = cursors ? ("TOKEN " + "z".repeat(Math.floor(size / cursors) - 7) + "\n").repeat(cursors) : size
			? "Paragraph text remains editable.\n"
					.repeat(Math.ceil(size / 33))
					.slice(0, size)
			: "# Writer\nalpha beta alpha\n"
		let source = "---\ntitle: Worker fixture\n---\n" + body
		yield* Effect.promise(async () => {
			let response = await fetch("/__tests/seed", {
				method: "POST",
				headers: { "content-type": "application/json" },
				body: JSON.stringify({
					subject: account.subject,
					document: documentId,
					source,
				}),
			})
			if (!response.ok) throw Error("Server seed failed " + response.status)
		})
		let worker = yield* createWorkerSessionClient(
			new URL("/worker.js", location.href).href,
		)
		let options = {
			document: documentId,
			subject: account.subject,
			origin: location.origin,
		}
		let opened = yield* worker.Open(options)
		yield* worker.Release({ handle: opened.handle })
		let initial = yield* worker.Begin()
		let queue = yield* makeEditorQueue(worker, initial)
		let tasks = yield* FiberSet.make()
		let scopedFork = yield* FiberSet.runtime(tasks)()
		let editor = new EditorElement()
		let bindings = workerEditorBindings(
			editor,
			queue,
			task => {
				scopedFork(task)
			},
			error => failures.push(String(error)),
		)
		editor.configure({
			authority: "worker",
			content: initial.source,
			atoms: [],
			editable: true,
			preferences: editorPreferences({ autocomplete: false }),
			documents: () => [],
			isDrawing: () => false,
			openDrawing() {},
			openDocument() {},
			comments: () => [],
			createLinkedDocument: async () => "unused",
			report: error => failures.push(String(error)),
			change: bindings.change,
			undo() {},
			select() {},
			requestAsset() {},
			attach() {},
			command() {},
			find() {},
		})
		document.body.append(editor)
		yield* Effect.addFinalizer(() => Effect.sync(() => editor.remove()))
		let peer = yield* createWorkerSessionClient(
			new URL("/worker.js", location.href).href,
		)
		let peerOpened = yield* peer.Open(options)
		yield* peer.Release({ handle: peerOpened.handle })
		let expectedBase = source
        let typed = ""
        let prefixes: string[] = []
        let peerWork: Promise<void> = Promise.resolve()
        let peerCompleted = 0
        let peerTimes: { index: number; start: number; replaced: number; end: number }[] = []
        window.workerEditorProof = {
            gauge() {
                let state = queue.state()
                return { sequence: state.sequence, acknowledged: state.acknowledged, applied: state.applied, pending: state.pending.length, inserted: state.pending.reduce((count, edit) => count + edit.ranges.reduce((sum, range) => sum + range.text.length, 0), 0), failures: failures.length }
            },
            startSustained(text, peers) {
                typed = text
                expectedBase = editor.source?.value ?? ""
                prefixes = []
                peerCompleted = 0
                peerTimes = []
                peerWork = (async () => {
                    for (let index = 0; index < peers; index++) {
                        await new Promise(resolve => setTimeout(resolve, index === 0 ? 0 : 500))
                        let start = performance.now()
                        if (index > 0) await Effect.runPromise(peer.Sync())
                        let view = await Effect.runPromise(peer.Read())
                        let prefix = "PEER" + index + "\n"
                        let next = await Effect.runPromise(peer.Replace({ handle: view.handle, ranges: [{ start: 0, remove: 0, text: prefix }] }))
                        await Effect.runPromise(peer.Release({ handle: view.handle }))
                        await Effect.runPromise(peer.Release({ handle: next.handle }))
                        await Effect.runPromise(peer.Flush())
                        await Effect.runPromise(peer.Sync())
                        let replaced = performance.now()
                        prefixes.unshift(prefix)
                        peerCompleted++
                        peerTimes.push({ index, start, replaced, end: performance.now() })
                    }
                })().catch(error => { failures.push(String(error)) })
            },
            async finishSustained(reopen = true) {
                await peerWork
                await Effect.runPromise(worker.Flush())
                await Effect.runPromise(worker.Sync())
                let merged = await Effect.runPromise(bindings.merge().saved)
                let view = await Effect.runPromise(worker.Read())
                await Effect.runPromise(worker.Release({ handle: view.handle }))
                let expected = prefixes.join("") + expectedBase + typed
                let digest = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(expected))
                let sha256 = Array.from(new Uint8Array(digest), byte => byte.toString(16).padStart(2, "0")).join("")
                let response = await fetch("/__tests/accepted?document=" + encodeURIComponent(documentId), { headers: { [accountHeader]: account.subject } })
                if (!response.ok) throw Error("Snapshot gate " + response.status)
                let accepted = await response.json()
                let localReopenExact = false
                if (reopen) {
                    await Effect.runPromise(peer.Sync())
                    await fetch("/__tests/network?blocked=1")
                    try {
                        await Effect.runPromise(peer.Close())
                        await Effect.runPromise(worker.Close())
                        let reopened = await Effect.runPromise(worker.Open(options))
                        await Effect.runPromise(worker.Release({ handle: reopened.handle }))
                        localReopenExact = reopened.source === expected
                    } finally {
                        await fetch("/__tests/network?blocked=0")
                    }
                }
                return { merged, editorExact: editor.source?.value === expected, workerExact: view.source === expected, localReopenExact, sha256, accepted, pending: view.pending, rejected: view.rejected, unsaved: view.unsaved, peerCompleted, peerTimes, failures, expectedCharacters: expected.length }
            },
			inspect() {
				let text = editor.source
				let selected = editor.mergeSelection()
				if (!text || !selected) throw Error("Editor removed")
				let state = queue.state()
				return {
					source: text.value,
					start: selected.start,
					end: selected.end,
					busy: selected.busy,
					pending: state.pending.length,
					readonly: text.readOnly,
					queueSource: state.source,
					failures,
				}
			},
			format(action) {
				editor.format(action)
			},
			merge() {
				return Effect.runPromise(bindings.merge().saved)
			},
			async peer() {
				await Effect.runPromise(peer.Sync())
				let view = await Effect.runPromise(peer.Read())
				let next = await Effect.runPromise(
					peer.Replace({
						handle: view.handle,
						ranges: [{ start: 0, remove: 0, text: "PEER\n" }],
					}),
				)
				await Effect.runPromise(peer.Release({ handle: view.handle }))
				await Effect.runPromise(peer.Release({ handle: next.handle }))
				await Effect.runPromise(peer.Flush())
				await Effect.runPromise(worker.Sync())
			},
			async flush() {
				await Effect.runPromise(worker.Flush())
				let view = await Effect.runPromise(worker.Read())
				await Effect.runPromise(worker.Release({ handle: view.handle }))
				return view.source
			},
			async dispose() {
				await Effect.runPromise(Fiber.interrupt(lifetime))
				document.body.dataset.closed = "true"
			},
		}
		document.body.dataset.ready = "true"
		return yield* Effect.never
	}),
)
lifetime = Effect.runFork(
	program.pipe(
		Effect.provide(FetchHttpClient.layer),
		Effect.catch(error =>
			Effect.sync(() => {
				document.body.dataset.failure = String(error)
			}),
		),
		Effect.andThen(Effect.never),
	),
)
