import { mkdtemp, mkdir, writeFile } from "node:fs/promises"
import { join } from "node:path"
import { tmpdir } from "node:os"
import * as Effect from "effect/Effect"
import { startServer } from "./src/server/index"
import { provisionFixture, acceptedFixture } from "./domain-seed"

let built = await Bun.build({ entrypoints: ["src/client/worker-document/worker.ts", "domain-worker-fixture.ts"], target: "browser", outdir: "assets", naming: { entry: "[name].js" } })
if (!built.success) throw new AggregateError(built.logs, "Worker fixture build failed")
let backend: Effect.Success<ReturnType<typeof startServer>>
let blocked = false
let css = await Bun.file("src/client/components/editor.css").text() + await Bun.file("src/client/components/editor-viewport.css").text()
let server = Bun.serve({
	port: 0,
	fetch: async (request): Promise<Response> => {
		let url = new URL(request.url)
		if (url.pathname === "/__tests/accepted") return acceptedFixture(request,backend.server.url,server.url.origin)
		if (url.pathname === "/__tests/seed" && request.method === "POST") return provisionFixture(request, backend.server.url, server.url.origin)
		if (url.pathname === "/") return new Response('<!doctype html><title>Worker library proof</title><style>*{box-sizing:border-box}:root{--font-mono:monospace;--foreground:#111;--background:#fff}' + css + '</style><script type="module" src="/domain-worker-fixture.js"></script>', { headers: { "content-type": "text/html" } })
		if (url.pathname === "/__tests/network") {
			blocked = url.searchParams.get("blocked") === "1"
			return new Response("ok")
		}
		if (url.pathname === "/worker.js" || url.pathname === "/domain-worker-fixture.js") return new Response(Bun.file("assets" + url.pathname), { headers: { "content-type": "text/javascript" } })
		if (blocked && url.pathname.startsWith("/api/")) return new Response("Offline fixture", { status: 503 })
		let headers = new Headers(request.headers)
		headers.set("host", server.url.host)
		return fetch(new URL(url.pathname + url.search, backend.server.url), {
			decompress: false,
			method: request.method,
			headers,
			body: request.method === "GET" || request.method === "HEAD" ? undefined : await request.arrayBuffer(),
		})
	},
})
let runRoot = process.env.DOMAIN_RUN_ROOT ?? await mkdtemp(join(tmpdir(), "alkalye-domain-worker-"))
let directory = join(runRoot, "data")
await mkdir(directory, { recursive: true })
await writeFile(join(runRoot, "run.private.json"), JSON.stringify({ directory, origin: server.url.origin, created: new Date().toISOString() }))
backend = await Effect.runPromise(startServer({ port: 0, data: directory, build: false, publicOrigin: server.url.origin }))
let stopping = false
for (let signal of ["SIGINT", "SIGTERM"]) process.once(signal, async () => {
	if (stopping) return
	stopping = true
	await server.stop(true)
	await Effect.runPromise(backend.close())
 await writeFile(join(runRoot, "closed.private.json"), JSON.stringify({ closed: true, time: new Date().toISOString() }))

	process.exit(0)
})
console.log(server.url.href)
