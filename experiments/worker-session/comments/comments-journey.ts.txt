import { assert, browser, evaluate, wait } from "./tests/browser-driver"
let url = process.argv[2]
if (!url) throw Error("URL required")
let session = "worker-comments-" + Date.now()
let proof = "window.workerEditorProof"
let text = 'document.querySelector("alkalye-editor").source'
try {
 browser(session, "open", url)
 wait(session, "document.body.dataset.ready==='true'||!!document.body.dataset.failure")
 assert(evaluate(session, "!document.body.dataset.failure"), "Actual comment Editor opens with worker authority")
 let original = evaluate(session, `${proof}.inspect().source`)
 if (typeof original !== "string") throw Error("Source unavailable")
 evaluate(session, `${text}.focus();${text}.setSelectionRange(${original.length},${original.length})`)
 browser(session, "keyboard", "inserttext", "same😀 same😀")
 let source = original + "same😀 same😀"
 let start = source.lastIndexOf("same😀")
 wait(session, `${proof}.inspect().source===${JSON.stringify(source)}&&${proof}.inspect().pending===0`)
 evaluate(session, `${text}.setSelectionRange(${start},${start+6});${text}.dispatchEvent(new MouseEvent('contextmenu',{bubbles:true,cancelable:true}))`)
 wait(session, "!!document.querySelector('[data-context-menu]:popover-open')")
 evaluate(session, `(function(){let item=[...document.querySelectorAll('[data-context-menu]:popover-open>[role=menuitem]')].find(item=>item.querySelector(':scope>span')?.textContent==='Add comment');if(!item)throw Error('Menu item missing');item.dataset.commentAction='true'})()`)
 browser(session, "click", "[data-comment-action=true]")
 wait(session, "!!document.querySelector('dialog[open] textarea')")
 evaluate(session, `${proof}.peerRange(${source.indexOf("alpha")},0,'PEER')`)
 browser(session, "fill", "dialog textarea", "Unicode selected comment")
 browser(session, "find", "role", "button", "click", "--name", "Save comment", "--exact")
 wait(session, "!document.querySelector('dialog[open]')")
 assert(evaluate(session, `${proof}.threads().then(({threads})=>threads.length===1&&threads[0].fields.quote==='same😀'&&threads[0].range.start===${start}&&threads[0].range.end===${start+6})`), "Actual menu composer retains exact repeated Unicode occurrence")
 assert(evaluate(session, `${proof}.threads().then(value=>typeof value.epoch==='string'&&typeof value.sequence==='number')`), "Threads DTO tags editor-basis coordinates despite unseen peer insertion")
 evaluate(session, `${proof}.peerRange(${start+4},6,'')`)
 assert(evaluate(session, `${proof}.threads().then(({threads})=>threads[0].orphaned&&threads[0].range.start===${start})`), "Authoritative orphan status stays distinct from retained editor-basis selection")
 evaluate(session, `${proof}.merge()`)
 assert(evaluate(session, `${proof}.threads().then(({threads})=>threads.length===1&&threads[0].orphaned&&!threads[0].range)`), "Peer deletion orphans captured occurrence without attaching earlier identical quote")
 evaluate(session, `${proof}.merge()`)
 let next = evaluate(session, `${proof}.inspect().source`)
 if(typeof next!=="string")throw Error("Merged source unavailable")
 let alpha = next.indexOf("alpha")
 evaluate(session, `${proof}.captureComment(${alpha},${alpha+5}).then(value=>{window.commentToken=value.token})`)
 let token = evaluate(session, "window.commentToken")
 if(typeof token!=="string")throw Error("Comment token unavailable")
 let first=evaluate(session, `${proof}.submitComment(${JSON.stringify(token)},'Retry body').then(value=>value.action)`)
 let repeated=evaluate(session, `${proof}.submitComment(${JSON.stringify(token)},'Retry body').then(value=>value.action)`)
 assert(first===repeated, "Existing composer retry preserves action identity")
 assert(evaluate(session, `${proof}.threads().then(({threads})=>threads.length===2)`), "Retry creates one thread")
 evaluate(session, `${proof}.commentAdmission(false)`)
 assert(evaluate(session, `${proof}.submitComment(${JSON.stringify(token)},'Denied body').then(()=>false,error=>error.reason==='PermissionDenied')`), "Readonly UI admission returns typed permission failure without losing composer")
 evaluate(session, `${proof}.commentAdmission(true)`)
 evaluate(session, `${proof}.submitComment(${JSON.stringify(token)},'Reauthorized body')`)
 assert(evaluate(session, `${proof}.threads().then(({threads})=>threads.some(thread=>thread.fields.body==='Reauthorized body'))`), "Composer resumes after admission returns")
 assert(evaluate(session, `${proof}.captureComment(${next.indexOf("😀")},${next.indexOf("😀")+1}).then(()=>false,error=>error.reason==='SelectionUnavailable')`), "Half-surrogate comment selection refused without writing malformed quote")
 evaluate(session, `${proof}.commentsEnabled(false)`)
 assert(evaluate(session, `${proof}.submitComment(${JSON.stringify(token)},'Disabled body').then(()=>false,error=>error.reason==='CommentsDisabled')`), "Disabled discussion refuses typed submission while preserving composer")
 evaluate(session, `${proof}.commentsEnabled(true)`)
 evaluate(session, `${proof}.releaseComment(${JSON.stringify(token)})`)
 assert(evaluate(session, `${proof}.submitComment(${JSON.stringify(token)},'Released').then(()=>false,error=>error.reason==='CommentUnavailable')`), "Released capability refuses reuse")
 assert(evaluate(session, `${proof}.flush()`)===next, "Comment metadata changes preserve document source")
 evaluate(session, `${proof}.dispose()`)
 assert(evaluate(session, "document.body.dataset.closed==='true'&&!document.querySelector('alkalye-editor')"), "Scope cleanup releases actual editor and worker")
} finally { browser(session,"close") }
