import { assert, browser, evaluate, wait } from "./tests/browser-driver"

let url = process.argv[2]
if (!url) throw Error("URL required")
let session = "worker-menu-" + Date.now()
let proof = "window.workerEditorProof"
let text = 'document.querySelector("alkalye-editor").source'
function choose(label: string) {
	evaluate(session, `(function(){let item=[...document.querySelectorAll('[data-context-menu]:popover-open>[role=menuitem]')].find(item=>item.querySelector(':scope>span')?.textContent===${JSON.stringify(label)});if(!item)throw Error('Menu item missing');item.dataset.clipboardAction=${JSON.stringify(label)}})()`)
	browser(session, "click", `[data-clipboard-action="${label}"]`)
}
function menu(start: number, end: number) {
	evaluate(session, `${text}.focus();${text}.setSelectionRange(${start},${end});${text}.dispatchEvent(new MouseEvent('contextmenu',{bubbles:true,cancelable:true}))`)
	wait(session, "!!document.querySelector('[data-context-menu]:popover-open')")
}
try {
	browser(session, "set", "viewport", "1280", "800")
	browser(session, "open", url)
	wait(session, "document.body.dataset.ready==='true'||!!document.body.dataset.failure")
	assert(evaluate(session, "!document.body.dataset.failure"), "Actual worker menu fixture opens")
	let original = evaluate(session, `${proof}.inspect().source`)
	if (typeof original !== "string") throw Error("Source unavailable")
	let expected = original
	let beta = expected.indexOf("beta")
	menu(beta, beta + 4)
	choose("Copy")
	wait(session, `${proof}.written()==='beta'`)
	assert(evaluate(session, `${proof}.inspect().source===${JSON.stringify(expected)}`), "Actual menu Copy writes exact selection without deleting")
	evaluate(session, `${proof}.settleWrite()`)
	menu(beta, beta + 4)
	choose("Paste")
	wait(session, `${proof}.captured()===1`)
	evaluate(session, `${proof}.settleClipboard('BETA')`)
	expected = expected.slice(0, beta) + "BETA" + expected.slice(beta + 4)
	wait(session, `${proof}.inspect().source===${JSON.stringify(expected)}&&${proof}.inspect().pending===0`)
	assert(evaluate(session, `${proof}.flush()`) === expected, "Actual menu Paste has exact HTTP durability")
	menu(beta, beta + 4)
	choose("Cut")
	wait(session, `${proof}.captured()===2&&${proof}.written()==='BETA'`)
	assert(evaluate(session, `${proof}.inspect().source===${JSON.stringify(expected)}`), "Menu Cut preserves text until clipboard write succeeds")
	evaluate(session, `${proof}.settleWrite()`)
	expected = expected.slice(0, beta) + expected.slice(beta + 4)
	wait(session, `${proof}.inspect().source===${JSON.stringify(expected)}&&${proof}.inspect().pending===0`)
	assert(evaluate(session, `${proof}.flush()`) === expected, "Successful menu Cut deletes only captured span durably")
	let alpha = expected.indexOf("alpha")
	menu(alpha, alpha + 5)
	choose("Cut")
	wait(session, `${proof}.captured()===3&&${proof}.written()==='alpha'`)
	evaluate(session, `${proof}.denyWrite()`)
	wait(session, `${proof}.inspect().failures.length===1`)
	assert(evaluate(session, `${proof}.inspect().source===${JSON.stringify(expected)}&&!${proof}.inspect().readonly`), "Denied menu Cut retains text and native typing authority")
	menu(alpha, alpha + 5)
	choose("Cut")
	wait(session, `${proof}.captured()===4&&${proof}.written()==='alpha'`)
	evaluate(session, `${proof}.peer(true,'alpha')`)
	evaluate(session, `${proof}.settleWrite()`)
	wait(session, `${proof}.inspect().failures.length===2`)
	assert(evaluate(session, `${proof}.inspect().source===${JSON.stringify(expected)}&&!${proof}.inspect().readonly`), "Peer-changed menu Cut refuses deletion and retains draft")
	evaluate(session, `${text}.setSelectionRange(0,0)`)
	assert(evaluate(session, `${proof}.merge()`) === true, "Refused menu Cut retains usable worker basis")
	expected = expected.replace("alpha", "REMOTE")
	wait(session, `${proof}.inspect().source===${JSON.stringify(expected)}`)
	alpha = expected.indexOf("alpha")
	menu(alpha, alpha + 5)
	choose("Format")
	choose("Bold")
	expected = expected.slice(0, alpha) + "**alpha**" + expected.slice(alpha + 5)
	wait(session, `${proof}.inspect().source===${JSON.stringify(expected)}&&${proof}.inspect().pending===0`)
	assert(evaluate(session, `${proof}.flush()`) === expected, "Actual menu native formatting queues exact markup")
	menu(expected.indexOf("bar"), expected.indexOf("bar") + 3)
	choose("Cut")
	wait(session, `${proof}.captured()===5&&${proof}.written()==='bar'`)
	evaluate(session, `${proof}.dispose()`)
	evaluate(session, `${proof}.settleWrite()`)
	assert(evaluate(session, "document.body.dataset.closed==='true'&&!document.querySelector('alkalye-editor')"), "Disposal interrupts pending menu Cut before native deletion")
} finally { browser(session, "close") }
